"""One frozen light event: source/capture qualification, never an image-quality oracle."""
import hashlib,json,struct,traceback
from pathlib import Path
import numpy as np
from native_log_check import scan
ROOT=Path(__file__).resolve().parents[2];P=ROOT/'LocalComparison/Review/local-light-move-response-01';N=P/'native-window';D=N/'run'
load=lambda p:json.loads(p.read_text(encoding='utf-8-sig'))
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
def xyz(v):return [v[k] for k in ['x','y','z']]
def point(row,pos):return row['positionType']['w']==1 and xyz(row['positionType'])==pos and row['directionRange']['w']==8
r={'sourceQualified':False,'visualResponse':'Requires actual image review; pixel differences alone do not establish causality','physicalTruthProven':False,'performanceAccepted':False}
try:
    contract=load(P/'input.json')
    for key in ['positionA','positionB']:contract[key]=[struct.unpack('<f',struct.pack('<f',value))[0] for value in contract[key]]
    result=load(D/'light-move-result.json');end=load(N/'native-exit.json');event=load(D/'event.json');r.update(result=result,nativeExit=end,event=event)
    hits=scan(N/'player.log');r['nativeLogHits']=hits
    assert not hits and end['exitCode']==0 and not end['watchdogTerminated'],'Native failure'
    assert result['status']=='collected' and result['errors']==0 and result['actualCameraFrames']==180 and result['captures']==5 and result['moved'] and result['normalQuitObserved'],'Incomplete capture'
    assert result['api']=='Direct3D12' and result['view']=='IndirectOnly' and not result['debugBuild'] and result['testCaptureDisabled'] and result['auditProbeDisabled'],'Configuration/source isolation'
    assert result['lastEngine']-result['firstEngine']==179 and event['logical']==120 and event['engine']==result['firstEngine']+120 and event['previousRenderEngine']==event['engine']-1,'Event/camera frame mismatch'
    assert xyz(event['from'])==contract['positionA'] and xyz(event['to'])==contract['positionB']
    frames={};lights={};pixels={};r['images']=[]
    for n in contract['captures']:
        f=load(D/f'frame-{n:04d}.json');l=load(D/f'lights-{n:04d}.json');frames[n]=f;lights[n]=l
        assert f['captured'] and f['logicalFrame']==l['logical']==n and f['engineFrame']==l['engine']==result['firstEngine']+n,'Capture/snapshot frame mismatch'
        assert l['lamp']['instance']==event['lampInstance'] and xyz(l['lamp']['position'])==contract['positionA' if n<120 else 'positionB'],'Lamp source mismatch'
        a=np.fromfile(D/f['raw'],dtype='<f4');assert a.size==640*360*4 and np.isfinite(a).all(),'Nonfinite/wrong HDR'
        pixels[n]=a.reshape(360,640,4)[:,:,:3]
        png=(D/f['png']).read_bytes();assert png[:8]==b'\x89PNG\r\n\x1a\n' and struct.unpack('>II',png[16:24])==(640,360)
        assert struct.unpack('<I',(D/f['exr']).read_bytes()[:4])[0]==20000630
        selected=[x for x in l['selected'] if point(x,contract['positionA' if n<120 else 'positionB'])];assert len(selected)==1,'Lamp absent or ambiguous in selected table'
        d=l['diagnostics'];assert d['hardwareBackend'] and d['hardwareMeshes']>0 and d['readyCards']>0 and d['hardwareMaterialUnknownRecords']==0,'Expected native preparation absent'
        r['images'].append({'logical':n,'engine':l['engine'],'gi':l['gi'],'selectedCount':l['count'],'changeCount':len(l['changes']),'regionCount':len(l['regions']),
            'meanLuminance':f['meanLuminance'],'maxRgb':f['maxRgb'],'files':{f[k]:sha(D/f[k]) for k in ['raw','png','exr']}})
    old=next(x for x in lights[119]['selected'] if point(x,contract['positionA']));new=next(x for x in lights[120]['selected'] if point(x,contract['positionB']))
    assert old in lights[120]['changes'] and new in lights[120]['changes'],'Missing actual old/new light change records'
    r['oldNewLightChangesVerified']=True;r['eventDomains']=lights[120]['regions']
    for n,f in frames.items():
        assert all(f[k]==frames[119][k] for k in ['cameraPosition','cameraRotation','doorLocalPosition','projection']),'Fixed camera/door changed'
        for current,previous in zip(f['lights'],frames[119]['lights']):
            expected=dict(previous)
            if expected['name']==contract['lightName'] and n>=120:expected['position']=dict(zip(['x','y','z'],contract['positionB']))
            assert current==expected,'Other captured light state changed'
    r['descriptiveMeanAbsoluteRgbDelta']={f'{a}->{b}':float(np.abs(pixels[b]-pixels[a]).mean()) for a,b in [(90,119),(119,120),(119,135),(119,179)]}
    r['sourceQualified']=True
except Exception as e:r.update(error=str(e),traceback=traceback.format_exc())
out=N/'qualification-addendum.json';assert not out.exists(),'Preserve first qualification';out.write_text(json.dumps(r,indent=2)+'\n',encoding='utf-8')
print(json.dumps({k:v for k,v in r.items() if k not in ['eventDomains','images']},indent=2));raise SystemExit(0 if r['sourceQualified'] else 1)
