#pragma kernel CopyDepth
#pragma target 5.0
#pragma use_dxc
Texture2D<float> _Source;
RWStructuredBuffer<float> _DepthValues;
int2 _Size;
[numthreads(8,8,1)]
void CopyDepth(uint3 id:SV_DispatchThreadID)
{
    if(any(id.xy>=(uint2)_Size))return;
    _DepthValues[id.y*_Size.x+id.x]=_Source.Load(int3(id.xy,0));
}
