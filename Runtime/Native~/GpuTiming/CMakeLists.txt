cmake_minimum_required(VERSION 3.20)
project(HybridLumenGIGpuTiming LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The native GPU timing plugin supports Windows x64 only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(UNITY_PLUGIN_API_DIR "" CACHE PATH "Unity PluginAPI headers (not distributed with this package)")
foreach(header IUnityInterface.h IUnityGraphics.h IUnityGraphicsD3D12.h)
    if(NOT EXISTS "${UNITY_PLUGIN_API_DIR}/${header}")
        message(FATAL_ERROR "Missing ${header}; provide UNITY_PLUGIN_API_DIR or use Tests/Build-NativeGpuTiming.ps1")
    endif()
endforeach()
add_library(HybridLumenGIGpuTiming SHARED GpuTimingPlugin.cpp NativeAbi.h QueryPool.h)
target_include_directories(HybridLumenGIGpuTiming SYSTEM PRIVATE "${UNITY_PLUGIN_API_DIR}")
target_compile_definitions(HybridLumenGIGpuTiming PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(HybridLumenGIGpuTiming PRIVATE d3d12 dxgi)
add_executable(HLGIQueryPoolTests tests/QueryPoolTests.cpp QueryPool.h NativeAbi.h)
target_include_directories(HLGIQueryPoolTests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(HLGINativeTimingSmoke tests/NativeTimingSmoke.cpp NativeAbi.h)
target_include_directories(HLGINativeTimingSmoke PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(HLGINativeTimingSmoke SYSTEM PRIVATE "${UNITY_PLUGIN_API_DIR}")
target_compile_definitions(HLGINativeTimingSmoke PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(HLGINativeTimingSmoke PRIVATE d3d12 dxgi)
if(MSVC)
    target_compile_options(HybridLumenGIGpuTiming PRIVATE /W4 /WX /permissive- /utf-8)
    target_compile_options(HLGIQueryPoolTests PRIVATE /W4 /WX /permissive- /utf-8)
    target_compile_options(HLGINativeTimingSmoke PRIVATE /W4 /WX /permissive- /utf-8)
endif()
