    static Pass Prepare(RenderGraph graph,params Batch.Source[] sources)
    {Check(Pass.TryCreate(graph,sources,1L<<30,out var pass,out var reason),"prepare "+reason);return pass;}
    static Batch OwnedBatch(Pass p)=>(Batch)typeof(Pass).GetField("batch",BindingFlags.Instance|BindingFlags.NonPublic).GetValue(p);
    static int ResidentPasses=>((ICollection)typeof(Pass).GetField("Resident",BindingFlags.Static|BindingFlags.NonPublic).GetValue(null)).Count;
    static void Clean()
    {
        Pass.PollRetired();
        Check(Batch.TotalAllocatedBytes==0&&Batch.TotalMaterialBytes==0&&Batch.ResidentBatchCount==0&&
            Pool.ResidentBatchCount==0&&GraphicsBuffer.Live==0&&RTHandle.Live==0&&ResidentPasses==0,"all test resources released");
        Check(RTHandles.OwnedAllocations==0,"RenderTexture ownership overload never selected");
    }
    static void Complete(Pass p,CommandBuffer cmd)
    {
        if(p.State==Batch.BatchState.Prepared){p.Dispose();Clean();return;}
        cmd.ThrowAt=cmd.ThrowBeforeAt=-1;cmd.OnCall=null;
        if(p.State!=Batch.BatchState.SealedAwaitingFence&&p.State!=Batch.BatchState.Released)
            Check(p.Finish(cmd,out var why),"Finish "+why);
        cmd.NextFence.Complete=true;Pass.PollRetired();p.Dispose();Clean();
    }
    static void RecoverForTestOnly(Pass p,CommandBuffer cmd)
    {
        // Pass intentionally has a one-shot Finish. This reflection is ONLY to clean up
        // injected permanent quarantine between test cases, not a production recovery API.
        cmd.ThrowAt=cmd.ThrowBeforeAt=-1;cmd.OnCall=null;cmd.NextFence=new FenceSignal();
        Check(OwnedBatch(p).SealAfterConsumers(cmd,out var reason),"test-only same-command final fence "+reason);
        cmd.NextFence.Complete=true;Pass.PollRetired();p.Dispose();Clean();
    }
    static void Record(Pass p,CommandBuffer cmd)
    {Check(p.RecordAndBind(cmd,new ComputeShader(),11,13,out var reason),"record/both binds "+reason);}
    static void ImportsAndFrozenBindings()
    {
        scenarios++;var graph=new RenderGraph();var f=new Fixture(submesh:3);var original=f.Freeze();var input=new[]{original};
        var pass=Prepare(graph,input);input[0]=default;
        Check(pass.Buffers.Count==4&&graph.ImportedBuffers.Count==4&&pass.SourceTextures.Count==0,"four constant resources, no source images");
        Check(graph.ImportedBuffers.Select(x=>x.Buffer.Stride).SequenceEqual(new[]{240,32,64,16}),"material/binding/metadata/texel import order and strides");
        for(int i=0;i<4;i++)Check(!graph.ImportedBuffers[i].ForceRelease&&ReferenceEquals(pass.Buffers[i].Buffer,graph.ImportedBuffers[i].Buffer),"graph borrows exact buffer "+i);
        Check(pass.ByteCount==352&&pass.State==Batch.BatchState.Prepared&&pass.MatchesSources(new[]{f.Freeze()}),"prepared owns copied sources, not caller array");
        var cmd=new CommandBuffer();Record(pass,cmd);
        Check(cmd.Copies.Count==3&&cmd.Dispatches.Count==0,"constant dummy initialized, tables uploaded without texture dispatch");
        Check(cmd.Copies.OfType<Record[]>().Single()[0].identity.x>0&&cmd.Copies.OfType<Binding[]>().Single()[0].source.w==3,"actual record and submesh bind upload");
        Check(cmd.Bound.Count==8&&cmd.Bound.Take(4).All(b=>b.Kernel==11)&&cmd.Bound.Skip(4).All(b=>b.Kernel==13),"both kernels receive every resource before success");
        foreach(int kernel in new[]{11,13})
            Check(cmd.Bound.Where(b=>b.Kernel==kernel).Select(b=>b.Buffer.Stride).OrderBy(x=>x).SequenceEqual(new[]{16,32,64,240}),"four bindings per consumer");
        Check(cmd.Events.All(x=>!x.StartsWith("fence:")),"no early retirement fence");
        cmd.Events.Add("consumer-cache");cmd.Events.Add("consumer-trace");
        Check(pass.Finish(cmd,out _),"finish after both actual caller dispatch sites");
        Check(cmd.Events.Last().StartsWith("fence:")&&cmd.Events.IndexOf("consumer-trace")==cmd.Events.Count-2,"final fence follows consumers");
        Check(!pass.RecordAndBind(cmd,new(),11,13,out _)&&!pass.MatchesSources(new[]{original})&&!pass.Finish(cmd,out _),"finished cannot be consumed again");
        for(int i=0;i<100;i++)Pass.PollRetired();Check(GraphicsBuffer.Live==4&&Batch.TotalAllocatedBytes==352,"pending fence does not imply submission/completion");
        cmd.NextFence.Complete=true;Pass.PollRetired();Clean();
    }

        {
            int at=RTHandles.BorrowedAllocations+index;if(returnNull)RTHandles.NullAt=at;else RTHandles.FailAt=at;
            Check(!Pass.TryCreate(new(),new[]{source},1L<<30,out _,out _),"wrapper create null/exception cleanup");

        }
        for(int i=1;i<=4;i++)
        {GraphicsBuffer.FailAllocationAt=GraphicsBuffer.Allocations+i;Check(!Pass.TryCreate(new(),new[]{source},1L<<30,out _,out _),"native buffer allocation failure");GraphicsBuffer.FailAllocationAt=-1;Clean();}
        Check(!Pass.TryCreate(null,new[]{source},1L<<30,out _,out _)&&!Pass.TryCreate(new(),null,1L<<30,out _,out _),"missing graph/sources rejects");Clean();
    }

    static void EveryRecordingFailure()
    {
        scenarios++;var source=new Fixture(Texture()).Freeze();var baseline=Prepare(new(),source);var bc=new CommandBuffer();Record(baseline,bc);
        int allCalls=bc.Calls,firstBind=bc.Events.FindIndex(e=>e.StartsWith("bind:11:"))+1,secondBind=bc.Events.FindIndex(e=>e.StartsWith("bind:13:"))+1;
        Check(firstBind>0&&secondBind==firstBind+4&&allCalls==secondBind+3,"measured independent two-kernel boundary");Complete(baseline,bc);
        foreach(bool before in new[]{false,true})for(int call=1;call<=allCalls;call++)
        {
            var p=Prepare(new(),source);var c=new CommandBuffer();if(before)c.ThrowBeforeAt=call;else c.ThrowAt=call;
            bool allowConsumers=p.RecordAndBind(c,new(),11,13,out _);if(allowConsumers)c.Events.Add("unsafe-consumer");
            Check(!allowConsumers&&!c.Events.Contains("unsafe-consumer")&&p.State==Batch.BatchState.Quarantined,"any record/bind failure forbids both consumers "+call+" before="+before);
            Check(c.Bound.Count(b=>b.Kernel==13)<4,"second consumer cannot have fully successful resources after a failed append");
            long bytes=Batch.TotalAllocatedBytes;p.Dispose();Pass.PollRetired();Check(bytes>0&&Batch.TotalAllocatedBytes==bytes,"unknown append cannot be treated as cancellation");
            c.ThrowAt=c.ThrowBeforeAt=-1;Check(p.Finish(c,out _),"finally can fence failed/partially recorded pass");
            Check(c.Events.Last().StartsWith("fence:"),"failure still finalizes after partial append");c.NextFence.Complete=true;Pass.PollRetired();Clean();
        }
        foreach(int kernel in new[]{-1,int.MinValue})
        {var p=Prepare(new(),source);var c=new CommandBuffer();Check(!p.RecordAndBind(c,new(),kernel,13,out _)&&!p.RecordAndBind(c,new(),11,kernel,out _)&&c.Calls==0,"invalid kernel no record");p.Dispose();Clean();}
        var noShader=Prepare(new(),source);var cn=new CommandBuffer();Check(!noShader.RecordAndBind(cn,null,11,13,out _)&&cn.Calls==0,"null compute no append");noShader.Dispose();Clean();
        var duplicate=Prepare(new(),source);var cd=new CommandBuffer();Record(duplicate,cd);int calls=cd.Calls;
        Check(!duplicate.RecordAndBind(cd,new(),11,13,out _)&&cd.Calls==calls,"second record rejected without appending");Complete(duplicate,cd);
    }
    static void FinishFailuresAndAbandonment()
    {
        scenarios++;var source=new Fixture(Texture()).Freeze();
        var never=Prepare(new(),source);long bytes=Batch.TotalAllocatedBytes;for(int i=0;i<1000;i++)Pass.PollRetired();
        Check(never.State==Batch.BatchState.Prepared&&Batch.TotalAllocatedBytes==bytes,"abandoned graph is retained until owning feature cancels, no age release");
        never.Dispose();var late=new CommandBuffer();Check(!never.RecordAndBind(late,new(),11,13,out _)&&late.Calls==0,"late graph after owner Dispose cannot append");Clean();
        var pristine=Prepare(new(),source);var cp=new CommandBuffer();Check(pristine.Finish(cp,out _)&&cp.Calls==0,"pristine finally uses no unnecessary fence");Clean();
        foreach(bool before in new[]{false,true})
        {
            var p=Prepare(new(),source);var c=new CommandBuffer();Record(p,c);
            if(before)c.ThrowBeforeAt=c.Calls+1;else c.ThrowAt=c.Calls+1;
            Check(!p.Finish(c,out _)&&p.State==Batch.BatchState.Quarantined,"fence recording exception quarantines");
            long held=Batch.TotalAllocatedBytes;Check(held>0&&RTHandle.Live==1&&!p.Finish(c,out _),"one-shot Finish cannot invent retry evidence");
            c.NextFence.Complete=true;for(int i=0;i<3;i++)Pass.PollRetired();Check(Batch.TotalAllocatedBytes==held,"failed append doesn't establish valid fence even when adapter signal completes");
            RecoverForTestOnly(p,c);
        }
        var invalid=Prepare(new(),source);var ci=new CommandBuffer();Record(invalid,ci);ci.NextFence=null;
        Check(!invalid.Finish(ci,out _)&&Batch.TotalAllocatedBytes>0&&RTHandle.Live==1,"default fence cannot free borrowed graph resources");RecoverForTestOnly(invalid,ci);
        var foreign=Prepare(new(),source);var cf=new CommandBuffer();Record(foreign,cf);
        Check(!foreign.Finish(new(),out _)&&Batch.TotalAllocatedBytes>0,"another command buffer fence not retirement evidence");RecoverForTestOnly(foreign,cf);
        var readFailure=Prepare(new(),source);var cr=new CommandBuffer();Record(readFailure,cr);Check(readFailure.Finish(cr,out _),"valid pending seal");
        cr.NextFence.Throw=true;Pass.PollRetired();Check(readFailure.State==Batch.BatchState.Quarantined&&RTHandle.Live==1,"fence passed getter failure retains resources");
        cr.NextFence.Throw=false;cr.NextFence.Complete=true;Pass.PollRetired();Check(RTHandle.Live==1,"later boolean alone doesn't restore invalidated evidence");RecoverForTestOnly(readFailure,cr);
        var canceled=Prepare(new(),source);var cc=new CommandBuffer();Record(canceled,cc);canceled.Dispose();Check(Batch.TotalAllocatedBytes>0,"Dispose between record and finally cannot free");
        Check(canceled.Finish(cc,out _),"Finish after canceled recording is retirement only");Check(!canceled.RecordAndBind(cc,new(),11,13,out _),"retirement doesn't reenable consume");
        cc.NextFence.Complete=true;Pass.PollRetired();Clean();
    }

        Check(initial.SourceTextures.Count==2&&!initial.ReusesTextureResidency&&initial.TextureUploadByteCount>0,"first pass imports both real upload sources");
        Check(initial.Finish(first,out _),"first actual pass final fence");first.NextFence.Complete=true;Pass.PollRetired();

        AsyncGPUReadback.CompleteNext(data);Pass.PollRetired();

    {
