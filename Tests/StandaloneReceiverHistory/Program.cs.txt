    private static void Revalidation()
    {
        var camera=new Camera();var registry=new Registry();var a=R();var b=R();var original=P(registry,camera,a,b);
        Check(registry.IsCurrent(original,D(camera,a,b),Budget),"unchanged sources remain current");
        Check(registry.IsCurrent(original,D(camera,b,a),Budget),"enumeration reorder alone is not a source change");
        Check(!registry.IsCurrent(original,D(camera,a),Budget),"removed source revokes source set");
        Check(!registry.IsCurrent(original,D(camera,a,b,R()),Budget),"new source revokes source set");
        Check(!registry.IsCurrent(original,D(camera,a,a),Budget),"duplicate cannot replace another current renderer");
        Check(!registry.IsCurrent(original,D(new Camera(),a,b),Budget),"different camera rejected despite same sources");
        camera.cullingMask=3;Check(!registry.IsCurrent(original,D(camera,a,b),Budget),"camera layer context change detected");camera.cullingMask=-1;
        var selected=new Discovery.Snapshot(camera,new []{new Discovery.Candidate(a),new Discovery.Candidate(b)},layers:3);
        Check(!registry.IsCurrent(original,selected,Budget),"selected-layer context change detected");
        var newlyUnknown=new Discovery.Snapshot(camera,new []{new Discovery.Candidate(a),new Discovery.Candidate(b)},unknown:true);
        Check(!registry.IsCurrent(original,newlyUnknown,Budget),"unknown-source global status change revokes source proof even with same candidate references");
        camera.cullingMatrix=Matrix4x4.Translate(1);Check(!registry.IsCurrent(original,D(camera,a,b),Budget),"culling matrix change detected");camera.cullingMatrix=Matrix4x4.identity;
        a.Matrix=Matrix4x4.Translate(2);Check(!registry.IsCurrent(original,D(camera,a,b),Budget),"current source transform change invalidates pending capture");a.Matrix=Matrix4x4.identity;
        a.Bounds=new Bounds{Center=9,Extent=1};Check(!registry.IsCurrent(original,D(camera,a,b),Budget),"source bounds change detected");a.Bounds=new Bounds{Extent=1};
        var material=a.Materials[0];a.Materials[0]=new Material();Check(!registry.IsCurrent(original,D(camera,a,b),Budget),"same-size material slot replacement detected");a.Materials[0]=material;
        a.Mpb[0]++;Check(!registry.IsCurrent(original,D(camera,a,b),Budget),"MPB changed after preparation rejected");a.Mpb[0]--;
        material.Value++;Check(!registry.IsCurrent(original,D(camera,a,b),Budget),"material scalar changed after preparation rejected");material.Value--;
        a.Mesh.Content="unnotified change";Check(!registry.IsCurrent(original,D(camera,a,b),Budget),"same-reference unnotified mesh edit rejected before capture");
        var fresh=P(registry,camera,a,b);Check(registry.IsCurrent(fresh,D(camera,a,b),Budget),"new prepared version restores current state");
        Check(!registry.IsCurrent(fresh,D(camera,a,b),0),"revalidation byte exhaustion fails closed");
        Check(!new Registry().IsCurrent(fresh,D(camera,a,b),Budget),"foreign registry snapshot rejected");

        a.Mesh.Destroyed=true;Check(!registry.IsCurrent(fresh,D(camera,a,b),Budget),"destroyed mesh cannot retain valid source receipt");a.Mesh.Destroyed=false;

    private static void RandomFrameOracle()
    {
        var random=new Random(0x484C47);var camera=new Camera();var registry=new Registry();var renderer=R();var life=new Lifecycle();var history=new History();
        int frame=0,geometryVersion=0,materialVersion=0;bool previousSucceeded=false;int previousFrame=-1,previousGeometry=-1,previousMaterial=-1,previousContext=0;
        for(int step=0;step<400;step++)
        {
            frame+=random.Next(4)==0?2:random.Next(5)==0?0:1;
            int edit=random.Next(8);if(edit==0){geometryVersion++;renderer.Mesh.Content="content-"+geometryVersion;}else if(edit==1){materialVersion++;renderer.Mpb[0]=materialVersion;}
            renderer.Matrix=Matrix4x4.Translate(step%17);
            int contextKey=random.Next(10)==0?1:0;
            var context=new Lifecycle.FrameContext(100,60,50,30,contextKey,0,100,60);
            var snapshot=P(registry,camera,renderer);var token=life.BeginFrame(frame,context);
            Check(history.Begin(token,snapshot),"random valid source frame begins "+step);
            bool expected=previousSucceeded&&frame==previousFrame+1&&contextKey==previousContext&&geometryVersion==previousGeometry&&materialVersion==previousMaterial;
            bool actual=history.TryGetPrevious(snapshot.Records[0],out var old);
            Check(actual==expected,"independent successful f-1/version/context oracle "+step);
            if(actual) Check(old.Token==snapshot.Records[0].Token,"random matched source token "+step);
            int outcome=random.Next(5);
            if(outcome<=1)
            {
                Check(history.MarkCaptureRecorded(token,true)&&life.Complete(token)&&history.Complete(token),"random successful receipt "+step);
                previousSucceeded=true;previousFrame=frame;previousGeometry=geometryVersion;previousMaterial=materialVersion;previousContext=contextKey;
            }
            else if(outcome==2)
            {
                history.MarkCaptureRecorded(token,false);life.Complete(token);Check(!history.Complete(token),"random source failure cannot publish "+step);previousSucceeded=false;
            }
            else if(outcome==3)
            {
                history.MarkCaptureRecorded(token,true);Check(!history.Complete(token),"random missing screen completion cannot publish "+step);previousSucceeded=false;
            }
            else
            {
                history.MarkCaptureRecorded(token,true);life.Complete(token);history.Complete(token);life.Invalidate();
                Check(!history.IsPublished,"random revoked screen receipt "+step);previousSucceeded=false;
            }
            Check(history.IsPublished==previousSucceeded,"random publication oracle "+step);
        }
    }

        Check(history.TryGetPrevious(same.Records[0],out var previous)&&ReferenceEquals(previous,first.Records[0]),"unchanged adjacent skin pose accepts f-1 history");
        Publish(history,life,t1);
        bone.localToWorldMatrix=Matrix4x4.Translate(.25f);
        Check(!registry.IsCurrent(same,D(camera,skin),Budget),"bone change after preparation rejects pending source capture");
        var changed=P(registry,camera,skin);var t2=life.BeginFrame(102,Context);history.Begin(t2,changed);
        Check(changed.Records[0].Token==same.Records[0].Token&&changed.Records[0].GeometryGeneration!=same.Records[0].GeometryGeneration,"real bone motion keeps owner identity but changes geometry generation");
        Check(!history.TryGetPrevious(changed.Records[0],out _),"moving skin cannot reuse different-pose history");Publish(history,life,t2);
        var stopped=P(registry,camera,skin);var t3=life.BeginFrame(103,Context);history.Begin(t3,stopped);
        Check(history.TryGetPrevious(stopped.Records[0],out previous)&&ReferenceEquals(previous,changed.Records[0]),"the next unchanged pose restores adjacent-frame history");Publish(history,life,t3);
        skin.Cloth=new Cloth();
        Check(!registry.IsCurrent(stopped,D(camera,skin),Budget),"adding Cloth revokes an already prepared skin source");
        var cloth=P(registry,camera,skin);Check(cloth.Records.Count==0&&cloth.HasUnknownSources,"Cloth remains unsupported instead of inheriting stable bone history");
    }
