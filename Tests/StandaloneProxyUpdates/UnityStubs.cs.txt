    public static class UnityApiGuard
    {
        public static readonly int MainThread=Environment.CurrentManagedThreadId;
        public static int Calls;
        public static void Access()
        {
            System.Threading.Interlocked.Increment(ref Calls);
            if(Environment.CurrentManagedThreadId!=MainThread) throw new InvalidOperationException("Unity API accessed outside the main thread");
        }
    }

    public class Component:Object
    {
        public GameObject gameObject=new();
        public readonly Dictionary<Type,object> Components=new();
        public T GetComponent<T>() where T:class
        { UnityApiGuard.Access();return this is T self?self:Components.TryGetValue(typeof(T),out object value)?value as T:null; }
    }

    public struct Vector3 : IEquatable<Vector3>
    {
        public float x,y,z;
        public Vector3(float x,float y,float z){this.x=x;this.y=y;this.z=z;}
        public static Vector3 one=>new(1,1,1);
        public float sqrMagnitude=>x*x+y*y+z*z;
        public static Vector3 operator -(Vector3 a,Vector3 b)=>new(a.x-b.x,a.y-b.y,a.z-b.z);
        public bool Equals(Vector3 b)=>x==b.x&&y==b.y&&z==b.z;
        public override bool Equals(object b)=>b is Vector3 other&&Equals(other);
        public override int GetHashCode()=>HashCode.Combine(x,y,z);
        public static bool operator ==(Vector3 a,Vector3 b)=>a.Equals(b);
        public static bool operator !=(Vector3 a,Vector3 b)=>!a.Equals(b);
    }
    public struct Bounds : IEquatable<Bounds>
    {
        public Vector3 center,size;
        public Bounds(Vector3 center,Vector3 size){this.center=center;this.size=size;}
        public bool Equals(Bounds b)=>center==b.center&&size==b.size;
        public override bool Equals(object b)=>b is Bounds other&&Equals(other);
        public override int GetHashCode()=>HashCode.Combine(center,size);
        public static bool operator ==(Bounds a,Bounds b)=>a.Equals(b);
        public static bool operator !=(Bounds a,Bounds b)=>!a.Equals(b);
    }
    public struct Matrix4x4 : IEquatable<Matrix4x4>
    {
        public N.Matrix4x4 value;
        public static Matrix4x4 identity=>new(){value=N.Matrix4x4.Identity};
        public static Matrix4x4 Translate(Vector3 p)=>new(){value=N.Matrix4x4.CreateTranslation(p.x,p.y,p.z)};
        public bool Equals(Matrix4x4 b)=>value.Equals(b.value);
        public override bool Equals(object b)=>b is Matrix4x4 other&&Equals(other);
        public override int GetHashCode()=>value.GetHashCode();
        public static bool operator ==(Matrix4x4 a,Matrix4x4 b)=>a.Equals(b);
        public static bool operator !=(Matrix4x4 a,Matrix4x4 b)=>!a.Equals(b);
    }
    public struct Color : IEquatable<Color>
    {
        public float r,g,b,a;
        public Color(float r,float g,float b,float a=1){this.r=r;this.g=g;this.b=b;this.a=a;}
        public static Color white=>new(1,1,1); public static Color black=>new(0,0,0);
        public Color linear=>new(L(r),L(g),L(b),a);
        static float L(float x)=>x<=.04045f?x/12.92f:MathF.Pow((x+.055f)/1.055f,2.4f);
        public static Color operator *(Color a,Color b)=>new(a.r*b.r,a.g*b.g,a.b*b.b,a.a*b.a);
        public static Color operator +(Color a,Color b)=>new(a.r+b.r,a.g+b.g,a.b+b.b,a.a+b.a);
        public bool Equals(Color x)=>r==x.r&&g==x.g&&b==x.b&&a==x.a;
        public override bool Equals(object x)=>x is Color other&&Equals(other);
        public override int GetHashCode()=>HashCode.Combine(r,g,b,a);
        public static bool operator ==(Color a,Color b)=>a.Equals(b);
        public static bool operator !=(Color a,Color b)=>!a.Equals(b);
    }

    {
        public readonly Dictionary<string,Color> Colors=new();
