    public sealed class ComputeBuffer
    {
        public readonly int count,stride;
        public readonly object[] data;
        public string name;
        public bool released;
        public int lastUploadCount;
        public ComputeBuffer(int count,int stride){this.count=count;this.stride=stride;data=new object[count];}
        public void SetData(Array values,int sourceIndex,int destinationIndex,int length)
        {
            if(released)throw new InvalidOperationException("Uploading released buffer.");
            if(sourceIndex<0||destinationIndex<0||length<0||destinationIndex+length>count)throw new ArgumentOutOfRangeException();
            for(int i=0;i<length;i++)data[destinationIndex+i]=values.GetValue(sourceIndex+i);
            lastUploadCount=length;
        }
        public void Release(){released=true;}
    }
