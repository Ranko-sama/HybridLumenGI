    // Bounded source-wiring checks only, not a C# parser or an execution model.
    // Every boundary must exist exactly once before a relative order is tested.
    static int UniqueSourceIndex(string source,string marker)
    {
        int index=source.IndexOf(marker,StringComparison.Ordinal);
        Check(index>=0,"Missing source boundary: "+marker);
        Check(source.IndexOf(marker,index+marker.Length,StringComparison.Ordinal)<0,"Ambiguous source boundary: "+marker);
        return index;
    }
    static string SourceSection(string source,string start,string end)
    {
        int first=UniqueSourceIndex(source,start),last=UniqueSourceIndex(source,end);
        Check(last>first,"Reversed source boundaries: "+start);
        return source.Substring(first+start.Length,last-first-start.Length);
    }
    static string CompactWiring(string source) => Regex.Replace(
        Regex.Replace(source,@"//[^\r\n]*|/\*[\s\S]*?\*/",string.Empty),@"\s+",string.Empty);
    static void CheckTimedRender(string section,string phase,string renderOperation,string expectedTail)
    {
        string code=CompactWiring(section);
        string begin="d.frameTiming.Begin(HybridLumenGIGpuFrameTiming.Phase."+phase+",ctx.cmd);";
        string end="d.frameTiming.End(HybridLumenGIGpuFrameTiming.Phase."+phase+",ctx.cmd);";
        if(phase=="Transport") { begin=begin.Replace("ctx.cmd","cmd");end=end.Replace("ctx.cmd","cmd"); }
        string failure="catch{d.state.InvalidateFailedRender();throw;}";
        UniqueSourceIndex(code,begin);UniqueSourceIndex(code,end);UniqueSourceIndex(code,failure);
        // The real render body alone is in the history-invalidating try/catch;
        // safe timing Begin is before it and End is its finally cleanup.
        var match=Regex.Match(code,@"^\{"+Regex.Escape(begin)+@"try\{(?<body>[\s\S]*)\}"+
            Regex.Escape(failure)+@"finally\{"+Regex.Escape(end)+@"\}(?<tail>[\s\S]*)$");
        Check(match.Success,phase+" must keep rendering failure and timing cleanup in their own scopes");
        string body=match.Groups["body"].Value;
        UniqueSourceIndex(body,renderOperation);
        Check(!body.Contains("CompleteScreenHistory("),phase+" cannot publish before rendering and cleanup finish");
        Check(match.Groups["tail"].Value==CompactWiring(expectedTail),phase+" publication must follow cleanup without another history-invalidating guard");
    }

    // Actual Execute callback, with dependency spies only. This tests control

    sealed class CommandBuffer { }
    static class HybridLumenGIGpuFrameTiming { internal enum Phase { Transport } }
    sealed class TimingSpy
    {
        readonly List<string> events;
        internal TimingSpy(List<string> events){this.events=events;}
        internal void Begin(HybridLumenGIGpuFrameTiming.Phase phase,CommandBuffer cmd){events.Add("begin");}
        internal void End(HybridLumenGIGpuFrameTiming.Phase phase,CommandBuffer cmd){events.Add("end");}
    }
    sealed class StateSpy
    {
        readonly List<string> events;
        internal string hardwareMaterialStatus="ready";
        internal StateSpy(List<string> events){this.events=events;}
        internal void InvalidateFailedRender(){events.Add("invalidate");}
    }
    sealed class MaterialPassSpy
    {
        readonly List<string> events;
        internal bool fail,throwOnFinish;
        internal CommandBuffer expected;
        internal MaterialPassSpy(List<string> events){this.events=events;}
        internal bool Finish(CommandBuffer cmd,out string reason)
        {
            Check(ReferenceEquals(expected,cmd),"Final material fence uses the exact consumer command buffer");
            events.Add("finish");reason="retirement unproven";
            if(throwOnFinish)throw new InvalidOperationException("finish failure");
            return !fail;
        }
    }
    sealed class Work
    { internal TimingSpy frameTiming;internal StateSpy state;internal MaterialPassSpy hitMaterials; }
    sealed class TransportAdapter
    {
        internal readonly List<string> events=new();
        internal readonly HashSet<MaterialPassSpy> materialPasses=new();
        internal readonly CommandBuffer command=new();
        internal bool failRender;
        internal Action duringRender;
        internal Work CreateWork(bool materials)
        {
            var d=new Work{state=new StateSpy(events),frameTiming=new TimingSpy(events)};
            if(materials){d.hitMaterials=new MaterialPassSpy(events){expected=command};materialPasses.Add(d.hitMaterials);}
            return d;
        }
        void ExecuteTransport(Work d,CommandBuffer cmd)
        {
            Check(ReferenceEquals(command,cmd),"Render consumes the captured command buffer");
            events.Add("render");duringRender?.Invoke();
            if(failRender)throw new InvalidOperationException("render failure");
        }
        internal void Run(Work d)=>Execute(d,command);
        __TRANSPORT_EXECUTE_METHOD__
    }
    static void CheckActualTransportCleanup()
    {
        foreach(bool hasMaterials in new[]{false,true})
        foreach(bool renderFails in new[]{false,true})
        foreach(bool fenceFails in new[]{false,true})
        {
            var owner=new TransportAdapter{failRender=renderFails};var work=owner.CreateWork(hasMaterials);
            if(hasMaterials)work.hitMaterials.fail=fenceFails;
            Exception caught=null;try{owner.Run(work);}catch(Exception ex){caught=ex;}
            var expected=new List<string>{"begin","render"};
            if(renderFails)expected.Add("invalidate");if(hasMaterials)expected.Add("finish");expected.Add("end");
            Check(owner.events.SequenceEqual(expected),"Execute cleanup order independent of rendering and fence outcomes");
            Check((caught!=null)==renderFails,"Retirement failure must not turn successful rendering into an exception");
            if(renderFails)Check(caught.Message=="render failure","Render failure is preserved");
            Check(owner.materialPasses.Count==0,"Executed pass is removed from the owner; batch retains its own fence obligations");
            Check(work.state.hardwareMaterialStatus==(hasMaterials&&fenceFails?"retirement unproven":"ready"),"Unproven retirement is visible");
        }
        {
            var owner=new TransportAdapter();var work=owner.CreateWork(true);work.hitMaterials.throwOnFinish=true;
            Exception caught=null;try{owner.Run(work);}catch(Exception ex){caught=ex;}
            Check(caught?.Message=="finish failure","Unexpected cleanup exception stays observable");
            Check(owner.events.SequenceEqual(new[]{"begin","render","finish","end"}),"Timing cleanup still runs if material cleanup throws");
            Check(owner.materialPasses.Contains(work.hitMaterials),"Unexpected cleanup failure keeps the owner reference for teardown");
        }
        {
            var owner=new TransportAdapter();var work=owner.CreateWork(true);var captured=work.hitMaterials;
            MaterialPassSpy later=null;
            owner.duringRender=()=>{later=new MaterialPassSpy(owner.events);owner.materialPasses.Add(later);};
            owner.Run(work);
            Check(!owner.materialPasses.Contains(captured)&&owner.materialPasses.Contains(later),"Finishing this graph must not cancel another graph's batch");
        }
    }

    // Only a call/order spy for the actual Feature helpers below. The real

    sealed class ReceiverHistorySpy
    {
        readonly History lifecycle;
        internal int invalidations;
        internal readonly List<History.FrameToken> completions=new();
        internal ReceiverHistorySpy(History lifecycle) { this.lifecycle=lifecycle; }
        internal void Invalidate()
        {
            Check(!lifecycle.HasCompletedHistory && lifecycle.LastCompletedFrame==-1,
                "Feature must revoke the lifecycle before notifying receiver invalidation");
            invalidations++;
        }
        internal void Complete(History.FrameToken token)
        {
            Check(lifecycle.IsCompleted(token) && !lifecycle.IsCurrent(token),
                "Receiver snapshot publication must follow successful completion of this exact lifecycle receipt");
            completions.Add(token);
        }
    }

    // A log oracle labels image provenance by the last effective event. It uses
    // neither runtime token sequence numbers nor its private pending/valid state.
    sealed class Oracle
    {
        public enum Kind { Begin, Publish, Reset }
        public readonly List<(Kind kind,int ticket,int frame)> Events = new();
        public bool Current(int ticket) => Events.Count>0 && Events[^1].kind==Kind.Begin && Events[^1].ticket==ticket;
        public bool Reuse(int frame) => frame>=0 && Events.Count>0 && Events[^1].kind==Kind.Publish &&
            (long)Events[^1].frame+1==frame;
        public void Begin(int ticket,int frame) => Events.Add((frame>=0?Kind.Begin:Kind.Reset,ticket,frame));
        public bool Complete(int ticket)
        {
            if(!Current(ticket)) return false;
            Events.Add((Kind.Publish,ticket,Events[^1].frame));return true;
        }
        public void Reset() => Events.Add((Kind.Reset,-1,-1));
        public int CompletedFrame => Events.Count>0 && Events[^1].kind==Kind.Publish?Events[^1].frame:-1;
        public bool Completed(int ticket) => Events.Count>0 && Events[^1].kind==Kind.Publish && Events[^1].ticket==ticket;
    }

    // Fixtures carry independent image-coordinate labels: the oracle never calls
    // FrameContext.Equals/IsValid or inspects runtime token/context fields.
    sealed class ContextOracle
    {
        public readonly List<(string kind,int ticket,int frame,string context)> Events=new();
        public bool Current(int ticket) => Events.Count>0 && Events[^1].kind=="begin" && Events[^1].ticket==ticket;
        public bool Reuse(int frame,string context) => frame>=0 && context!=null && Events.Count>0 &&
            Events[^1].kind=="publish" && (long)Events[^1].frame+1==frame && Events[^1].context==context;
        public void Begin(int ticket,int frame,string context) => Events.Add((frame>=0&&context!=null?"begin":"reset",ticket,frame,context));
        public bool Complete(int ticket)
        {
            if(!Current(ticket)) return false;
            var prior=Events[^1];Events.Add(("publish",ticket,prior.frame,prior.context));return true;
        }
        public void Reset() => Events.Add(("reset",-1,-1,null));
        public int CompletedFrame => Events.Count>0 && Events[^1].kind=="publish"?Events[^1].frame:-1;
        public bool Completed(int ticket) => Events.Count>0 && Events[^1].kind=="publish" && Events[^1].ticket==ticket;
    }
