// Executes the actual runtime sampler on D3D12 WARP, not a Unity host or DXR test.
#define WIN32_LEAN_AND_MEAN
#define NOMINMAX
#include <Windows.h>
#include <d3d12.h>
#include <dxgi1_4.h>
#include <wrl/client.h>
#include <cstdint>
#include <cstring>
#include <filesystem>
#include <fstream>
#include <iostream>
#include <stdexcept>
#include <vector>
using Microsoft::WRL::ComPtr;
namespace {
void Hr(HRESULT h,const char* operation){if(FAILED(h)){std::cerr<<operation<<" HRESULT="<<std::hex<<static_cast<unsigned long>(h)<<std::dec<<'\n';throw std::runtime_error(operation);}}
std::vector<std::uint8_t> Read(const std::filesystem::path& path){
    std::ifstream stream(path,std::ios::binary|std::ios::ate);if(!stream)throw std::runtime_error("open input");
    auto size=stream.tellg();if(size<=0||size>64*1024*1024)throw std::runtime_error("input size");
    std::vector<std::uint8_t> result(static_cast<std::size_t>(size));stream.seekg(0);stream.read(reinterpret_cast<char*>(result.data()),size);
    if(!stream)throw std::runtime_error("read input");return result;
}
ComPtr<ID3D12Resource> Buffer(ID3D12Device* device,UINT64 bytes,D3D12_HEAP_TYPE type,D3D12_RESOURCE_STATES state,D3D12_RESOURCE_FLAGS flags=D3D12_RESOURCE_FLAG_NONE){
    D3D12_HEAP_PROPERTIES heap{};heap.Type=type;heap.CreationNodeMask=1;heap.VisibleNodeMask=1;
    D3D12_RESOURCE_DESC desc{};desc.Dimension=D3D12_RESOURCE_DIMENSION_BUFFER;desc.Width=bytes;desc.Height=1;desc.DepthOrArraySize=1;desc.MipLevels=1;
    desc.SampleDesc.Count=1;desc.Layout=D3D12_TEXTURE_LAYOUT_ROW_MAJOR;desc.Flags=flags;
    ComPtr<ID3D12Resource> result;Hr(device->CreateCommittedResource(&heap,D3D12_HEAP_FLAG_NONE,&desc,state,nullptr,IID_PPV_ARGS(&result)),"CreateCommittedResource");return result;
}
ComPtr<ID3D12Resource> Upload(ID3D12Device* device,const std::uint8_t* source,UINT64 bytes){
    auto result=Buffer(device,bytes,D3D12_HEAP_TYPE_UPLOAD,D3D12_RESOURCE_STATE_GENERIC_READ);
    void* mapped=nullptr;D3D12_RANGE empty{};Hr(result->Map(0,&empty,&mapped),"Map upload");
    std::memcpy(mapped,source,static_cast<std::size_t>(bytes));result->Unmap(0,nullptr);return result;
}
}
int wmain(int argc,wchar_t** argv){
    try{
        if(argc!=4)throw std::runtime_error("Usage: RayTexturesWarp shader.dxil inputs.bin outputs.bin");
        auto shader=Read(argv[1]),input=Read(argv[2]);if(input.size()<12)throw std::runtime_error("missing counts");
        std::uint32_t counts[3]{};std::memcpy(counts,input.data(),12);
        const UINT count=counts[0],metadataCount=counts[1],texelCount=counts[2];
        constexpr UINT InputStride=32,OutputStride=32,MetadataStride=64,TexelStride=16;
        const UINT64 metadataBytes=static_cast<UINT64>(metadataCount)*MetadataStride,texelBytes=static_cast<UINT64>(texelCount)*TexelStride,inputBytes=static_cast<UINT64>(count)*InputStride;
        if(count==0||count>100000||metadataCount==0||texelCount==0||input.size()!=12+metadataBytes+texelBytes+inputBytes)throw std::runtime_error("case ABI/size");
        ComPtr<IDXGIFactory4> factory;ComPtr<IDXGIAdapter> adapter;ComPtr<ID3D12Device> device;
        HRESULT h=CreateDXGIFactory2(0,IID_PPV_ARGS(&factory));if(FAILED(h)){std::cerr<<"WARP environment unavailable: factory\n";return 2;}
        h=factory->EnumWarpAdapter(IID_PPV_ARGS(&adapter));if(FAILED(h)){std::cerr<<"WARP environment unavailable: adapter\n";return 2;}
        h=D3D12CreateDevice(adapter.Get(),D3D_FEATURE_LEVEL_11_0,IID_PPV_ARGS(&device));if(FAILED(h)){std::cerr<<"WARP environment unavailable: D3D12 device\n";return 2;}
        DXGI_ADAPTER_DESC adapterDescription{};Hr(adapter->GetDesc(&adapterDescription),"GetDesc");
        D3D12_COMMAND_QUEUE_DESC queueDesc{};queueDesc.Type=D3D12_COMMAND_LIST_TYPE_DIRECT;
        ComPtr<ID3D12CommandQueue> queue;Hr(device->CreateCommandQueue(&queueDesc,IID_PPV_ARGS(&queue)),"CreateCommandQueue");
        D3D12_DESCRIPTOR_RANGE ranges[2]{};
        ranges[0].RangeType=D3D12_DESCRIPTOR_RANGE_TYPE_SRV;ranges[0].NumDescriptors=3;
        ranges[1].RangeType=D3D12_DESCRIPTOR_RANGE_TYPE_UAV;ranges[1].NumDescriptors=1;
        D3D12_ROOT_PARAMETER parameters[3]{};
        for(UINT i=0;i<2;i++){parameters[i].ParameterType=D3D12_ROOT_PARAMETER_TYPE_DESCRIPTOR_TABLE;parameters[i].DescriptorTable.NumDescriptorRanges=1;parameters[i].DescriptorTable.pDescriptorRanges=&ranges[i];}
        parameters[2].ParameterType=D3D12_ROOT_PARAMETER_TYPE_32BIT_CONSTANTS;parameters[2].Constants.Num32BitValues=1;
        D3D12_ROOT_SIGNATURE_DESC signatureDesc{};signatureDesc.NumParameters=3;signatureDesc.pParameters=parameters;
        ComPtr<ID3DBlob> signatureBytes,diagnostics;Hr(D3D12SerializeRootSignature(&signatureDesc,D3D_ROOT_SIGNATURE_VERSION_1,&signatureBytes,&diagnostics),"SerializeRootSignature");
        ComPtr<ID3D12RootSignature> signature;Hr(device->CreateRootSignature(0,signatureBytes->GetBufferPointer(),signatureBytes->GetBufferSize(),IID_PPV_ARGS(&signature)),"CreateRootSignature");
        D3D12_COMPUTE_PIPELINE_STATE_DESC pipelineDesc{};pipelineDesc.pRootSignature=signature.Get();pipelineDesc.CS={shader.data(),shader.size()};
        ComPtr<ID3D12PipelineState> pipeline;Hr(device->CreateComputePipelineState(&pipelineDesc,IID_PPV_ARGS(&pipeline)),"CreateComputePipelineState actual sampler DXIL");
        auto metadata=Upload(device.Get(),input.data()+12,metadataBytes);
        auto texels=Upload(device.Get(),input.data()+12+static_cast<std::size_t>(metadataBytes),texelBytes);
        auto inputs=Upload(device.Get(),input.data()+12+static_cast<std::size_t>(metadataBytes+texelBytes),inputBytes);
        auto outputs=Buffer(device.Get(),static_cast<UINT64>(OutputStride)*count,D3D12_HEAP_TYPE_DEFAULT,D3D12_RESOURCE_STATE_UNORDERED_ACCESS,D3D12_RESOURCE_FLAG_ALLOW_UNORDERED_ACCESS);
        auto readback=Buffer(device.Get(),static_cast<UINT64>(OutputStride)*count,D3D12_HEAP_TYPE_READBACK,D3D12_RESOURCE_STATE_COPY_DEST);
        // Real descriptors are required: root SRVs alone do not carry element
        // counts for the sampler's StructuredBuffer.GetDimensions validation.
        D3D12_DESCRIPTOR_HEAP_DESC heapDesc{};heapDesc.Type=D3D12_DESCRIPTOR_HEAP_TYPE_CBV_SRV_UAV;heapDesc.NumDescriptors=4;heapDesc.Flags=D3D12_DESCRIPTOR_HEAP_FLAG_SHADER_VISIBLE;
        ComPtr<ID3D12DescriptorHeap> heap;Hr(device->CreateDescriptorHeap(&heapDesc,IID_PPV_ARGS(&heap)),"CreateDescriptorHeap");
        const UINT increment=device->GetDescriptorHandleIncrementSize(heapDesc.Type);
        auto cpu=heap->GetCPUDescriptorHandleForHeapStart();
        ID3D12Resource* resources[]={metadata.Get(),texels.Get(),inputs.Get()};const UINT elements[]={metadataCount,texelCount,count},strides[]={MetadataStride,TexelStride,InputStride};
        for(UINT i=0;i<3;i++){
            D3D12_SHADER_RESOURCE_VIEW_DESC view{};view.ViewDimension=D3D12_SRV_DIMENSION_BUFFER;view.Shader4ComponentMapping=D3D12_DEFAULT_SHADER_4_COMPONENT_MAPPING;
            view.Buffer.NumElements=elements[i];view.Buffer.StructureByteStride=strides[i];device->CreateShaderResourceView(resources[i],&view,cpu);cpu.ptr+=increment;
        }
        D3D12_UNORDERED_ACCESS_VIEW_DESC view{};view.ViewDimension=D3D12_UAV_DIMENSION_BUFFER;view.Buffer.NumElements=count;view.Buffer.StructureByteStride=OutputStride;
        device->CreateUnorderedAccessView(outputs.Get(),nullptr,&view,cpu);
        ComPtr<ID3D12CommandAllocator> allocator;Hr(device->CreateCommandAllocator(D3D12_COMMAND_LIST_TYPE_DIRECT,IID_PPV_ARGS(&allocator)),"CreateCommandAllocator");
        ComPtr<ID3D12GraphicsCommandList> list;Hr(device->CreateCommandList(0,D3D12_COMMAND_LIST_TYPE_DIRECT,allocator.Get(),pipeline.Get(),IID_PPV_ARGS(&list)),"CreateCommandList");
        ID3D12DescriptorHeap* heaps[]={heap.Get()};list->SetDescriptorHeaps(1,heaps);list->SetComputeRootSignature(signature.Get());
        auto gpu=heap->GetGPUDescriptorHandleForHeapStart();list->SetComputeRootDescriptorTable(0,gpu);gpu.ptr+=3ull*increment;list->SetComputeRootDescriptorTable(1,gpu);list->SetComputeRoot32BitConstant(2,count,0);
        list->Dispatch((count+63)/64,1,1);
        D3D12_RESOURCE_BARRIER barriers[2]{};barriers[0].Type=D3D12_RESOURCE_BARRIER_TYPE_UAV;barriers[0].UAV.pResource=outputs.Get();
        barriers[1].Type=D3D12_RESOURCE_BARRIER_TYPE_TRANSITION;barriers[1].Transition.pResource=outputs.Get();barriers[1].Transition.StateBefore=D3D12_RESOURCE_STATE_UNORDERED_ACCESS;barriers[1].Transition.StateAfter=D3D12_RESOURCE_STATE_COPY_SOURCE;barriers[1].Transition.Subresource=D3D12_RESOURCE_BARRIER_ALL_SUBRESOURCES;
        list->ResourceBarrier(2,barriers);list->CopyResource(readback.Get(),outputs.Get());Hr(list->Close(),"Close");ID3D12CommandList* lists[]={list.Get()};queue->ExecuteCommandLists(1,lists);
        ComPtr<ID3D12Fence> fence;Hr(device->CreateFence(0,D3D12_FENCE_FLAG_NONE,IID_PPV_ARGS(&fence)),"CreateFence");Hr(queue->Signal(fence.Get(),1),"Signal");
        HANDLE event=CreateEventW(nullptr,FALSE,FALSE,nullptr);if(!event)throw std::runtime_error("CreateEvent");h=fence->SetEventOnCompletion(1,event);if(FAILED(h)){CloseHandle(event);Hr(h,"SetEventOnCompletion");}
        DWORD wait=WaitForSingleObject(event,30000);CloseHandle(event);if(wait!=WAIT_OBJECT_0)throw std::runtime_error("WARP dispatch timeout");Hr(device->GetDeviceRemovedReason(),"DeviceRemovedReason");
        void* mapped=nullptr;D3D12_RANGE range{0,static_cast<SIZE_T>(OutputStride)*count},empty{};Hr(readback->Map(0,&range,&mapped),"Map readback");
        std::ofstream result(std::filesystem::path(argv[3]),std::ios::binary);result.write(static_cast<const char*>(mapped),static_cast<std::streamsize>(OutputStride)*count);readback->Unmap(0,&empty);if(!result)throw std::runtime_error("write output");result.close();
        std::cout<<"Actual RayTextures HLSL executed on D3D12 WARP: "<<count<<" cases, metadata="<<metadataCount<<" texels="<<texelCount<<"; adapterVendor="<<adapterDescription.VendorId<<" adapterDevice="<<adapterDescription.DeviceId<<". No Unity or physical GPU.\n";return 0;
    }catch(const std::exception& error){std::cerr<<error.what()<<'\n';return 1;}
}
