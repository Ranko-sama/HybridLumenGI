    {
        public int renderQueue=2000;static readonly Dictionary<string,int> ids=new();

    public class Bag
    {
        public readonly Dictionary<int,object> Values=new(); public int Writes;
        public bool HasProperty(int id)=>Values.ContainsKey(id); public bool HasProperty(string s)=>HasProperty(Shader.PropertyToID(s));
        public bool HasFloat(int id)=>Values.TryGetValue(id,out var o)&&o is float;
        public bool HasVector(int id)=>Values.TryGetValue(id,out var o)&&(o is Vector4||o is Color);
        public bool HasTexture(int id)=>Values.TryGetValue(id,out var o)&&(o==null||o is Texture);
        public float GetFloat(int id)=>Values.TryGetValue(id,out var o)&&o is float f?f:0;public float GetFloat(string s)=>GetFloat(Shader.PropertyToID(s));
        public Color GetColor(int id)=>Values.TryGetValue(id,out var o)&&o is Color c?c:default;
        public Vector4 GetVector(int id)=>Values.TryGetValue(id,out var o)?o is Vector4 v?v:o is Color c?(Vector4)c:default:default;
        public Texture GetTexture(int id)=>Values.TryGetValue(id,out var o)?o as Texture:null;
        public void Set(string s,object o){Values[Shader.PropertyToID(s)]=o;Writes++;}
        public void Clear(){Values.Clear();Writes++;}
    }
