    static (Source source,Renderer renderer,Material mat) Fixture()
    {
        var m=new Material(new Shader{name="Universal Render Pipeline/Lit"});

        foreach(var n in Colors)m.Set(n,new Color(.2f,.3f,.4f));foreach(var n in Textures)m.Set(n,new Texture2D());

    static Source.Snapshot Capture(Source s,Renderer r,Material m,int i=0)
    {Check(s.TryCapture(r,i,m,out var result,out var reason)&&result!=null&&reason==null,"capture: "+reason);return result;}
    static void IdentityAndImmutability()
    {
        scenarios++;var(s,r,m)=Fixture();m.shaderKeywords=new[]{"_EMISSION","_NORMALMAP"};var before=Capture(s,r,m);
        int writes=m.Data.Writes+r.Block.Writes;Check(s.IsCurrent(r,0,before),"immediate revalidation");Check(m.Data.Writes+r.Block.Writes==writes,"no original object writes");
        var reordered=new[]{"_NORMALMAP","_EMISSION"};m.shaderKeywords=reordered;var after=Capture(s,r,m);Check(before.Equals(after)&&before.GetHashCode()==after.GetHashCode(),"keyword order is not identity");
        reordered[0]="_PARALLAXMAP";Check(!s.IsCurrent(r,0,before)&&before.Equals(after),"keyword array copied, old snapshots remain immutable");
        m.shaderKeywords=new[]{"_NORMALMAP","_EMISSION"};m.Set("_Metallic",.8f);var changed=Capture(s,r,m);Check(before.GetHashCode()==changed.GetHashCode()&&!before.Equals(changed),"deliberately equal hashes cannot alias material state");
        m.Set("_Metallic",0f);Check(s.IsCurrent(r,0,before),"returning to exact observable source is equal");
        var other=Fixture().mat;other.FakeInstanceId=m.FakeInstanceId;r.Materials[0]=other;Check(!s.IsCurrent(r,0,before)&&!s.TryCapture(r,0,m,out _,out _),"same instance-id substitute is not same source");r.Materials[0]=m;
        var oldShader=m.shader;m.shader=new Shader{name=oldShader.name,FakeInstanceId=oldShader.FakeInstanceId};Check(!s.IsCurrent(r,0,before),"same named shader is distinct by reference");m.shader=oldShader;
        r.Materials.Add(m);Check(!s.IsCurrent(r,0,before),"material binding layout change detected");r.Materials.RemoveAt(1);
        var otherRenderer=new Renderer();otherRenderer.Materials.Add(m);Check(!before.Equals(Capture(s,otherRenderer,m)),"ordinary renderer identity preserved without proxy");
        r.enabled=false;Check(s.IsCurrent(r,0,before),"material observation independent of enabled state");
        Check(typeof(Source.Snapshot).GetProperties(BindingFlags.Public|BindingFlags.Instance).All(p=>!p.CanWrite&&!p.PropertyType.IsArray),"public snapshot has no mutable arrays/setters");
        var values=new[]{new Source.Value(1,1,new Vector4(1,2,3,4))};var texels=Array.Empty<Source.TextureValue>();var names=new[]{"KEY"};var passes=new[]{"Pass"};var enabled=new[]{true};
        var internalA=new Source.Snapshot(r,m,0,1,values,texels,true,names,passes,enabled);var internalB=new Source.Snapshot(r,m,0,1,values,texels,true,names,passes,enabled);
        values[0]=new Source.Value(9,3,new Vector4(4,3,2,1));names[0]="MUTATED";passes[0]="MUTATED";enabled[0]=false;
        Check(internalA.Equals(internalB),"same-assembly constructor input arrays cannot mutate frozen snapshots");
    }

    {

        own.Set("_Smoothness",.6f);var scoped=Capture(s,r,m);own.Clear();Check(!s.IsCurrent(r,0,scoped),"empty per-material restores renderer block");

    static void TextureStates()
    {
        scenarios++;
        foreach(string name in Textures)
        {
            var(s,r,m)=Fixture();var old=Capture(s,r,m);var t=m.GetTexture(Shader.PropertyToID(name));t.updateCount++;
            Check(!s.IsCurrent(r,0,old),"source texture content counter "+name);old=Capture(s,r,m);m.Set(name,new Texture2D());Check(!s.IsCurrent(r,0,old),"source texture identity "+name);
            r.Block.Set(name,t);old=Capture(s,r,m);t.updateCount++;Check(!s.IsCurrent(r,0,old),"override texture counter "+name);
            old=Capture(s,r,m);r.Block.Set(name,null);Check(!s.IsCurrent(r,0,old),"explicit null texture override "+name);Check(s.IsCurrent(r,0,Capture(s,r,m)),"null override round trip");
        }
        var(f,rr,mm)=Fixture();var tex=mm.GetTexture(Shader.PropertyToID("_BaseMap"));
        Action<Texture>[] changes={t=>t.filterMode++,t=>t.wrapModeU++,t=>t.wrapModeV++,t=>t.wrapModeW++,t=>t.anisoLevel++,t=>t.mipMapBias+=.25f,t=>t.width++,t=>t.height++,t=>t.mipmapCount++,t=>t.graphicsFormat++,t=>t.dimension++};
        foreach(var change in changes){var old=Capture(f,rr,mm);change(tex);Check(!f.IsCurrent(rr,0,old),"sampler/layout mutation with unchanged updateCount");}
        mm.Set("_EmissionMap",new RenderTexture());var rt=Capture(f,rr,mm);Check(!rt.CanReuseHistory&&!f.IsCurrent(rr,0,rt),"GPU-written RenderTexture cannot prove reusable content");
        mm.Set("_EmissionMap",new CustomRenderTexture());Check(!Capture(f,rr,mm).CanReuseHistory,"custom RT remains unknown");
        mm.Set("_EmissionMap",new Texture2D());var unchanged=Capture(f,rr,mm);Check(f.IsCurrent(rr,0,unchanged),"CPU-observable static content is current; unreported GPU writes are an explicit external contract");
        tex.Destroyed=true;Check(!f.TryCapture(rr,0,mm,out _,out _),"destroyed texture not mistaken for unassigned texture");
    }
    static void RejectionsAndLateMutation()
    {

        r.Block.Set("_Surface",1f);Check(!s.TryCapture(r,0,m,out _,out _),"effective transparent uniform rejected");r.Block.Clear();

        m.Destroyed=true;Check(!s.IsCurrent(r,0,old),"destroyed source rejected");
    }

    static void FrozenHitValues()
    {
        scenarios++;var(s,r,m)=Fixture();

        m.Set("_BaseColor",new Color(.125f,-2,17.5f,.6f));m.Set("_SpecColor",new Color(.25f,.5f,.75f,.8f));
        m.Set("_EmissionColor",new Color(12000.25f,-4,.03125f,.3f));

        m.Offsets[Shader.PropertyToID("_BaseMap")]=new Vector2(3,-4);

        m.shaderKeywords=new[]{"_EMISSION","_NORMALMAP","_SPECULAR_SETUP","_ALPHATEST_ON","UNRECOGNIZED_FUTURE_KEYWORD"};

        m.Data.Clear();m.shaderKeywords[0]="MUTATED";m.shaderKeywords=Array.Empty<string>();m.Destroyed=true;

        foreach(var item in expected)
        {
            Check(input.TryGetFloat(item.name,out var scalar,out var origin)&&scalar==item.value&&origin==HybridLumenGIMaterialValueOrigin.Material,"every frozen scalar and source: "+item.name);
            Check(input.TryGetFloat(Shader.PropertyToID(item.name),out var byId,out var idOrigin)&&byId==scalar&&idOrigin==origin,"property-id access: "+item.name);
            string property=item.name=="_Cull"?"SourceCull":item.name.Substring(1);

        for(int i=0;i<expectedKeywords.Length;i++)Check(input.GetKeyword(i)==expectedKeywords[i]&&input.IsKeywordEnabled(expectedKeywords[i]),"frozen sorted keyword "+i);
        Check(!input.IsKeywordEnabled("MUTATED")&&!input.IsKeywordEnabled(null),"late keyword edit not visible");
        foreach(var type in new[]{typeof(HybridLumenGIHitMaterialData),typeof(HybridLumenGIFrozenTextureView),typeof(HybridLumenGIFrozenColor)})
            Check(type.GetProperties(BindingFlags.Public|BindingFlags.Instance).All(p=>!p.CanWrite&&!p.PropertyType.IsArray),"no public setters/arrays: "+type.Name);
    }

        }
    }
