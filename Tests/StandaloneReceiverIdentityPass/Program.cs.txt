    static int Id(string name)=>Shader.PropertyToID("_HLGI_Identity"+name);
    static HybridLumenGIMaterialSource.IdentityCoverageData Coverage(int cull=2,bool alpha=false,bool smooth=false,Texture tex=null)
        =>new HybridLumenGIMaterialSource.IdentityCoverageData{Supported=true,SourceCull=cull,BaseColorAlpha=.7f,Cutoff=.3f,AlphaTest=alpha,SmoothnessFromAlbedoAlpha=smooth,BaseMap=tex,BaseMapST=new Vector4(2,3,.2f,.4f)};
    static Frame Make(int count=1,RectInt[] rects=null)
    {
        var rows=new Frame.Row[count];var draws=new Frame.Draw[count];
        for(int i=0;i<count;i++){rows[i]=new Frame.Row((uint)(100+i));draws[i]=new Frame.Draw(new HybridLumenGIReceiverRegistry.Record{Mesh=new Mesh(),Submesh=i},i,Coverage(i%3,true,false,new Texture2D()));}
        return new Frame(rows,draws,rects??Array.Empty<RectInt>());
    }
    static (RenderGraph graph,Pass.Result result,List<bool> receipts) Record(Pass pass,Frame frame=null,Func<bool> validate=null,int fw=19,int fh=11,int gw=7,int gh=5,
        Action<CommandBuffer,int,bool> timing=null,Action invalidated=null)
    {
        var graph=new RenderGraph();var receipts=new List<bool>();
        var depth=graph.CreateTexture(new TextureDesc(fw,fh));var capture=graph.CreateTexture(new TextureDesc(fw,fh));var output=graph.CreateTexture(new TextureDesc(gw,gh));
        var result=pass.Record(graph,frame??Make(),depth,capture,output,fw,fh,gw,gh,Matrix4x4.identity,validate??(()=>true),v=>receipts.Add(v),timing,invalidated);
        return (graph,result,receipts);
    }
    static void CompleteFences(){foreach(var f in CommandBuffer.Fences){f.Throw=false;f.Passed=true;}RenderPipelineManager.EndFrame();}
    static RecordedCommand Resolve(CommandBuffer cmd)=>cmd.Commands.Last(x=>x.Name=="Dispatch"&&x.Kernel==2);
    static IEnumerable<DrawInfo> Draws(CommandBuffer cmd)=>cmd.Commands.Where(x=>x.Name=="Draw").Select(x=>(DrawInfo)x.Value);
    static void Basic()
    {
        scenarios++;var pass=new Pass();int validCalls=0;
        var frame=Make(3,new[]{new RectInt(-5,1,12,9),new RectInt(18,10,7,8),new RectInt(30,0,2,2)});
        var data=Record(pass,frame,()=>{validCalls++;return true;});
        Check(data.result.IsValid&&data.result.ObjectCount==3,"valid result count");
        Check(data.receipts.Count==0&&validCalls==0,"not published or validated at graph construction");
        Check(data.graph.Buffers.Count==2&&data.graph.Buffers[0].desc.count==209&&data.graph.Buffers[0].desc.stride==12,"uint3 full-size candidates");
        Check(data.graph.Buffers[1].desc.count==3&&data.graph.Buffers[1].desc.stride==80&&Marshal.SizeOf<Frame.Row>()==80,"80B row ABI");
        Check(data.graph.Textures.Last().desc.colorFormat==UnityEngine.Experimental.Rendering.GraphicsFormat.R8G8B8A8_UNorm,"single dummy RGBA8");
        Check(data.graph.Passes.Count==4&&data.graph.Passes.All(x=>!x.Cull),"four uncullable stages");
        var p=data.graph.Passes;
        Check(p[0].Dependencies.Count(x=>x.type=="buffer"&&x.flags==AccessFlags.Write)==2,"init writes both buffers");
        Check(p[0].Dependencies.Any(x=>x.type=="texture"&&x.id==data.graph.Textures[1].handle.Id&&x.flags==AccessFlags.Read),"clear explicitly follows capture depth producer");
        Check(p[1].Dependencies.Single().flags==AccessFlags.ReadWrite&&p[2].Dependencies.Any(x=>x.type=="buffer"&&x.flags==AccessFlags.ReadWrite),"mask/capture ordered RW");
        Check(p[2].Dependencies.Count(x=>x.type=="texture"&&x.flags==AccessFlags.Read)==2&&p[2].Global,"both depth dependencies and raster state");
        Check(p[3].Dependencies.Count(x=>x.type=="buffer"&&x.flags==AccessFlags.Read)==2&&p[3].Dependencies.Count(x=>x.type=="texture"&&x.flags==AccessFlags.Write)==1,"resolve consumes both buffers");
        Check(p[3].Global,"resolve permits final native timing event global state");
        var cmd=new CommandBuffer();
        data.graph.ExecutePass(0,cmd);data.graph.ExecutePass(1,cmd);
        Check(validCalls==0&&data.receipts.Count==0,"validation immediately before capture only");
        data.graph.ExecutePass(2,cmd);Check(validCalls==1&&data.receipts.Count==0,"capture is not final receipt");
        data.graph.ExecutePass(3,cmd);Check(data.receipts.SequenceEqual(new[]{true}),"one successful resolve receipt");
        var ds=Draws(cmd).ToArray();Check(ds.Length==3,"all frozen draws");
        Check(ds.Select(x=>x.Block).Distinct().Count()==3&&ds.Select(x=>x.Material).Distinct().Count()==1,"separate blocks immutable common material");
        for(int i=0;i<3;i++)
        {
            Check(ReferenceEquals(ds[i].Mesh,frame.Draws[i].Record.Mesh)&&ds[i].Submesh==i&&ds[i].Pass==i,"source submesh and cull pass");
            Check(ds[i].Values[Id("OwnerIndexPlusOne")] is int&& (int)ds[i].Values[Id("OwnerIndexPlusOne")]==i+1,"integer owner and 0-based rows");
            var size=(Vector4)ds[i].Values[Id("FullSize")];Check(size.x==19&&size.y==11,"raster float4 size contract");
            Check(ReferenceEquals(ds[i].Values[Id("BaseMap")],frame.Draws[i].Coverage.BaseMap),"frozen alpha texture");
            Check(ds[i].Values.ContainsKey(Id("CameraDepth"))&&ds[i].Values.ContainsKey(Id("CaptureDepth")),"per-draw depth bindings");
        }
        var upload=(GraphicsBuffer)cmd.Commands.Single(x=>x.Name=="Upload").Value;
        Check(upload.Uploaded.Length==3&&((Frame.Row[])upload.Uploaded)[2].keyLow==102,"actual row upload");
        var masks=cmd.Commands.Where(x=>x.Name=="Dispatch"&&x.Kernel==1).ToArray();
        Check(masks.Length==2,"empty clamped rectangle skipped");
        Check(((int[])masks[0].Parameters[Id("UnknownRect")]).SequenceEqual(new[]{0,1,7,10}),"first mask clamped min/max");
        Check(((int[])masks[1].Parameters[Id("UnknownRect")]).SequenceEqual(new[]{18,10,19,11}),"second mask clamped min/max");
        Check(((int[])masks[0].Value).SequenceEqual(new[]{1,2,1}),"mask dispatch rectangle extent");
        var resolve=Resolve(cmd);Check((int)resolve.Parameters[Id("ObjectCount")]==3&&((int[])resolve.Parameters[Id("GiSize")]).SequenceEqual(new[]{7,5}),"resolve int ABI");
        Check(cmd.Commands.Single(x=>x.Name=="Uav").Index==1,"pixel u1");
        Check(cmd.Commands.Count(x=>x.Name=="ClearUav")==2,"UAV clear before/after draw");
        Check(cmd.Commands.Last().Name=="Fence"&&cmd.Commands.Last().Index==7,"fence after resolve includes all stages");
        Check(pass.PendingFrames==1,"unpassed fence retains lease");
        Material m=ds[0].Material;pass.Dispose();Check(!m.Destroyed,"dispose must not release before fence");
        CompleteFences();Check(m.Destroyed,"endFrame collects completed private material");
    }
    static void RejectAndFailure()
    {
        scenarios++;
        foreach(int mode in new[]{0,1,2,3})
        {
            var pass=new Pass();var data=Record(pass,Make(2),()=>mode==1?throw new Exception("source changed"):mode!=0);
            var cmd=new CommandBuffer{ThrowDrawAt=mode==2?1:-1,ThrowDispatchKernel=mode==3?0:-1};data.graph.Execute(cmd);
            Check(data.receipts.SequenceEqual(new[]{false}),"failure rejects receipt");
            Check((int)Resolve(cmd).Parameters[Id("ObjectCount")]==0,"failure resolves zeros via zero count");
            Check(cmd.Commands.Any(x=>x.Name=="ClearUav"),"failure still clears UAV");
            if(mode<2)Check(!Draws(cmd).Any(),"invalid source never draws");
            if(mode==2)Check(Draws(cmd).Count()==1&&pass.PendingFrames==1,"partial draw stays fenced");
            pass.Dispose();CompleteFences();
        }
    }

    static readonly string[] AllEvents={"0B","0E","1B","1E","2B","2E","3B","3E"};
    sealed class Timer
    {
        public readonly List<string> Events=new List<string>();
        public readonly List<CommandBuffer> Buffers=new List<CommandBuffer>();
        public int Invalidations,ThrowStage=-1;public bool ThrowBegin,ThrowInvalidate;
        public void Boundary(CommandBuffer cmd,int stage,bool begin)
        {
            Events.Add(stage+(begin?"B":"E"));Buffers.Add(cmd);
            cmd.Commands.Add(new RecordedCommand{Name=begin?"TimingBegin":"TimingEnd",Index=stage});
            if(stage==ThrowStage&&begin==ThrowBegin)throw new Exception("timing observer failure after marker attempt");
        }
        public void Invalidate(){Invalidations++;if(ThrowInvalidate)throw new Exception("timing invalidation observer failure");}
    }
    static int Marker(CommandBuffer cmd,int stage,bool begin)=>cmd.Commands.FindIndex(x=>x.Name==(begin?"TimingBegin":"TimingEnd")&&x.Index==stage);
    static string[] BodyCommands(CommandBuffer cmd)=>cmd.Commands.Where(x=>!x.Name.StartsWith("Timing")).Select(x=>x.Name+":"+x.Kernel+":"+x.Index).ToArray();
    static void TimingOrderAndDependencies()
    {
        scenarios++;var pass=new Pass();var timer=new Timer();var data=Record(pass,Make(2,new[]{new RectInt(1,2,3,4)}),timing:timer.Boundary,invalidated:timer.Invalidate);
        Check(timer.Events.Count==0&&timer.Invalidations==0,"constructing a complete graph does not emit GPU timing");
        var graph=data.graph;
        // Independent dataflow assertions identify the actual resource handles, not pass names.
        int candidates=graph.Buffers[0].handle.Id,objects=graph.Buffers[1].handle.Id,capture=graph.Textures[1].handle.Id,output=graph.Textures[2].handle.Id;
        Check(graph.Passes[0].Dependencies.Any(d=>d.type=="texture"&&d.id==capture&&d.flags==AccessFlags.Read),"capture-depth producer is required by first timed callback");
        for(int i=0;i<4;i++)Check(graph.Passes[i].Dependencies.Any(d=>d.type=="buffer"&&d.id==candidates&&d.flags==(i==0?AccessFlags.Write:i==3?AccessFlags.Read:AccessFlags.ReadWrite)),"candidate dataflow orders all four stages");
        Check(graph.Passes[0].Dependencies.Any(d=>d.type=="buffer"&&d.id==objects&&d.flags==AccessFlags.Write)&&graph.Passes[3].Dependencies.Any(d=>d.type=="buffer"&&d.id==objects&&d.flags==AccessFlags.Read),"row upload precedes resolve");
        Check(graph.Passes[3].Dependencies.Any(d=>d.type=="texture"&&d.id==output&&d.flags==AccessFlags.Write)&&graph.Passes[3].Global,"resolve produces declared downstream texture and permits end event");
        var cmd=new CommandBuffer();graph.Execute(cmd);
        Check(timer.Events.SequenceEqual(AllEvents)&&timer.Invalidations==0,"exact independent begin/end stage order");
        Check(cmd.Commands.First().Name=="TimingBegin"&&cmd.Commands.First().Index==0,"begin clear before any command including upload");
        int upload=cmd.Commands.FindIndex(x=>x.Name=="Upload");int clear=cmd.Commands.FindIndex(x=>x.Name=="Dispatch"&&x.Kernel==0);
        Check(Marker(cmd,0,true)<upload&&upload<clear&&clear<Marker(cmd,0,false),"upload and clear enclosed in stage zero");
        int mask=cmd.Commands.FindIndex(x=>x.Name=="Dispatch"&&x.Kernel==1);
        Check(Marker(cmd,1,true)<mask&&mask<Marker(cmd,1,false),"unknown mask enclosed");
        int firstUav=cmd.Commands.FindIndex(x=>x.Name=="ClearUav"),lastUav=cmd.Commands.FindLastIndex(x=>x.Name=="ClearUav");
        Check(Marker(cmd,2,true)<firstUav&&lastUav<Marker(cmd,2,false),"capture includes both UAV cleanup calls");
        int resolve=cmd.Commands.FindIndex(x=>x.Name=="Dispatch"&&x.Kernel==2),fence=cmd.Commands.FindIndex(x=>x.Name=="Fence");
        Check(Marker(cmd,3,true)<resolve&&resolve<fence&&fence<Marker(cmd,3,false),"resolve includes dispatch and retirement fence");
        Check(data.receipts.SequenceEqual(new[]{true}),"timed success preserves identity receipt");pass.Dispose();CompleteFences();
    }

    }

    static void TimingGraphFailures()
    {
        scenarios++;
        // Every allocation/builder boundary, including after the final resolve callback exists.
        for(int fault=0;fault<11;fault++)
        {
            var pass=new Pass();var graph=new RenderGraph();var d=graph.CreateTexture(new TextureDesc(19,11));var c=graph.CreateTexture(new TextureDesc(19,11));var o=graph.CreateTexture(new TextureDesc(7,5));
            graph.ThrowCreateBufferAt=fault<2?fault:-1;graph.ThrowCreateTextureAt=fault==2?3:-1;
            graph.ThrowAddAt=fault>=3&&fault<7?fault-3:-1;graph.ThrowBuilderDisposeAt=fault>=7?fault-7:-1;
            var timer=new Timer();var receipts=new List<bool>();
            var result=pass.Record(graph,Make(),d,c,o,19,11,7,5,Matrix4x4.identity,()=>true,v=>receipts.Add(v),timer.Boundary,timer.Invalidate);
            Check(!result.IsValid&&receipts.SequenceEqual(new[]{false})&&timer.Invalidations==1,"construction fault rejects timing and identity immediately "+fault);
            Check(timer.Events.Count==0,"graph construction never starts timing "+fault);
