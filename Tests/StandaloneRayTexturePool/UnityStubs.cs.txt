    public sealed class GraphicsBuffer : IDisposable
    {
        public enum Target { Structured }
        public static int Allocations, FailAllocationAt = -1, Live;
        public readonly int Count, Stride;
        public int count=>Count;public int stride=>Stride;
        public bool Released, ThrowDispose;
        public GraphicsBuffer(Target target, int count, int stride)
        {
            if (++Allocations == FailAllocationAt) throw new InvalidOperationException("allocation failure");
            Count=count; Stride=stride; Live++;
        }
        public void Dispose()
        {
            if (ThrowDispose) throw new InvalidOperationException("dispose failure");
            if (Released) throw new InvalidOperationException("double dispose");
            Released=true; Live--;
        }
    }

    public sealed class CommandBuffer
    {
        public readonly List<string> Events=new List<string>();
        public readonly List<Array> Copies=new List<Array>();
        public readonly List<UnityEngine.GraphicsBuffer> Buffers=new List<UnityEngine.GraphicsBuffer>();
        public readonly Dictionary<int,int> Integers=new Dictionary<int,int>();
        public readonly List<(int Kernel,int X,int Y,int Z,Dictionary<int,int> Parameters)> Dispatches=new();
        public int Calls, ThrowAt=-1;
        private void Call(string name,UnityEngine.GraphicsBuffer buffer)
        {
            Events.Add(name); Buffers.Add(buffer);
            if(buffer!=null&&buffer.Released) throw new InvalidOperationException("use after release");
            if(++Calls==ThrowAt) throw new InvalidOperationException("unknown native recording result");
        }
        public void SetBufferData(UnityEngine.GraphicsBuffer buffer,Array source)
        { Call("metadata-initialize",buffer); Copies.Add((Array)source.Clone()); }
        public readonly List<(int Source,int Destination,int Count)> CopyRanges=new();
        public void SetBufferData(UnityEngine.GraphicsBuffer buffer,Array source,int sourceStart,int destinationStart,int count)
        { Call("metadata-range",buffer); Copies.Add((Array)source.Clone()); CopyRanges.Add((sourceStart,destinationStart,count)); }
        public void SetComputeBufferParam(UnityEngine.ComputeShader shader,int kernel,int name,UnityEngine.GraphicsBuffer buffer)
        { Call("bind:"+name,buffer); }
        public void SetComputeIntParam(UnityEngine.ComputeShader shader,int name,int value)
        { Call("int:"+name,null);Integers[name]=value; }
        public void SetComputeTextureParam(UnityEngine.ComputeShader shader,int kernel,int name,UnityEngine.Texture source)
        { Call("texture:"+kernel,null); }
        public void DispatchCompute(UnityEngine.ComputeShader shader,int kernel,int x,int y,int z)
        { Call("dispatch:"+kernel,null);Dispatches.Add((kernel,x,y,z,new Dictionary<int,int>(Integers))); }
    }

    public readonly struct AsyncGPUReadbackRequest
    {
        private readonly System.Array data;
        public bool hasError {get;}
        public AsyncGPUReadbackRequest(System.Array data,bool error){this.data=data;hasError=error;}
        public T[] GetData<T>() where T:struct => (T[])data;
    }
    public static class AsyncGPUReadback
    {
        public static bool ThrowDuringRequest;
        public static readonly System.Collections.Generic.Queue<(UnityEngine.GraphicsBuffer Buffer,System.Action<AsyncGPUReadbackRequest> Callback)> Pending=new();
        public static AsyncGPUReadbackRequest Request(UnityEngine.GraphicsBuffer buffer,System.Action<AsyncGPUReadbackRequest> callback)
        {
            if(buffer.Released)throw new System.Exception("readback after release");
            Pending.Enqueue((buffer,callback));
            if(ThrowDuringRequest)throw new System.Exception("unknown native request result");
            return default;
        }
        public static void CompleteNext(System.Array data,bool error=false)
        {var next=Pending.Dequeue();if(next.Buffer.Released)throw new System.Exception("pending readback buffer released");next.Callback(new AsyncGPUReadbackRequest(data,error));}
    }
