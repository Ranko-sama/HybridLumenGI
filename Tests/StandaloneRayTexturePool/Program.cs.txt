        Check(lease.State==Pool.LeaseState.Released,"released only safe");

    static void ImporterDispatchOracle()
    {
        scenarios++;using var pool=new Pool();var shader=new ComputeShader();var importer=new HybridLumenGIRayTextureImporter(shader);
        int expectedDispatches=0,expectedTextures=0;
        var sizes=new[]{(1,1,1),(13,9,4),(257,3,9),(17,41,3),(16384,1,15),(1,16384,15),(7,9,1),(64,64,7)};
        foreach(var size in sizes)
        {
            var source=Source(size.Item1,size.Item2,size.Item3);var view=Observe(source);var lease=Prepare(pool,view);
            var command=new CommandBuffer();Check(pool.RecordUploads(command,lease,importer.Record,out var error),"actual importer "+error);
            Check(importer.LastRejection==null&&command.Dispatches.Count==size.Item3+2,"Begin+every logical mip+Finish");
            Check(command.Dispatches[0].Kernel==0&&command.Dispatches[0].X==1&&command.Dispatches[0].Y==1&&command.Dispatches.Last().Kernel==2,"dispatch stage order");
            int width=size.Item1,height=size.Item2;
            for(int mip=0;mip<size.Item3;mip++)
            {
                var dispatch=command.Dispatches[mip+1];int x=0,y=0;
                for(int pixel=0;pixel<width;pixel+=8)x++;
                for(int pixel=0;pixel<height;pixel+=8)y++;
                Check(dispatch.Kernel==1&&dispatch.X==x&&dispatch.Y==y&&dispatch.Z==1,"independent tiles cover mip exactly once");
                Check(dispatch.Parameters[Shader.PropertyToID("_HLGI_CopyTextureMip")]==mip,"absolute logical mip parameter");
                Check((x-1)*8<width&&x*8>=width&&(y-1)*8<height&&y*8>=height,"NPOT/tail dispatch envelope");
                width=Math.Max(1,width/2);height=Math.Max(1,height/2);
            }
            foreach(var dispatch in command.Dispatches)
            {
                Check(unchecked((uint)dispatch.Parameters[Shader.PropertyToID("_HLGI_CopyTextureGeneration")])==lease.Generation,"generation preserved");
                Check(dispatch.Parameters[Shader.PropertyToID("_HLGI_CopyTextureCapacity")]==lease.TexelCapacity,"full-chain reserved capacity");
            }
            Check(command.Copies.Count==1&&((HybridLumenGIRayTextureMetadata[])command.Copies[0]).All(x=>x.flags==0),"CPU never publishes ready even on importer success");
            expectedDispatches+=size.Item3+2;expectedTextures++;
            Check(importer.RecordedDispatches==expectedDispatches&&importer.RecordedTextures==expectedTextures,"recorded counters reflect successful API calls only");
            Finish(pool,lease);
        }
        Check(shader.Lookups==3,"all kernel identities cached only after complete lookup");
        var generationField=typeof(Pool).GetField("lastGeneration",BindingFlags.NonPublic|BindingFlags.Static);
        uint previous=(uint)generationField.GetValue(null);generationField.SetValue(null,0x80000000u);
        var high=Prepare(pool,Observe(Source()));var highCommand=new CommandBuffer();Check(pool.RecordUploads(highCommand,high,importer.Record,out _),"high-bit generation record");
        Check(highCommand.Integers[Shader.PropertyToID("_HLGI_CopyTextureGeneration")]<0&&unchecked((uint)highCommand.Integers[Shader.PropertyToID("_HLGI_CopyTextureGeneration")])==high.Generation,"uint generation uses integer bits, never float conversion");
        Finish(pool,high);generationField.SetValue(null,previous);Clean();
    }

    static void PackedHalfLayoutsAndBudget()
    {

    static void SharedFrameQuotaAndProgress()
    {
        scenarios++;SystemInfo.supportsAsyncGPUReadback=true;using var aPool=new Pool();using var bPool=new Pool();
        var sources=new[]{Observe(Source(13,9,4)),Observe(Source(7,5,3))};var pools=new[]{aPool,bPool};
        var expected=new int[2];var starts=new int[2];var finishes=new int[2];var done=new bool[2];
        var tables=new Dictionary<GraphicsBuffer,HybridLumenGIRayTextureMetadata[]>();var importer=new HybridLumenGIRayTextureImporter(new ComputeShader());
        for(int frame=0;frame<32&&!done.All(x=>x);frame++)
        {
            Pool.BeginFrame(1000+frame,320);Check(Pool.UploadBytesUsed==0,"new frame resets shared byte allowance");
            for(int camera=0;camera<2;camera++)
            {
                var lease=Prepare(pools[camera],sources[camera]);pools[camera].GetResources(lease,out var metadata,out _);
                if(!tables.ContainsKey(metadata))tables[metadata]=new HybridLumenGIRayTextureMetadata[metadata.count];
                if(frame==1&&camera==0)
                {Finish(pools[camera],lease);Check(Pool.UploadBytesUsed==0,"canceled continuation graph consumes no quota");lease=Prepare(pools[camera],sources[camera]);}
                var cmd=new CommandBuffer();Check(pools[camera].RecordUploads(cmd,lease,importer.Record,out _),"actual importer progressive recording");
                int rangeId=Shader.PropertyToID("_HLGI_CopyRangeStart"),countId=Shader.PropertyToID("_HLGI_CopyRangeCount");long recorded=cmd.CopyRanges.Count*64;
                foreach(var dispatch in cmd.Dispatches)
                {
                    if(dispatch.Kernel==0)starts[camera]++;
                    if(dispatch.Kernel==2){finishes[camera]++;done[camera]=true;}
                    if(dispatch.Kernel!=3)continue;
                    Check(dispatch.Parameters[rangeId]==expected[camera],"ranges continue exactly at prior frame cursor");
                    int count=dispatch.Parameters[countId];expected[camera]+=count;recorded+=(long)count*16;
                    Check(dispatch.X==(count+63)/64&&dispatch.Y==1&&dispatch.Z==1,"actual range dispatch covers declared physical elements");
                }
                for(int i=0;i<cmd.CopyRanges.Count;i++)tables[metadata][cmd.CopyRanges[i].Destination]=((HybridLumenGIRayTextureMetadata[])cmd.Copies[i])[0];
                if(frame==0&&camera==1)Check(!pools[camera].IsHandleCurrent(lease,lease.GetHandle(0))&&cmd.Calls==0,"exhausted frame never exposes uninitialized descriptor or records copy");
                Check(recorded<=320&&Pool.UploadBytesUsed<=320,"all cameras share one actual recorded byte cap");
                long used=Pool.UploadBytesUsed;Pool.BeginFrame(1000+frame,3200);Check(Pool.UploadBytesUsed==used&&Pool.UploadBudgetBytes==320,"same-frame camera cannot replenish or enlarge quota");
                Check(pools[camera].MarkSubmitted(lease,Fence(true),out _),"progress final consumer fence");Pool.PollRetired();
                while(AsyncGPUReadback.Pending.Count>0)
                {
                    var target=AsyncGPUReadback.Pending.Peek().Buffer;var table=(HybridLumenGIRayTextureMetadata[])tables[target].Clone();
                    for(int i=0;i<table.Length;i++)if(table[i].generation!=0)table[i].flags=1;
                    AsyncGPUReadback.CompleteNext(table);Pool.PollRetired();
                }
                if(!done[camera])Check(finishes[camera]==0,"partial chain never records Finish or ready proof");
            }
        }
        Check(done.All(x=>x)&&starts.All(x=>x==1)&&finishes.All(x=>x==1),"both large pages make bounded cross-frame progress with one Begin and Finish each");
        for(int i=0;i<2;i++){Check(expected[i]==(i==0?148:42),"every physical element copied exactly once");var cached=Prepare(pools[i],sources[i]);Check(cached.UploadCount==0,"completed range job becomes reusable only after readback");Finish(pools[i],cached);}
        Pool.ClearReusable();SystemInfo.supportsAsyncGPUReadback=false;Clean();
    }
