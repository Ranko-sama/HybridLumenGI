bool Zero(Array values)
{
    if(values==null||values.Length!=33||Marshal.SizeOf(values.GetType().GetElementType())!=32)return false;
    foreach(object record in values)foreach(var field in record.GetType().GetFields())
    {
        Vector4 v=(Vector4)field.GetValue(record);if(v.x!=0||v.y!=0||v.z!=0||v.w!=0)return false;
    }
    return true;
}
