Entry[] Data(HybridLumenGISurfaceCache surface,int slots=1,int grid=2)
{
    var data=new Entry[surface.CardCount*grid*grid];
    for(int card=0;card<surface.CardCount;card++)for(int local=0;local<grid*grid;local++)
        if(surface.CardOwners[card]>=0)data[card*grid*grid+local]=Valid(101+surface.CardOwners[card],3,7,surface.Generations[card]);
    return data;
}
void Layout(HybridLumenGISurfaceCache surface,int[][] cards,int capacity)
{
    surface.Cards=cards.Select(indices=>(int[])indices.Clone()).ToArray();
    surface.CardOwners=Enumerable.Repeat(-1,capacity).ToArray();
    surface.Generations=Enumerable.Range(0,capacity).Select(card=>(uint)(100+card)).ToArray();
    for(int slot=0;slot<cards.Length;slot++)
    {
        surface.Layouts[slot]++;surface.RequiredCoverage[slot]=true;
        foreach(int card in cards[slot])surface.CardOwners[card]=slot;
    }
}

foreach(string change in new[]{"asset","texture","texels","mesh","bounds","resolution","watertight","error","transform","removed"})
{
    var f=Setup();f.scene.Proxies[0].MeshSdf=new HybridLumenGIMeshSdf();Record(f);
    var asset=f.scene.Proxies[0].MeshSdf;
    switch(change)
    {
        case "asset":f.scene.Proxies[0].MeshSdf=new HybridLumenGIMeshSdf();break;
        case "texture":asset.DistanceTexture=new Texture3D();break;
        case "texels":asset.DistanceTexture.updateCount++;break;
        case "mesh":asset.SourceMesh=new Mesh();break;
        case "bounds":asset.LocalBounds=new Bounds(1);break;
        case "resolution":asset.Resolution=32;break;
        case "watertight":asset.Watertight=false;break;
        case "error":asset.InterpolationError+=.01f;break;
        case "transform":f.scene.Proxies[0].SdfLocalToWorld=new Matrix4x4(1);break;
        case "removed":f.scene.Proxies[0].MeshSdf=null;break;
    }
    f.command.Complete(0,Data(f.surface));
