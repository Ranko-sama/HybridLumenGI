    public sealed class CommandBuffer
    {
        public bool ThrowOnRequest;
        public readonly List<Action<AsyncGPUReadbackRequest>> Callbacks=new();
        public void RequestAsyncReadback(UnityEngine.ComputeBuffer buffer,Action<AsyncGPUReadbackRequest> callback)
        {

    public sealed class HybridLumenGIScene
    {
        public struct ProxyGpuData{public UnityEngine.Vector4 identity,centerShape,extentsRoughness;}
        public struct DirtyRegion
        {
            public int StableId;
            public bool HasPrevious,HasCurrent;
            public UnityEngine.Bounds PreviousBounds,CurrentBounds;
        }
        public IReadOnlyList<DirtyRegion> DirtyRegions { get; } = Array.Empty<DirtyRegion>();
        public readonly record struct Ticket(int Id,int Revision,bool Geometry,bool Material,bool Lighting);
        public readonly ProxyGpuData[] Data;public readonly HybridLumenGIProxy[] Proxies;
        public readonly List<Ticket> Reports=new();public readonly Dictionary<int,Ticket> Latest=new();
        public int Capacity=>Data.Length;public int WarmingCount=>Data.Count(d=>d.identity.w==2);
        public HybridLumenGIScene(int count)
        {
            Data=new ProxyGpuData[count];Proxies=new HybridLumenGIProxy[count];
            for(int i=0;i<count;i++){Data[i].identity=new UnityEngine.Vector4(101+i,3,0,2);Proxies[i]=new HybridLumenGIProxy{StableId=101+i,name="Owner "+(101+i)};}
        }
        public bool IsWarming(int slot)=>slot>=0&&slot<Data.Length&&Data[slot].identity.w==2;
        public HybridLumenGIProxy GetProxy(int slot)=>Proxies[slot];
        public bool ReportReadiness(int id,int revision,bool geometry,bool material,bool lighting)
        {
            // Mirror Scene's actual current-identity/revision guard; this is not a GPU result.
            if(!Data.Any(d=>d.identity.w>.5f&&(int)d.identity.x==id&&(int)d.identity.y==revision))return false;
            var ticket=new Ticket(id,revision,geometry,material,lighting);Reports.Add(ticket);Latest[id]=ticket;return true;
        }
        public bool Ready(int id,int revision)=>Latest.TryGetValue(id,out var ticket)&&ticket.Revision==revision&&ticket.Geometry&&ticket.Material&&ticket.Lighting;
    }

    public sealed class HybridLumenGISurfaceCache
    {
        public uint AtlasGeneration=1;public uint[] Generations;public readonly bool[] Ready,SourceCurrent,RequiredCoverage;
        public readonly int[] Ids,Revisions;public int[] CardOwners;public int[][] Cards;public readonly uint[] Layouts;
        public int CardCount=>CardOwners.Length;
        public bool[] ReadyMask => CardOwners.Select(slot => slot >= 0 && slot < Ready.Length && Ready[slot]).ToArray();
        public HybridLumenGISurfaceCache(int slots)
        {
            Generations=new uint[slots*6];Ready=new bool[slots];SourceCurrent=new bool[slots];RequiredCoverage=new bool[slots];Ids=new int[slots];Revisions=new int[slots];
            CardOwners=new int[slots*6];Cards=new int[slots][];Layouts=new uint[slots];
            for(int i=0;i<slots;i++)
            {
                Ready[i]=SourceCurrent[i]=true;Ids[i]=101+i;Revisions[i]=3;Layouts[i]=1;Cards[i]=new int[6];
                for(int face=0;face<6;face++){int card=i*6+face;Generations[card]=(uint)(10+card);CardOwners[card]=i;Cards[i][face]=card;}
            }
        }
        public uint GetGeneration(int card)=>Generations[card];
        public int GetCardCount(int slot)=>Cards[slot].Length;
        public int GetCardIndex(int slot,int ordinal)=>Cards[slot][ordinal];
        public uint GetLayoutGeneration(int slot)=>Layouts[slot];
        public bool RequiresCardCoverage(int slot)=>RequiredCoverage[slot];
        public bool IsSlotReady(int slot,int id,int revision)=>slot>=0&&slot<Ready.Length&&Ready[slot]&&Ids[slot]==id&&Revisions[slot]==revision;
        public bool IsSlotSourceCurrent(int slot,int id,int revision)=>IsSlotReady(slot,id,revision)&&SourceCurrent[slot];
    }
