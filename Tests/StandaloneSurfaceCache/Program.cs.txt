    static HybridLumenGIProxy Proxy(int id,int charts,int phaseMaterialCount=3)
    {
        var source=Mesh(phaseMaterialCount,"source "+id);
        var renderer=new MeshRenderer{Filter=new MeshFilter{sharedMesh=source}};
        for(int m=0;m<phaseMaterialCount;m++)
        {
            var mat=new Material(new Shader{name="Universal Render Pipeline/Lit"});mat.SetFloat("_Surface",0);mat.SetFloat("_Cull",2);mat.SetFloat("_Metallic",id*10+m);renderer.Materials.Add(mat);
        }
        var proxy=new HybridLumenGIProxy{StableId=id,SourceRenderer=renderer};
        if(charts>=0)
        {
            var asset=new HybridLumenGISurfaceCharts{CaptureMesh=Mesh(charts,"capture "+id),Charts=new HybridLumenGISurfaceCharts.Chart[charts]};
            for(int c=0;c<charts;c++)asset.Charts[c]=new(){stableId=(uint)(101+c*17),sourceSubmesh=(c*2+1)%phaseMaterialCount,localToCard=Matrix4x4.identity,direction=Vector3.forward,triangleIndices=new[]{c}};
            proxy.SurfaceCharts=asset;
        }
        return proxy;
    }
    static CommandBuffer Capture(HybridLumenGISurfaceCache cache){var command=new CommandBuffer();cache.RecordCapture(command,Viewport);return command;}
    static void ValidateRanges(HybridLumenGISurfaceCache cache,int[] counts)
    {
        var ranges=Ranges(cache);var indices=Private<uint[]>(cache,"cardIndices");int cursor=0;var seen=new HashSet<int>();var metadata=Metadata(cache);
        for(int slot=0;slot<counts.Length;slot++)
        {
            Check(cache.GetCardCount(slot)==counts[slot],"required physical set count "+slot);
            if(counts[slot]==0){Check(U(ranges[slot],"count")==0&&U(ranges[slot],"layoutGeneration")==0,"retired empty range "+slot);continue;}
            Check(U(ranges[slot],"first")==cursor&&U(ranges[slot],"count")==counts[slot],"indirect range offset/count "+slot);
            Check(U(ranges[slot],"layoutGeneration")==cache.GetLayoutGeneration(slot),"owner range layout "+slot);
            foreach(int index in Indices(cache,slot))
            {
                Check(index>=0&&index<cache.CardCount&&seen.Add(index),"unique physical pool ownership");
                Check(indices[cursor++]==index&&cache.CardOwners[index]==slot&&U(metadata[index],"ownerSlot")==slot,"explicit indirect owner metadata");
            }
        }
        Check(indices.Skip(cursor).All(i=>i==uint.MaxValue),"unused indices never reference a physical card");
    }

    static void AtlasAllocationRecovery()
    {
        scenarios++;
        var scene = new HybridLumenGIScene(1);
        scene.Set(0, Proxy(42, 1));
        // Request the optional attachment without supplying any baked texels.
        scene.ShadowmaskSources = Array.Empty<LightmapData>();
        var config = Config(1);
        using var cache = new HybridLumenGISurfaceCache();
        int bufferStart = ComputeBuffer.All.Count, textureStart = RTHandle.All.Count;
        cache.Prepare(scene, config);
        Capture(cache);
        var oldBuffers = ComputeBuffer.All.Skip(bufferStart).ToArray();
        var oldTextures = RTHandle.All.Skip(textureStart).ToArray();
        long maskBytes = cache.AllocatedBytes;
        long unmaskedBytes = maskBytes - (long)cache.AtlasWidth * cache.AtlasHeight * 8;
        uint originalGeneration = cache.AtlasGeneration;

        scene.ShadowmaskSources = null;
        bufferStart = ComputeBuffer.All.Count;
        textureStart = RTHandle.All.Count;
        RTHandles.FailNextAllocationNamed = "HLGI Card Depth";
        bool failed = false;
        try { cache.Prepare(scene, config); }
        catch (InvalidOperationException error) when (error.Message == "Injected atlas allocation failure") { failed = true; }
        Check(failed && cache.AtlasGeneration == originalGeneration && cache.AllocatedBytes == maskBytes &&
            cache.ShadowmaskAtlas == oldTextures[5] && cache.DiffuseAtlas == oldTextures[0] && cache.IsSlotReady(0, 42, 1) &&
            oldBuffers.All(b => !b.Released) && oldTextures.All(t => !t.Released) && HybridLumenGIResourceRetirement.Pending.Count == 0,
            "failed late allocation preserves the complete previous atlas and cannot publish the partial one");
        Check(ComputeBuffer.All.Skip(bufferStart).Count(b => b.Released) == 5 && RTHandle.All.Skip(textureStart).Count(t => t.Released) == 4,
            "unpublished replacement buffers and textures are released after failure");

        bufferStart = ComputeBuffer.All.Count;
        textureStart = RTHandle.All.Count;
        cache.Prepare(scene, config);
        var unmaskedBuffers = ComputeBuffer.All.Skip(bufferStart).ToArray();
        var unmaskedTextures = RTHandle.All.Skip(textureStart).ToArray();
        Check(cache.AtlasGeneration == originalGeneration + 1 && cache.ShadowmaskAtlas == null && cache.DepthAtlas != null && !cache.IsSlotReady(0, 42, 1) &&
            cache.AllocatedBytes == unmaskedBytes + maskBytes && unmaskedBuffers.Length == 5 && unmaskedTextures.Length == 5,
            "same configuration retries a complete replacement and keeps the old atlas charged");
        Capture(cache);
        Check(cache.IsSlotReady(0, 42, 1), "the successful retry records and publishes its own cards");

        scene.ShadowmaskSources = Array.Empty<LightmapData>();
        cache.Prepare(scene, config);
        var currentAtlas = cache.DiffuseAtlas;
        var completions = HybridLumenGIResourceRetirement.Pending;
        Check(completions.Count == 2 && cache.ShadowmaskAtlas != null && cache.AllocatedBytes == 2 * maskBytes + unmaskedBytes &&
            oldBuffers.All(b => !b.Released) && oldTextures.All(t => !t.Released) && unmaskedBuffers.All(b => !b.Released) && unmaskedTextures.All(t => !t.Released),
            "two rapid changes retain both detached atlases while their readers are incomplete");
        completions.Dequeue()();
        Check(oldBuffers.All(b => b.Released) && oldTextures.All(t => t.Released) && unmaskedBuffers.All(b => !b.Released) &&
            unmaskedTextures.All(t => !t.Released) && !currentAtlas.Released && cache.AllocatedBytes == unmaskedBytes + maskBytes,
            "the first completion releases and uncharges only the first generation");
        completions.Dequeue()();
        Check(unmaskedBuffers.All(b => b.Released) && unmaskedTextures.All(t => t.Released) && !currentAtlas.Released && cache.AllocatedBytes == maskBytes,
            "the second completion releases and uncharges only the unmasked generation");
    }
