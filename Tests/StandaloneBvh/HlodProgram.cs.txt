using HybridLumenGI;
using UnityEngine;
static void Require(bool condition,string message){if(!condition)throw new Exception(message);}
static (HybridLumenGIProxy root,HybridLumenGIProxy[] children) Group(int count)
{
    HybridLumenGIProxy.ActiveProxies.Clear();
    var root=new HybridLumenGIProxy{StableId=1000,FarFieldOnly=true,bounds=new Bounds(Vector3.zero,new Vector3(12,12,12))};
    HybridLumenGIProxy.ActiveProxies.Add(root);
    var children=new HybridLumenGIProxy[count];
    for(int i=0;i<count;i++)
    {
        children[i]=new HybridLumenGIProxy{StableId=i+1,HLODParent=root,bounds=new Bounds(new Vector3((i%8)-4,(i/8)-2,0),Vector3.one)};
        HybridLumenGIProxy.ActiveProxies.Add(children[i]);
    }
    return(root,children);
}
static void Prepare(HybridLumenGIScene s,bool far,int budget=2)=>s.Prepare(far?new Vector3(100,0,0):Vector3.zero,32,20,1000,16,budget);
static int Slot(HybridLumenGIScene s,int id)=>Array.FindIndex(s.Data,x=>(int)x.identity.x==id&&x.identity.w>.5f);
static void ReportAll(HybridLumenGIScene s)
{
    for(int i=0;i<s.Capacity;i++)if(s.Data[i].identity.w>.5f)
        Require(s.ReportReadiness((int)s.Data[i].identity.x,(int)s.Data[i].identity.y,true,true,true),"Current receipt rejected.");
}
static void RootPublished(HybridLumenGIScene s,int id)
{
    int slot=Slot(s,id);Require(slot>=0&&s.IsPublished(slot)&&s.PublishedCount==1,"Old root representation not held.");
    Require(s.Bvh.NodeCount==1&&s.Bvh.Nodes[0].proxySlot==slot,"Warm geometry leaked into traversal.");
}
// The same cold-start inputs run against frozen old source and the candidate.
// Only assertions differ; the baseline then skips the pre-existing control suite.
static void ColdStartRootFallback(bool baseline)
{
    var g=Group(33);
    using var s=new HybridLumenGIScene();
    void BlockedState(bool rootReady,string stage)
    {
        Require(s.PendingCapacity==1,stage+": detail capacity blockage must count this group exactly once.");
        Require(s.PendingUploadCount==0,stage+": the root or empty baseline must have no queued upload.");
        Require(g.children.All(c=>Slot(s,c.StableId)<0),stage+": oversized detail group was partially admitted.");
        if(baseline)
            Require(s.ValidCount==0&&s.WarmingCount==0&&s.PublishedCount==0&&s.Bvh.NodeCount==0&&Slot(s,g.root.StableId)<0,
                stage+": frozen source did not reproduce the empty cold-start representation.");
        else if(rootReady)
        {
            RootPublished(s,g.root.StableId);
            Require(s.ValidCount==1&&s.WarmingCount==0,stage+": fallback root was not the sole representation.");
        }
        else
        {
            int rootSlot=Slot(s,g.root.StableId);
            Require(rootSlot>=0&&s.IsWarming(rootSlot)&&s.ValidCount==1&&s.WarmingCount==1&&s.PublishedCount==0&&s.Bvh.NodeCount==0,
                stage+": root was not retained in warming until complete readiness.");
            Require(s.DirtyRegions.Count==0,stage+": unready root invalidated the live world.");
        }
        Console.WriteLine($"COLD33 {stage}: valid={s.ValidCount} warming={s.WarmingCount} published={s.PublishedCount} bvh={s.Bvh.NodeCount} pendingCapacity={s.PendingCapacity}");
    }
    Prepare(s,false,1);BlockedState(false,"initial");
    Prepare(s,false,1);BlockedState(false,"without-receipt");
    Require(!s.ReportReadiness(g.root.StableId,g.root.Revision+1,true,true,true),"Cold root accepted a wrong revision.");
    Prepare(s,false,1);BlockedState(false,"wrong-revision");
    Require(s.ReportReadiness(g.root.StableId,g.root.Revision,true,true,false)==!baseline,
        "Cold root partial receipt admission differs from the frozen expectation.");
    Prepare(s,false,1);BlockedState(false,"lighting-not-ready");
    Require(s.ReportReadiness(g.root.StableId,g.root.Revision,true,true,true)==!baseline,
        "Cold root complete receipt admission differs from the frozen expectation.");
    Prepare(s,false,1);BlockedState(true,"complete-root-receipt");
    Prepare(s,false,1);BlockedState(true,"root-held-capacity-blocked");

    // Keep the same scene and camera. Two live leaves now fit beside the root.
    foreach(var child in g.children.Skip(2))
    {child.isActiveAndEnabled=false;HybridLumenGIProxy.ActiveProxies.Remove(child);}
    var remaining=g.children.Take(2).ToArray();
    void DetailWarmState(int count,string stage)
    {
        Require(s.PendingCapacity==0&&s.PendingUploadCount==2-count&&s.WarmingCount==count,
            stage+": fitting detail staging violated capacity or upload budget.");
        Require(s.ValidCount==count+(baseline?0:1)&&s.PublishedCount==(baseline?0:1),
            stage+": fitting detail staging lost coverage or published a partial group.");
        Require(remaining.Count(c=>Slot(s,c.StableId)>=0)==count&&remaining.All(c=>!s.IsPublished(Slot(s,c.StableId))),
            stage+": detail residency/publication is not atomic.");
        if(baseline)Require(s.Bvh.NodeCount==0&&Slot(s,g.root.StableId)<0,stage+": old source invented a root.");
        else RootPublished(s,g.root.StableId);
        Console.WriteLine($"COLD33 {stage}: warming={s.WarmingCount} published={s.PublishedCount} bvh={s.Bvh.NodeCount} pendingUpload={s.PendingUploadCount}");
    }
    Prepare(s,false,1);DetailWarmState(1,"shrink-first-upload");
    int accepted=0;
    foreach(var child in remaining)if(s.ReportReadiness(child.StableId,child.Revision,true,true,true))accepted++;
    Require(accepted==1,"First detail upload did not accept exactly one current receipt.");
    Prepare(s,false,1);DetailWarmState(2,"shrink-second-upload");
    Prepare(s,false,1);DetailWarmState(2,"second-detail-not-ready");
    foreach(var child in remaining)
        Require(s.ReportReadiness(child.StableId,child.Revision,true,true,true),"Complete current detail receipt rejected.");
    Prepare(s,false,1);
    Require(s.ValidCount==2&&s.PublishedCount==2&&s.WarmingCount==0&&s.PendingCapacity==0&&s.PendingUploadCount==0&&s.Bvh.NodeCount==3,
        "Fitting detail group did not atomically publish after all current receipts.");
    Require(Slot(s,g.root.StableId)<0&&remaining.All(c=>s.IsPublished(Slot(s,c.StableId))),
        "Fallback root remained locked or duplicated the published leaves.");
    if(!baseline)
        Require(s.DirtyRegions.Any(x=>x.StableId==g.root.StableId&&x.HasPrevious&&!x.HasCurrent&&
            (x.Kind&HybridLumenGIScene.ChangeKind.Representation)!=0),"Fallback switch lost root representation removal.");
    Console.WriteLine("COLD33 fitting-detail-commit: published=2 warming=0 bvh=3 rootAbsent=true");
}
string coldExpectation=Environment.GetEnvironmentVariable("HLGI_HLOD_COLD_START_EXPECTATION");
Require(string.IsNullOrEmpty(coldExpectation)||coldExpectation=="candidate"||coldExpectation=="baseline","Unknown cold-start expectation.");
bool coldBaseline=coldExpectation=="baseline";
ColdStartRootFallback(coldBaseline);
if(coldBaseline)
{
    Console.WriteLine("PASS: frozen-source cold-start hole established with the same 33/32 inputs; fitting-detail sequence remains functional. Candidate-only controls were not executed.");
    return;
}
// An empty scene can fit all 32 leaves: do not reserve an unnecessary root slot.
var fitting=Group(32);
using(var s=new HybridLumenGIScene())
{
    Prepare(s,false,32);
    Require(s.ValidCount==32&&s.WarmingCount==32&&s.PublishedCount==0&&s.Bvh.NodeCount==0&&s.PendingCapacity==0&&s.PendingUploadCount==0,
        "Exactly fitting cold detail group did not stage directly.");
    Require(Slot(s,fitting.root.StableId)<0&&fitting.children.All(c=>s.IsWarming(Slot(s,c.StableId))),
        "Exactly fitting detail group unnecessarily staged the root.");
    ReportAll(s);Prepare(s,false,32);
    Require(s.ValidCount==32&&s.PublishedCount==32&&s.WarmingCount==0&&s.Bvh.NodeCount==63&&s.PendingCapacity==0&&
        Slot(s,fitting.root.StableId)<0&&fitting.children.All(c=>s.IsPublished(Slot(s,c.StableId))),
        "Exactly fitting details failed whole-group publication.");
}
Console.WriteLine("PASS: cold root fallback/readiness, fitting-detail release and exact-capacity direct-detail control.");
var g=Group(8);
using(var s=new HybridLumenGIScene())
{
    Prepare(s,true,1);Require(s.WarmingCount==1&&s.PublishedCount==0&&s.Bvh.NodeCount==0,"Initial representation published before readiness.");
    Require(s.DirtyRegions.Count==0,"Pure warm-up invalidated the live world's caches.");
    Require(!s.ReportReadiness(g.root.StableId,g.root.Revision+1,true,true,true),"Stale/wrong generation accepted.");
    s.ReportReadiness(g.root.StableId,g.root.Revision,true,true,false);Prepare(s,true,1);
    Require(s.PublishedCount==0,"Lighting readiness was ignored.");
    ReportAll(s);Prepare(s,true,1);RootPublished(s,g.root.StableId);
    for(int frame=0;frame<4;frame++)
    {
        Prepare(s,false,2);RootPublished(s,g.root.StableId);
        Require(s.WarmingCount==(frame+1)*2,"Warm-up upload budget or whole-group reservation failed.");
        ReportAll(s);
    }
    // A ready receipt for an older revision must not let the entire group publish.
    int lightingSlot=Slot(s,g.children[0].StableId);g.children[0].SourceRenderer.shadowCastingMode=UnityEngine.Rendering.ShadowCastingMode.Off;g.children[0].Revision++;
    Prepare(s,false,2);RootPublished(s,g.root.StableId);
    Require(s.IsWarming(lightingSlot)&&s.LightingPolicies[lightingSlot].shadowCastingMode==0u,"Lighting-only edit removed staging geometry or failed to refresh policy.");
    Require(!s.ReportReadiness(g.children[0].StableId,g.children[0].Revision-1,true,true,true),"Old lighting-policy receipt survived revision change.");
    g.children[7].bounds.center+=new Vector3(.1f,0,0);g.children[7].Revision++;
    Prepare(s,false,2);RootPublished(s,g.root.StableId);
    Require(!s.ReportReadiness(g.children[7].StableId,g.children[7].Revision-1,true,true,true),"Old geometry revision receipt survived replacement.");
    ReportAll(s);s.InvalidateReadiness();Prepare(s,false,2);RootPublished(s,g.root.StableId);
    ReportAll(s);Prepare(s,false,2);
    Require(s.PublishedCount==8&&s.WarmingCount==0&&s.Bvh.NodeCount==15,"New detail group was not atomically published.");
    Require(Slot(s,g.root.StableId)<0,"Old root remained a duplicate occluder after commit.");
    Require(s.DirtyRegions.Any(x=>x.StableId==g.root.StableId&&x.HasPrevious&&!x.HasCurrent),"Commit lost old representation invalidation.");
    Require(s.DirtyRegions.All(x=>(x.Kind&HybridLumenGIScene.ChangeKind.Representation)!=0),"Pure publication was not distinguished from hard geometry edits.");
    foreach(var child in g.children)Require(s.IsPublished(Slot(s,child.StableId)),"Partial detail publication.");
    // Cancel a reverse transition after the root becomes ready, before publication.
    Prepare(s,true,1);Require(s.PublishedCount==8&&s.WarmingCount==1,"Reverse transition dropped details before root ready.");
    ReportAll(s);Prepare(s,false,1);
    Require(s.PublishedCount==8&&s.WarmingCount==0&&Slot(s,g.root.StableId)<0,"Cancelled warm-up published or retained a stale target.");
    Require(!s.ReportReadiness(g.root.StableId,g.root.Revision,true,true,true),"Retired staging identity accepted a receipt.");
}
