using HybridLumenGI;

static void Require(bool condition,string message){if(!condition)throw new Exception(message);}
static (HybridLumenGIProxy root,HybridLumenGIProxy[] children) Group(int count)
{
    HybridLumenGIProxy.ActiveProxies.Clear();
    var root=new HybridLumenGIProxy{StableId=1000,FarFieldOnly=true,bounds=new Bounds(Vector3.zero,new Vector3(12,12,12))};
    HybridLumenGIProxy.ActiveProxies.Add(root);
    var children=new HybridLumenGIProxy[count];
    for(int i=0;i<count;i++)
    {
        children[i]=new HybridLumenGIProxy{StableId=i+1,HLODParent=root,bounds=new Bounds(new Vector3((i%8)-4,(i/8)-2,0),Vector3.one)};
        HybridLumenGIProxy.ActiveProxies.Add(children[i]);
    }
    return(root,children);
}
static void Prepare(HybridLumenGIScene s,bool far,int budget=2)=>s.Prepare(far?new Vector3(100,0,0):Vector3.zero,32,20,1000,16,budget);
static int Slot(HybridLumenGIScene s,int id)=>Array.FindIndex(s.Data,x=>(int)x.identity.x==id&&x.identity.w>.5f);
static void ReportAll(HybridLumenGIScene s)
{
    for(int i=0;i<s.Capacity;i++)if(s.Data[i].identity.w>.5f)
        Require(s.ReportReadiness((int)s.Data[i].identity.x,(int)s.Data[i].identity.y,true,true,true),"Current receipt rejected.");
}
static void RootPublished(HybridLumenGIScene s,int id)
{
    int slot=Slot(s,id);Require(slot>=0&&s.IsPublished(slot)&&s.PublishedCount==1,"Old root representation not held.");
    Require(s.Bvh.NodeCount==1&&s.Bvh.Nodes[0].proxySlot==slot,"Warm geometry leaked into traversal.");
}

static void IncompleteCapacityFallback(bool baseline)
{
    var group = Group(33);
    var declarations = HybridLumenGIStreamedRepresentation.Active;
    declarations.Clear();
    foreach (var leaf in group.children)
    {
        leaf.isActiveAndEnabled = false;
        declarations.Add(new HybridLumenGIStreamedRepresentation { Proxy = leaf });
    }
    using var scene = new HybridLumenGIScene();
    Prepare(scene, false, 32);
    int rootSlot = Slot(scene, group.root.StableId);
    Require(rootSlot >= 0 && scene.IsWarming(rootSlot) && scene.ValidCount == 1, "The cold root must start warming before the first leaf binds.");
    Require(scene.ReportReadiness(group.root.StableId, group.root.Revision, true, true, false), "Current partial root receipt was rejected.");

    group.children[0].isActiveAndEnabled = true;
    Prepare(scene, false, 32);
    Console.WriteLine($"Declared 33 / bound 1 / capacity 32: warming={scene.WarmingCount}, published={scene.PublishedCount}, pendingCapacity={scene.PendingCapacity}.");
    if (baseline)
    {
        Require(scene.ValidCount == 0 && scene.PendingCapacity == 0 &&
            !scene.ReportReadiness(group.root.StableId, group.root.Revision, true, true, true),
            "Release201 did not reproduce loss of the warming root after a leaf bound.");
        declarations.Clear();
        return;
    }

    Require(scene.IsWarming(rootSlot) && scene.ValidCount == 1 && scene.PendingCapacity == 1 && scene.Bvh.NodeCount == 0,
        "The oversized declared set must hold the original warming root without publishing partial geometry.");
    Require(Slot(scene, group.children[0].StableId) < 0, "The single bound leaf reserved a slot for an oversized set.");
    Require(scene.ReportReadiness(group.root.StableId, group.root.Revision, true, true, true), "Complete current root receipt was rejected.");
    Prepare(scene, false, 32);
    RootPublished(scene, group.root.StableId);
    Require(scene.PendingCapacity == 1, "The still-oversized declaration must report one blocked group.");
    Console.WriteLine("Complete root receipt: published=1, pendingCapacity=1.");

    foreach (var removed in declarations.Skip(2)) removed.isActiveAndEnabled = false;
    Prepare(scene, false, 32);
    RootPublished(scene, group.root.StableId);
    Require(scene.WarmingCount == 0 && scene.PendingCapacity == 0, "A fitting set with one unbound leaf staged partially or kept the old capacity block.");
    group.children[1].isActiveAndEnabled = true;
    Prepare(scene, false, 32);
    RootPublished(scene, group.root.StableId);
    Require(scene.WarmingCount == 2, "The complete fitting leaf set did not stage beside the old root.");
    var first = group.children[0];
    Require(scene.ReportReadiness(first.StableId, first.Revision, true, true, true), "The first complete leaf receipt was rejected.");
    Prepare(scene, false, 32);
    RootPublished(scene, group.root.StableId);
    var second = group.children[1];
    Require(scene.ReportReadiness(second.StableId, second.Revision, true, true, true), "The second complete leaf receipt was rejected.");
    Prepare(scene, false, 32);
    Require(scene.PublishedCount == 2 && scene.WarmingCount == 0 && scene.PendingCapacity == 0 &&
        scene.Bvh.NodeCount == 3 && Slot(scene, group.root.StableId) < 0,
        "The two ready leaves did not replace the root as a complete set.");
    declarations.Clear();
    Console.WriteLine("PASS incomplete capacity fallback: one ready root, then atomic replacement by both fitting leaves.");
}

static void ColdStartRootFallback(bool baseline)
{
    var g=Group(33);
    using var s=new HybridLumenGIScene();
    void BlockedState(bool rootReady,string stage)
    {
        Require(s.PendingCapacity==1,stage+": detail capacity blockage must count this group exactly once.");
        Require(s.PendingUploadCount==0,stage+": the root or empty baseline must have no queued upload.");
        Require(g.children.All(c=>Slot(s,c.StableId)<0),stage+": oversized detail group was partially admitted.");
        if(baseline)
            Require(s.ValidCount==0&&s.WarmingCount==0&&s.PublishedCount==0&&s.Bvh.NodeCount==0&&Slot(s,g.root.StableId)<0,
                stage+": frozen source did not reproduce the empty cold-start representation.");
        else if(rootReady)
        {
            RootPublished(s,g.root.StableId);
            Require(s.ValidCount==1&&s.WarmingCount==0,stage+": fallback root was not the sole representation.");
        }
        else
        {
            int rootSlot=Slot(s,g.root.StableId);
            Require(rootSlot>=0&&s.IsWarming(rootSlot)&&s.ValidCount==1&&s.WarmingCount==1&&s.PublishedCount==0&&s.Bvh.NodeCount==0,
                stage+": root was not retained in warming until complete readiness.");
            Require(s.DirtyRegions.Count==0,stage+": unready root invalidated the live world.");
        }
        Console.WriteLine($"COLD33 {stage}: valid={s.ValidCount} warming={s.WarmingCount} published={s.PublishedCount} bvh={s.Bvh.NodeCount} pendingCapacity={s.PendingCapacity}");
    }
    Prepare(s,false,1);BlockedState(false,"initial");
    Prepare(s,false,1);BlockedState(false,"without-receipt");
    Require(!s.ReportReadiness(g.root.StableId,g.root.Revision+1,true,true,true),"Cold root accepted a wrong revision.");
    Prepare(s,false,1);BlockedState(false,"wrong-revision");
    Require(s.ReportReadiness(g.root.StableId,g.root.Revision,true,true,false)==!baseline,
        "Cold root partial receipt admission differs from the frozen expectation.");
    Prepare(s,false,1);BlockedState(false,"lighting-not-ready");
    Require(s.ReportReadiness(g.root.StableId,g.root.Revision,true,true,true)==!baseline,
        "Cold root complete receipt admission differs from the frozen expectation.");
    Prepare(s,false,1);BlockedState(true,"complete-root-receipt");
    Prepare(s,false,1);BlockedState(true,"root-held-capacity-blocked");

    // Keep the same scene and camera. Two live leaves now fit beside the root.
    foreach(var child in g.children.Skip(2))
    {child.isActiveAndEnabled=false;HybridLumenGIProxy.ActiveProxies.Remove(child);}
    var remaining=g.children.Take(2).ToArray();
    void DetailWarmState(int count,string stage)
    {
        Require(s.PendingCapacity==0&&s.PendingUploadCount==2-count&&s.WarmingCount==count,
            stage+": fitting detail staging violated capacity or upload budget.");
        Require(s.ValidCount==count+(baseline?0:1)&&s.PublishedCount==(baseline?0:1),
            stage+": fitting detail staging lost coverage or published a partial group.");
        Require(remaining.Count(c=>Slot(s,c.StableId)>=0)==count&&remaining.All(c=>!s.IsPublished(Slot(s,c.StableId))),
            stage+": detail residency/publication is not atomic.");
        if(baseline)Require(s.Bvh.NodeCount==0&&Slot(s,g.root.StableId)<0,stage+": old source invented a root.");
        else RootPublished(s,g.root.StableId);
        Console.WriteLine($"COLD33 {stage}: warming={s.WarmingCount} published={s.PublishedCount} bvh={s.Bvh.NodeCount} pendingUpload={s.PendingUploadCount}");
    }
    Prepare(s,false,1);DetailWarmState(1,"shrink-first-upload");
    int accepted=0;
    foreach(var child in remaining)if(s.ReportReadiness(child.StableId,child.Revision,true,true,true))accepted++;
    Require(accepted==1,"First detail upload did not accept exactly one current receipt.");
    Prepare(s,false,1);DetailWarmState(2,"shrink-second-upload");
    Prepare(s,false,1);DetailWarmState(2,"second-detail-not-ready");
    foreach(var child in remaining)
        Require(s.ReportReadiness(child.StableId,child.Revision,true,true,true),"Complete current detail receipt rejected.");
    Prepare(s,false,1);
    Require(s.ValidCount==2&&s.PublishedCount==2&&s.WarmingCount==0&&s.PendingCapacity==0&&s.PendingUploadCount==0&&s.Bvh.NodeCount==3,
        "Fitting detail group did not atomically publish after all current receipts.");
    Require(Slot(s,g.root.StableId)<0&&remaining.All(c=>s.IsPublished(Slot(s,c.StableId))),
        "Fallback root remained locked or duplicated the published leaves.");
    if(!baseline)
        Require(s.DirtyRegions.Any(x=>x.StableId==g.root.StableId&&x.HasPrevious&&!x.HasCurrent&&
            (x.Kind&HybridLumenGIScene.ChangeKind.Representation)!=0),"Fallback switch lost root representation removal.");
    Console.WriteLine("COLD33 fitting-detail-commit: published=2 warming=0 bvh=3 rootAbsent=true");
}

Require(string.IsNullOrEmpty(coldExpectation)||coldExpectation=="candidate"||coldExpectation=="baseline","Unknown cold-start expectation.");

static void ReadyHlodBehindOlderBuild(bool useQueue, bool exactRoot, bool fitsBesidePrepared = true)
{
    var group = Group(2);
    var independent = new HybridLumenGIProxy { StableId = 1001, bounds = new Bounds(new Vector3(0, 8, 0), Vector3.one) };
    if (!exactRoot) HybridLumenGIProxy.ActiveProxies.Add(independent);
    using var scene = new HybridLumenGIScene();
    Prepare(scene, true); ReportAll(scene);
    scene.Prepare(new Vector3(100, 0, 0), 32, 20, 1000, 16, 2, null, _ => false,
        (_, _) => throw new Exception("The first publication has no old group to defer."));
    Prepare(scene, false); ReportAll(scene);
    int rootSlot = Slot(scene, group.root.StableId);
    int exactSlot = exactRoot ? rootSlot : Slot(scene, independent.StableId);
    int independentCount = exactRoot ? 0 : 1;
    Predicate<int> preparedExact = slot => slot == exactSlot;
    const int triangleLimit = 1024;
    const int detailTriangles = 6;
    int preparedTriangles = fitsBesidePrepared ? 6 : 1020;
    bool ReadyExactTarget(int slot, bool removesExact)
    {
        Require(removesExact == exactRoot, "Handoff did not distinguish approximate and exact old geometry.");
        return scene.IsWarming(slot) && detailTriangles <= (removesExact ? triangleLimit : triangleLimit - preparedTriangles);
    }
    void RequireOldGroup()
    {
        Require(scene.IsPublished(rootSlot) && scene.IsPublished(exactSlot) && scene.PublishedCount == 1 + independentCount,
            "The full old group and independent exact geometry must stay published.");
        Require(scene.Bvh.NodeCount == 1 + 2 * independentCount && group.children.All(child => scene.IsWarming(Slot(scene, child.StableId))),
            "Warm detail geometry entered the live BVH.");
    }
    RequireOldGroup();

    var main = new object();
    var older = new object();
    var priorBuild = new object();
    var quota = HybridLumenGIRtasBudget.Settings.Default;
    quota.MaxBuildsPerFrame = 1;
    HybridLumenGIRtasBudget.ResetFrameAdmission();
    HybridLumenGIRtasBudget.BeginFrame(30, quota);
    Require(HybridLumenGIRtasBudget.TryConsumeBuild(priorBuild, 30, 6, 1, out _), "Expected the previous frame's one build.");
    Require(!HybridLumenGIRtasBudget.TryConsumeBuild(older, 30, 5, 1, out _) &&
        HybridLumenGIRtasBudget.PendingDemands == 1, "Expected a genuine older demand after the previous build.");
    HybridLumenGIRtasBudget.BeginFrame(31, quota);
    Require(HybridLumenGIRtasBudget.BuildsThisFrame == 0, "The current frame's build turn must be unused.");

    bool PrepareHandoff(bool blocked)
    {
        bool sawExactTarget = false;
        scene.Prepare(Vector3.zero, 32, 20, 1000, 16, 2, null, preparedExact, (slot, removesExact) =>
        {
            if (!ReadyExactTarget(slot, removesExact)) return false;
            sawExactTarget = true;
            return blocked;
        });
        return sawExactTarget;
    }
    bool blocked = useQueue ? HybridLumenGIRtasBudget.IsBuildTurnBlocked(main) :
        HybridLumenGIRtasBudget.BuildsThisFrame >= quota.MaxBuildsPerFrame;
    Require(PrepareHandoff(blocked) == fitsBesidePrepared, "Wrong exact target decision for the prepared geometry.");
    if (!useQueue)
    {
        Require(scene.PublishedCount == 2 && Slot(scene, group.root.StableId) < 0 &&
            !HybridLumenGIRtasBudget.TryConsumeBuild(main, 31, 12, 2, out string reason) &&
            reason.Contains("older active camera"), "The count-only gate must expose the late rejection.");
        return;
    }
    if (!fitsBesidePrepared)
    {
        Require(scene.PublishedCount == 3 && Slot(scene, group.root.StableId) < 0 && scene.IsPublished(exactSlot) &&
            HybridLumenGIRtasBudget.PendingDemands == 1, "Details that cannot join the retained RTAS waited for its build turn.");
        return;
    }

    RequireOldGroup();
    Require(scene.HasReadyExactHlodReplacement(preparedExact, ReadyExactTarget) && HybridLumenGIRtasBudget.PendingDemands == 1,
        "Holding the old group lost ready staging or changed the queue before recording.");
    if (!exactRoot)
    {
        var leaf = group.children[0];
        scene.ReportReadiness(leaf.StableId, leaf.Revision, true, false, true);
        Require(!scene.HasReadyExactHlodReplacement(preparedExact, ReadyExactTarget),
            "Revoked material readiness must remove the deferred handoff's queue eligibility.");
        ReportAll(scene);
    }
    // The real Feature joins the queue only after it records DXR with the retained root.
    Require(HybridLumenGIRtasBudget.RequestBuild(main, 31, 0, 0, out _), "The retained representation could not register its waiting turn.");
    Require(HybridLumenGIRtasBudget.TryConsumeBuild(older, 31, 5, 1, out _) &&
        HybridLumenGIRtasBudget.BuildsThisFrame == 1 && HybridLumenGIRtasBudget.PendingDemands == 1,
        "The held HLOD displaced the older camera or consumed its build.");
    HybridLumenGIRtasBudget.BeginFrame(32, quota);
    Require(!HybridLumenGIRtasBudget.IsBuildTurnBlocked(main), "The main camera did not reach the front.");
    Require(PrepareHandoff(false), "The queued handoff disappeared before using its turn.");
    int buildTriangles = 2 * detailTriangles + independentCount * preparedTriangles;
    int buildInstances = 2 + independentCount;
    Require(scene.PublishedCount == buildInstances && Slot(scene, group.root.StableId) < 0 &&
        (exactRoot || scene.IsPublished(exactSlot)) &&
        group.children.All(child => scene.IsPublished(Slot(scene, child.StableId))) &&
        HybridLumenGIRtasBudget.TryConsumeBuild(main, 32, buildTriangles, buildInstances, out _) &&
        HybridLumenGIRtasBudget.TriangleWorkThisFrame == buildTriangles && HybridLumenGIRtasBudget.InstanceWorkThisFrame == buildInstances &&
        HybridLumenGIRtasBudget.PendingDemands == 0, "The complete group did not take and pay for its own turn.");
}

static void SetResident(HybridLumenGIProxy proxy, bool resident)
{
    proxy.isActiveAndEnabled = resident;
    if (resident) HybridLumenGIProxy.ActiveProxies.Add(proxy);
    else HybridLumenGIProxy.ActiveProxies.Remove(proxy);
}
