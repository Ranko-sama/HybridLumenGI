sealed class Backend : HybridLumenGIChunkStreaming.IBackend
{
    public sealed class Op : HybridLumenGIChunkStreaming.IOperation
    { public string path;public int handle;public bool unload;public bool IsDone{get;set;} }
    public readonly Dictionary<string,int> loaded=new();public readonly List<Op> operations=new();
    public string failPath;public int attempts;private int nextHandle;
    public int LoadedHandle(string path)=>loaded.TryGetValue(path,out int handle)?handle:0;
    public HybridLumenGIChunkStreaming.IOperation BeginLoad(string path)
    {
        attempts++;if(path==failPath)throw new InvalidOperationException("Scene is not in Build Settings.");
        var op=new Op{path=path,handle=++nextHandle};operations.Add(op);return op;
    }
    public HybridLumenGIChunkStreaming.IOperation BeginUnload(string path,int handle)
    {
        attempts++;if(LoadedHandle(path)!=handle)throw new InvalidOperationException("Ownership changed.");
        var op=new Op{path=path,handle=handle,unload=true};operations.Add(op);return op;
    }
    public void Complete()
    {
        var op=operations.First(x=>!x.IsDone);
        if(op.unload)loaded.Remove(op.path);else loaded[op.path]=op.handle;
        op.IsDone=true;
    }
}

static class ChunkMultiviewChecks
{
    static void Require(bool value,string message){if(!value)throw new Exception(message);}
    static HybridLumenGIChunkStreaming.Definition Def(string id,float x)=>new(){id=id,scenePath=$"Assets/Chunks/{id}.unity",worldBounds=new Bounds(new Vector3(x,0,0),Vector3.one)};
    static readonly System.Reflection.MethodInfo MultiviewStep = typeof(HybridLumenGIChunkStreaming).GetMethod("Step",
        new[] { typeof(IReadOnlyList<Vector3>), typeof(float), typeof(float), typeof(int), typeof(int), typeof(int) });

    static void StepUnion(HybridLumenGIChunkStreaming stream, Vector3[] positions, int maxResidentScenes) =>
        MultiviewStep.Invoke(stream, new object[] { positions, 20f, 40f, maxResidentScenes, 1, 1 });

    public static void RunLoadFairness()
    {
        var backend = new Backend();
        var stream = new HybridLumenGIChunkStreaming(new[] { Def("A", 0), Def("B", 100), Def("C", 200) }, backend);
        var positions = new[] { new Vector3(110, 0, 0), Vector3.zero };
        var waiting = stream.States[1];
        int startedStep = 0;
        for (int step = 1; step <= 64; step++)
        {
            if (step > 1)
            {
                var previous = backend.operations.Last();
                backend.Complete();
                if (step == 2) positions[1] = new Vector3(200, 0, 0);
                else if (previous.unload)
                    positions[1] = previous.path.EndsWith("A.unity") ? Vector3.zero : new Vector3(200, 0, 0);
            }

            StepUnion(stream, positions, 3);
            Require(waiting.Wanted && stream.OperationsStarted == 1 && stream.ActiveOperations == 1 &&
                stream.States.Count(state => state.Owned || state.Loading) <= 3,
                "Continuous demand and the shared start, concurrency and residency limits must hold.");
            if (step == 1) Require(stream.States[0].Loading, "Fresh requests must still use the nearer scene first.");
            if (waiting.Loading) { startedStep = step; break; }

            // The outgoing scene has its fallback ready only after the incoming load starts.
            foreach (var state in stream.States)
                if (!state.Wanted && state.Owned && !state.Unloading)
                    Require(stream.ReportUnloadSafe(state.Id, state.Generation, true), "Expected a current safe-unload receipt.");
        }

        Require(startedStep > 0, "Continuously wanted scene B was still starved after 64 steps with room to load.");
        backend.Complete();
        StepUnion(stream, positions, 3);
        Require(waiting.Loaded && waiting.Owned, "The waiting scene did not finish its ordinary load.");
        Console.WriteLine($"PASS: Additive scene B started on step {startedStep} and completed within the original limits.");
    }

    static string Snapshot(HybridLumenGIChunkStreaming s,Backend b)=>System.Text.Json.JsonSerializer.Serialize(new{

        states=s.States.Select(x=>new{x.Id,x.Generation,x.Wanted,x.Loaded,x.Owned,x.Loading,x.Unloading,x.UnloadSafe,x.LastError}),
        operations=b.operations.Select(x=>new{x.path,x.handle,x.unload,x.IsDone})});
