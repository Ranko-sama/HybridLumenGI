sealed class Backend : HybridLumenGIChunkStreaming.IBackend
{
    public sealed class Op : HybridLumenGIChunkStreaming.IOperation
    { public string path;public int handle;public bool unload;public bool IsDone{get;set;} }
    public readonly Dictionary<string,int> loaded=new();public readonly List<Op> operations=new();
    public string failPath;public int attempts;private int nextHandle;
    public int LoadedHandle(string path)=>loaded.TryGetValue(path,out int handle)?handle:0;
    public HybridLumenGIChunkStreaming.IOperation BeginLoad(string path)
    {
        attempts++;if(path==failPath)throw new InvalidOperationException("Scene is not in Build Settings.");
        var op=new Op{path=path,handle=++nextHandle};operations.Add(op);return op;
    }
    public HybridLumenGIChunkStreaming.IOperation BeginUnload(string path,int handle)
    {
        attempts++;if(LoadedHandle(path)!=handle)throw new InvalidOperationException("Ownership changed.");
        var op=new Op{path=path,handle=handle,unload=true};operations.Add(op);return op;
    }
    public void Complete()
    {
        var op=operations.First(x=>!x.IsDone);
        if(op.unload)loaded.Remove(op.path);else loaded[op.path]=op.handle;
        op.IsDone=true;
    }
}
