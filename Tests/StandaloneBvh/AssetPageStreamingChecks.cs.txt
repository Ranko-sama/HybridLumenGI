using HybridLumenGI;
using Stream=HybridLumenGI.HybridLumenGIAssetPageStreaming;

static class AssetPageStreamingChecks
{
    static void Check(bool pass,string failure){if(!pass)throw new Exception(failure);}
    static Stream.Definition Page(string id,params string[] dependencies)=>new(){id=id,bundlePath=id+".bundle",assetName=id,kind=Stream.Kind.Geometry,fileBytes=10,residentBytes=20,uploadBytes=8,dependencies=dependencies};
    static Stream.Budgets Budget()=>new(){ioBytesPerStep=20,maxIoBytesInFlight=20,maxBundleBytes=100,maxResidentBytes=100,uploadBytesPerStep=8,concurrentOperations=2,startsPerStep=2};
    public static void Run()
    {
        var backend=new Provider();var stream=new Stream(new[]{Page("parent","dependency"),Page("dependency")},backend);var budget=Budget();
        stream.Request("parent",true);stream.Step(budget);
        Check(stream.IoBytesStarted==20 && stream.IoBytesInFlight==20 && stream.BundleBytes==20 && backend.operations.Count==2,"Bundle IO admission did not reserve exact declared bytes.");
        stream.Step(budget);Check(backend.operations.Count==2,"In-flight bundles did not retain concurrency/IO budgets.");
        backend.CompleteAll();stream.Step(budget);
        Check(backend.operations.Last().id=="dependency" && backend.operations.Last().phase=="asset" && stream.ResidentBytes==20,"Parent asset loaded before its dependency, or residency was not reserved at admission.");
        backend.CompleteAll();stream.Step(budget);Check(backend.operations.Last().id=="parent" && stream.UploadBytesStarted==8,"Dependency completion did not admit parent upload.");
        backend.CompleteAll();stream.Step(budget);Check(stream.TryAcquire("parent",out var lease),"Ready group could not acquire a lease.");
        var ticket=new Ticket();Check(stream.ReleaseAfter(lease,ticket) && !stream.ReleaseAfter(lease,ticket),"Retirement accepted duplicate receipt.");
        stream.Request("parent",false);stream.Step(budget);
        Check(stream.ResidentBytes==40 && backend.operations.Count==4 && stream.RetiringLeases==1,"Unproven GPU retirement freed a page/dependency.");
        ticket.done=true;stream.Step(budget);
        Check(backend.operations.Last().id=="parent" && backend.operations.Last().phase=="release" && stream.ResidentBytes==40,"Parent release failed to retain residency until native completion.");
        backend.CompleteAll();stream.Step(budget);
        Check(backend.operations.Last().id=="dependency" && stream.ResidentBytes==20,"Dependency was released before parent completion.");
        backend.CompleteAll();stream.Step(budget);Check(stream.ResidentBytes==0 && stream.BundleBytes==0 && stream.IoBytesInFlight==0,"Completed releases did not return reservations.");
        Check(!stream.ReleaseAfter(lease,new Ticket{done=true}),"Retired lease authorized another lifetime.");

        var cancelledProvider=new Provider();var cancelled=new Stream(new[]{Page("a")},cancelledProvider);
        cancelled.Request("a",true);cancelled.Step(budget);cancelled.Request("a",false);cancelled.Step(budget);
        Check(cancelled.IoBytesInFlight==10 && cancelledProvider.operations.Count==1,"Cancellation forgot an in-flight native operation.");
        cancelledProvider.CompleteAll();cancelled.Step(budget);
        Check(cancelledProvider.operations.Last().phase=="release" && cancelled.ResidentBytes==0,"Cancelled bundle completion loaded an unwanted asset.");
        cancelledProvider.CompleteAll();cancelled.Step(budget);Check(cancelled.BundleBytes==0,"Cancelled late completion leaked its owned bundle.");

        var cappedProvider=new Provider();var capped=new Stream(new[]{Page("a"),Page("b")},cappedProvider);
        var cap=Budget();cap.ioBytesPerStep=10;cap.maxBundleBytes=10;cap.maxResidentBytes=19;
        capped.Request("a",true);capped.Request("b",true);capped.Step(cap);
        Check(cappedProvider.operations.Count==1 && capped.BundleBytes==10,"IO/whole-bundle capacity was exceeded.");
        cappedProvider.CompleteAll();capped.Step(cap);Check(capped.ResidentBytes==0 && capped.BudgetBlocked>0,"Resident page exceeding capacity was admitted.");
        cap.maxResidentBytes=20;cap.uploadBytesPerStep=7;capped.Step(cap);Check(capped.ResidentBytes==0,"Upload admission exceeded per-step budget.");
        cap.uploadBytesPerStep=8;capped.Step(cap);Check(capped.ResidentBytes==20,"Admissible page remained blocked after budget increase.");

        var groups=new Stream(new[]{Page("meshA"),Page("textureA"),Page("meshB"),Page("textureB")},new Provider());
        var groupBudget=Budget();groupBudget.maxResidentBytes=60;var admitted=new HashSet<string>();
        Check(groups.TryAdmitGroup(new[]{"meshA","textureA"},groupBudget,admitted),"Complete group fitting capacity was not admitted.");
        Check(!groups.TryAdmitGroup(new[]{"meshB","textureB"},groupBudget,admitted) && admitted.Count==2,"Partial second group consumed material capacity and could deadlock both representations.");
        Check(groups.TryAdmitGroup(new[]{"meshB","textureA"},groupBudget,admitted) && admitted.Count==3,"Shared group dependency was charged twice.");

        var retryProvider=new Provider();var retry=new Stream(new[]{Page("a")},retryProvider);
        retry.Request("a",true);for(int i=0;i<3;i++){retry.Step(budget);retryProvider.CompleteAll();}
        uint firstGeneration=retry.GetState("a").Generation;retry.Request("a",false);retry.Step(budget);
        var noNewIo=budget;noNewIo.ioBytesPerStep=0;
        Check(!retry.TryAdmitGroup(new[]{"a"},noNewIo,new()),"Releasing page bypassed future bundle admission.");
        retryProvider.operations.Last().error="Native unload failed.";retryProvider.CompleteAll();retry.Step(budget);
        Check(retry.GetState("a").Status==Stream.Phase.ReleaseFailed && retry.ResidentBytes==20,"Failed native release lost reservations.");
        Check(!retry.TryAdmitGroup(new[]{"a"},noNewIo,new()),"Failed release bypassed future bundle admission.");
        retry.Request("a",true);int operations=retryProvider.operations.Count;
        for(int i=0;i<59;i++)retry.Step(budget);
        Check(retryProvider.operations.Count==operations,"Failed release ignored retry backoff or reloaded uncertain ownership.");
        retry.Step(budget);Check(retryProvider.operations.Last().phase=="release","Re-requested failed release did not complete quarantine before reuse.");
        retryProvider.CompleteAll();retry.Step(budget);
        Check(retry.GetState("a").Generation>firstGeneration && retry.ResidentBytes==0,"Page lifetime was reused after failed release.");

        var unknownProvider=new Provider();var unknown=new Stream(new[]{Page("a")},unknownProvider);
        unknown.Request("a",true);unknown.Step(budget);unknownProvider.operations[0].throws=true;unknown.Request("a",false);
        unknown.Step(budget);Check(unknown.IoBytesInFlight==10 && unknown.ActiveOperations==1 && unknownProvider.operations.Count==1,"Unknown native completion was treated as failure/completion.");
        bool rejected=false;try{_ = new Stream(new[]{Page("a","b"),Page("b","a")},new Provider());}catch(ArgumentException){rejected=true;}
        Check(rejected,"Cyclic page dependencies accepted.");
        foreach(string path in new[]{"../a","a/../b","C:/x","/root","a\\b","a//b"})Check(!Stream.RelativePath(path),"Unsafe relative bundle path accepted: "+path);
        RemainingWorkAdmission();
        Console.WriteLine("PASS: actual independent asset-page scheduler: dependency ordering, IO/upload/resident quotas, non-cancellable late completion, delayed consumer retirement, failed-release quarantine/retry, generations, unknown native completion and manifest/path rejection. Fake provider; Unity AssetBundle IO/GPU execution is not claimed.");
    }
    static void RemainingWorkAdmission()
    {
        var failures=new List<string>();
        void Case(string name,Action body)
        {
            try{body();Console.WriteLine("PASS: "+name);}
            catch(Exception error){failures.Add(name+": "+error.Message);Console.WriteLine("FAIL: "+failures.Last());}
        }
        Case("Ready shared page after lowering start budgets",()=>
        {
            var provider=new Provider();var small=Page("small","shared");small.fileBytes=1;small.residentBytes=2;small.uploadBytes=1;
            var stream=new Stream(new[]{Page("shared"),small},provider);var budget=Budget();
            stream.Request("shared",true);
            for(int i=0;i<3;i++){stream.Step(budget);provider.CompleteAll();}
            Check(stream.TryAcquire("shared",out var lease),"Shared page did not reach leased Ready state.");
            budget.ioBytesPerStep=budget.uploadBytesPerStep=1;
            // Same public scheduler sequence as AssetPageStreamer.Drive: whole groups,
            // then their accepted root requests, then one real scheduler step.
            for(int i=0;i<3;i++)
            {
                var admitted=new HashSet<string>();
                bool shared=stream.TryAdmitGroup(new[]{"shared"},budget,admitted);
                bool next=stream.TryAdmitGroup(new[]{"small"},budget,admitted);
                stream.Request("shared",shared);stream.Request("small",next);stream.Step(budget);provider.CompleteAll();
            }
            Check(stream.GetState("small").Status==Stream.Phase.Ready,"Small new page was blocked by the historical costs of its leased Ready dependency.");
            Check(lease.Asset!=null && provider.operations.Count(x=>x.id=="shared")==2,"Ready shared page was released or reloaded.");
            Check(stream.BundleBytes==11 && stream.ResidentBytes==22 && stream.IoBytesInFlight==0,"Shared reservation accounting changed.");
            var tooSmall=budget;tooSmall.maxBundleBytes=10;
            Check(!stream.TryAdmitGroup(new[]{"small"},tooSmall,new()),"Ready pages bypassed total bundle capacity.");
            tooSmall=budget;tooSmall.maxResidentBytes=21;
            Check(!stream.TryAdmitGroup(new[]{"small"},tooSmall,new()),"Ready pages bypassed total resident capacity.");
        });
        Case("BundleReady continuation and failed-stage readmission",()=>
        {
            var provider=new Provider();var stream=new Stream(new[]{Page("a")},provider);var budget=Budget();
            stream.Request("a",true);stream.Step(budget);provider.CompleteAll();
            var paused=budget;paused.startsPerStep=0;stream.Step(paused);
            Check(stream.GetState("a").Status==Stream.Phase.BundleReady && stream.IoBytesInFlight==0,"Bundle did not finish before the reduced budget.");
            budget.ioBytesPerStep=budget.maxIoBytesInFlight=0;
            Check(stream.TryAdmitGroup(new[]{"a"},budget,new()),"Completed bundle incorrectly required another file read.");
            var noUpload=budget;noUpload.uploadBytesPerStep=7;
            Check(!stream.TryAdmitGroup(new[]{"a"},noUpload,new()),"BundleReady page bypassed its remaining upload cost.");
            stream.Step(budget);
            Check(provider.operations.Count==2 && provider.operations.Last().phase=="asset" && stream.IoBytesStarted==0,"BundleReady continuation reread the file or failed to start its asset.");
            Check(stream.TryAdmitGroup(new[]{"a"},noUpload,new()),"Already in-flight upload was charged as a new upload.");
            provider.operations.Last().error="Asset load failed.";provider.CompleteAll();stream.Step(budget);
            Check(stream.GetState("a").Status==Stream.Phase.Failed && stream.BundleBytes==10 && stream.ResidentBytes==0,"Asset failure lost its retained bundle accounting.");
            Check(stream.TryAdmitGroup(new[]{"a"},budget,new()) && !stream.TryAdmitGroup(new[]{"a"},noUpload,new()),"Asset retry did not retain upload-only admission.");
            for(int i=0;i<60;i++)stream.Step(budget);
            provider.CompleteAll();stream.Step(budget);
            Check(stream.GetState("a").Status==Stream.Phase.Ready && provider.operations.Count(x=>x.phase=="bundle")==1,"Asset retry did not finish without rereading the retained bundle.");
            var failedProvider=new Provider();var failed=new Stream(new[]{Page("b")},failedProvider);
            failed.Request("b",true);failed.Step(Budget());failedProvider.operations[0].error="Bundle load failed.";
            failedProvider.CompleteAll();failed.Step(Budget());
            Check(failed.GetState("b").Status==Stream.Phase.Failed && !failed.TryAdmitGroup(new[]{"b"},budget,new()),"Failed bundle retry bypassed new I/O admission.");
        });
        Check(failures.Count==0,string.Join("; ",failures));
    }
    sealed class Ticket : Stream.ICompletion {internal bool done;public bool IsComplete=>done;}
    sealed class Provider : Stream.IProvider
    {
        internal sealed class Op : Stream.IOperation
        {
            internal string id,phase,error;internal bool done,throws;
            public bool IsDone=>throws?throw new InvalidOperationException("Unknown completion"):done;
            public string Error=>error;public object Result=>phase=="release" || error!=null?null:this;
        }
        internal readonly List<Op> operations=new();
        Op Add(string id,string phase){var op=new Op{id=id,phase=phase};operations.Add(op);return op;}
        public Stream.IOperation LoadBundle(Stream.Definition definition)=>Add(definition.id,"bundle");
        public Stream.IOperation LoadAsset(Stream.Definition definition,object bundle)=>Add(definition.id,"asset");
        public Stream.IOperation Release(Stream.Definition definition,object bundle,object asset)=>Add(definition.id,"release");
        internal void CompleteAll(){foreach(var op in operations)op.done=true;}
    }
}
