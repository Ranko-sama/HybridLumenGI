            public Snapshot(UnityEngine.Renderer r,UnityEngine.Material m,int index)
            {SourceRenderer=r;SourceMaterial=m;MaterialIndex=index;CanReuseHistory=!m.UsesRenderTexture;
             FrozenValue=r.SlotMpbValues.TryGetValue(index,out float slot)?slot:r.RendererMpbValue!=0?r.RendererMpbValue:m.SourceValue;}
        }
        public static readonly List<(UnityEngine.Renderer renderer,int index,UnityEngine.Material source)> Calls=new();
        public static int RejectIndex=-1;public static bool ThrowCapture,WrongSnapshot;public static Action<int> OnCapture;
        public bool TryCapture(UnityEngine.Renderer renderer,int index,UnityEngine.Material source,out Snapshot snapshot,out string reason)
        {
            Calls.Add((renderer,index,source));snapshot=null;reason=null;
            if(ThrowCapture)throw new Exception("source reader failure");
            if(index==RejectIndex||index<0||index>=renderer.Materials.Count||!ReferenceEquals(source,renderer.Materials[index]))
            {reason="rejected original source";return false;}
            snapshot=new(renderer,source,WrongSnapshot?index+1:index);OnCapture?.Invoke(index);return true;
        }
        public bool IsCurrent(UnityEngine.Renderer r,int index,Snapshot snapshot)=>throw new Exception("History API must not gate current RT capture");
    }

    // SurfaceCache is a recording boundary here; the actual cache/asset classes
    // have their own source-compiled suites. No triangle allocator is duplicated.
    public sealed class HybridLumenGISurfaceCache
    {
        public struct TriangleChartBinding { public uint cardIndex,chartId,layoutGeneration,geometryRevision; }
        public struct HardwareMaterial { public uint cullMode; public UnityEngine.Vector4 diffuse,emission; }
        public readonly HybridLumenGIScene Scene;
        public readonly Dictionary<int,uint> Geometry=new(),Layouts=new();
        public readonly Dictionary<(int,int,int),TriangleChartBinding> Bindings=new();
        public readonly Dictionary<(int,int),HardwareMaterial> Materials=new();
        public readonly HashSet<int> UnknownMaterial=new(),NoBinding=new();
        public int BindingQueries,MaterialQueries;
        public HybridLumenGISurfaceCache(HybridLumenGIScene scene){Scene=scene;}
        public uint GetGeometryRevision(int slot)=>Scene.Renderers[slot]!=null?Geometry.GetValueOrDefault(slot,1u):0;
        public uint GetLayoutGeneration(int slot)=>NoBinding.Contains(slot)?0:Layouts.GetValueOrDefault(slot,1u);
        public bool TryGetTriangleBinding(int slot,UnityEngine.Mesh mesh,int submesh,int primitive,out TriangleChartBinding binding)
        {
            BindingQueries++;binding=new(){cardIndex=uint.MaxValue};
            if(NoBinding.Contains(slot)||Scene.Renderers[slot]?.Filter?.sharedMesh!=mesh)return false;
            if(Bindings.TryGetValue((slot,submesh,primitive),out binding))return true;
            binding=new(){cardIndex=(uint)(slot*32+submesh+3),chartId=(uint)(submesh+11),layoutGeneration=GetLayoutGeneration(slot),geometryRevision=GetGeometryRevision(slot)};return true;
        }
        public bool TryGetHardwareMaterial(int slot,UnityEngine.Mesh mesh,int submesh,out HardwareMaterial material)
        {
            MaterialQueries++;material=default;
            if(UnknownMaterial.Contains(slot)||Scene.Renderers[slot]?.Filter?.sharedMesh!=mesh)return false;
            if(Materials.TryGetValue((slot,submesh),out material))return true;
            material=new(){cullMode=2,diffuse=new(.1f*(submesh+1),.2f,.3f,1),emission=new(submesh,2*submesh,3*submesh,0)};return true;
        }
    }
