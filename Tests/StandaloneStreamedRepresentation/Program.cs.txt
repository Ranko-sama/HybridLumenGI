sealed class Provider:Stream.IProvider
{
    public sealed class Op:Stream.IOperation{public string phase;public bool done;public object value;public bool IsDone=>done;public string Error=>null;public object Result=>value;}
    public readonly List<Op> operations=new();
    Op Add(string phase,object value){var op=new Op{phase=phase,value=value};operations.Add(op);return op;}
    public Stream.IOperation LoadBundle(Stream.Definition definition)=>Add("bundle",new object());
    public Stream.IOperation LoadAsset(Stream.Definition definition,object bundle)=>Add("asset",definition.kind==Stream.Kind.Geometry?new Mesh():definition.kind==Stream.Kind.Material?new Material():new Texture());
    public Stream.IOperation Release(Stream.Definition definition,object bundle,object asset)=>Add("release",null);
    public void CompleteAll(){foreach(var op in operations)op.done=true;}
}
