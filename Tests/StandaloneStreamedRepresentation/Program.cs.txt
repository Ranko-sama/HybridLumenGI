HybridLumenGIStreamedRepresentation AddMember(string geometry, HybridLumenGIProxy parent, float load, float unload)
{
    var member = new HybridLumenGIStreamedRepresentation();
    var renderer = new MeshRenderer { gameObject = member.gameObject };
    var filter = new MeshFilter { gameObject = member.gameObject };
    var proxy = new HybridLumenGIProxy { gameObject = member.gameObject, enabled = false, HLODParent = parent };
    member.transform.position = root.transform.position;
    member.Configure(nested.driver, proxy, filter, renderer, geometry, null, null, new[] { "material" }, new Bounds(Vector3.zero, new Vector3(2, 2, 2)));
    member.ConfigureDistances(load, unload);
    member.UseHlodPublicationForRendering(true);
    Fixture.Call(member, "OnEnable");
    return member;
}

Check(retry.state.BundleBytes == 0 && brokenPage.Generation == failedGeneration, "The original backoff must expire before retrying");
Check(retry.driver.SetRequests(secondGroup, secondPages), "The later group can request again after releasing its pages");
retry.Drive();
Check(brokenPage.Status == Stream.Phase.LoadingBundle && brokenPage.Generation != failedGeneration &&
    healthyPage.Status == Stream.Phase.Unloaded, "The waiting request must regain its original order on the exact retry step");
for (int step = 0; step < 16 && brokenPage.Status != Stream.Phase.Ready; step++)
{
    retry.provider.CompleteAll();
    retry.Drive();
}
Check(brokenPage.Status == Stream.Phase.Ready, "An eligible retry must still complete after the bundle becomes readable");
Console.WriteLine("PASS load retry admission: healthy group completes during backoff; the original request retries at expiry and completes.");
CheckFailedAssetRetirement();
Check(CheckBundleIoHeadroom(), "An admitted large page must make progress while smaller file reads overlap");
CheckWithdrawnBundleDependency();
Console.WriteLine("PASS multiview cold-demand capability; actual C# representation/driver/scheduler, adapted Unity objects and IO. No Unity/AssetBundle/GPU/HLOD native-chain acceptance.");

static void CheckWithdrawnBundleDependency()
{
    var fixture = new Fixture(new[] { Fixture.Page("material", Stream.Kind.Material, "texture"),
        Fixture.Page("texture", Stream.Kind.Texture), Fixture.Page("z-healthy", Stream.Kind.Geometry) });
    fixture.budget.startsPerStep = 1;
    fixture.ApplyBudget();
    var material = fixture.state.GetState("material");
    var texture = fixture.state.GetState("texture");
    var healthy = fixture.state.GetState("z-healthy");
    var requester = new object();
    Check(fixture.driver.SetRequests(requester, new[] { material.Id }), "Material request failed");
    fixture.Drive();
    Check(material.Status == Stream.Phase.LoadingBundle && texture.Status == Stream.Phase.Unloaded &&
        fixture.state.ActiveOperations == 1, "Expected only the material file to be in flight");
    var oldRead = fixture.provider.operations.Single();
    uint oldGeneration = material.Generation;

    fixture.driver.RemoveRequests(requester);
    Check(fixture.driver.SetRequests(new object(), new[] { healthy.Id }), "Healthy request failed");
    fixture.Drive();
    Console.WriteLine($"Withdrawn dependency: material={material.Status}, texture={texture.Status}, healthy={healthy.Status}, starts={fixture.state.OperationsStarted}.");
    Check(!oldRead.done && material.Status == Stream.Phase.LoadingBundle && material.Generation == oldGeneration &&
        texture.Status == Stream.Phase.Unloaded && healthy.Status == Stream.Phase.LoadingBundle,
        "Retaining a withdrawn in-flight material started its unrequested texture instead of the healthy page");

    oldRead.done = true;
    fixture.Drive();
    Check(material.Status == Stream.Phase.Releasing && texture.Status == Stream.Phase.Unloaded &&
        fixture.state.BundleBytes == 20 && fixture.state.ResidentBytes == 0,
        "The withdrawn bundle did not enter normal release with its reservation intact");
    fixture.provider.operations.Last().done = true;
    fixture.Drive();
    Check(material.Status == Stream.Phase.Unloaded && fixture.state.BundleBytes == 10,
        "Completing release did not return only the withdrawn material bundle's reservation");

    Check(fixture.driver.SetRequests(requester, new[] { material.Id }), "Renewed material request failed");
    for (int step = 0; step < 8 && material.Status != Stream.Phase.Ready; step++)
    {
        fixture.provider.CompleteAll();
        fixture.Drive();
    }
    Check(material.Status == Stream.Phase.Ready && material.Generation > oldGeneration &&
        texture.Status == Stream.Phase.Ready && healthy.Status == Stream.Phase.Ready,
        "A renewed material request could not load its dependency or recover after safe release");
}

static void CheckFailedAssetRetirement()
{
    var fixture = new Fixture(new[] { Fixture.Page("broken", Stream.Kind.Geometry, "shared"),
        Fixture.Page("healthy", Stream.Kind.Geometry, "shared"), Fixture.Page("shared", Stream.Kind.Material) });
    fixture.budget.maxBundleBytes = 20;
    fixture.budget.maxResidentBytes = 40;
    fixture.ApplyBudget();
    fixture.provider.FailingAsset = "broken";
    var first = new object();
    var second = new object();
    Check(fixture.driver.SetRequests(first, new[] { "broken" }, priority: 1) &&
        fixture.driver.SetRequests(second, new[] { "healthy" }), "Both complete groups must remain requested");

    var broken = fixture.state.GetState("broken");
    var healthy = fixture.state.GetState("healthy");
    for (int step = 0; step < 8 && broken.Status != Stream.Phase.Failed; step++)
    {
        fixture.provider.CompleteAll();
        fixture.Drive();
    }
    uint failedGeneration = broken.Generation;
    Check(broken.Status == Stream.Phase.Failed && failedGeneration > 0 && healthy.Status == Stream.Phase.Unloaded &&
        fixture.state.BundleBytes == 20 && fixture.state.ResidentBytes == 20 && fixture.driver.TryAcquire("shared", out _),
        "The asset failure must leave only its real bundle and the shared dependency charged");

    int elapsed = 0;
    void Advance(bool complete = true)
    {
        if (complete) fixture.provider.CompleteAll();
        fixture.Drive();
        elapsed++;
    }
    Advance(complete: false);
    Check(broken.Status == Stream.Phase.Releasing && fixture.state.BundleBytes == 20 && healthy.Status == Stream.Phase.Unloaded,
        "Asset load backoff must not delay releasing its unrequested bundle");
    Advance(complete: false);
    Check(broken.Status == Stream.Phase.Releasing && fixture.state.BundleBytes == 20 && healthy.Status == Stream.Phase.Unloaded,
        "An unfinished release must keep the failed bundle charged");
    Advance();
    Check(broken.Status == Stream.Phase.Failed && broken.Generation == failedGeneration && fixture.state.BundleBytes == 10,
        "Release completion must return only the failed bundle and preserve load backoff");
    for (; elapsed < 12 && healthy.Status != Stream.Phase.Ready;) Advance();
    int healthyReadyAt = elapsed;
    Check(healthy.Status == Stream.Phase.Ready && fixture.state.BundleBytes == 20 && fixture.state.ResidentBytes == 40 &&
        broken.Status == Stream.Phase.Failed && broken.Generation == failedGeneration,
        "The healthy group must finish during the asset load backoff within the original limits");

    fixture.driver.RemoveRequests(second);
    fixture.provider.FailingAsset = null;
    for (; elapsed < 59;) Advance();
    Check(broken.Status == Stream.Phase.Failed && broken.Generation == failedGeneration && fixture.state.BundleBytes == 10,
        "Releasing the failed bundle must not advance the load retry");
    Advance();
    Check(broken.Status == Stream.Phase.LoadingBundle && broken.Generation != failedGeneration,
        "The original request must retry on step 60 under a fresh generation");
    for (; elapsed < 68 && broken.Status != Stream.Phase.Ready;) Advance();
    Check(broken.Status == Stream.Phase.Ready && fixture.state.BundleBytes == 20 && fixture.state.ResidentBytes == 40,
        "The original request must recover once its asset can load");
    Console.WriteLine($"PASS failed asset retirement: release starts at step 1, healthy page ready at step {healthyReadyAt}, original retries at step 60 and becomes ready at step {elapsed}.");
}

static bool CheckBundleIoHeadroom()
{
    const long MiB = 1L << 20;
    var shared = Fixture.Page("shared", Stream.Kind.Material);
    shared.fileBytes = 10 * MiB;
    var large = Fixture.Page("large", Stream.Kind.Geometry, shared.id);
    large.fileBytes = 50 * MiB;
    var smallPages = Enumerable.Range(0, 6).Select(index =>
    {
        var page = Fixture.Page($"small-{index}", Stream.Kind.Geometry, shared.id);
        page.fileBytes = 30 * MiB;
        return page;
    }).ToArray();
    var definitions = new[] { shared, large }.Concat(smallPages).ToArray();
    var fixture = new Fixture(definitions);
    fixture.budget = Stream.Budgets.Default;
    fixture.budget.maxIoBytesInFlight = 60 * MiB;
    fixture.ApplyBudget();

    Check(fixture.driver.SetRequests(new object(), new[] { shared.id }), "Shared material request was rejected");
    for (int step = 0; step < 3; step++)
    {
        fixture.provider.CompleteAll();
        fixture.Drive();
    }
    Check(fixture.state.GetState(shared.id).Status == Stream.Phase.Ready, "Shared dependency must be ready before the overlap");

    void RequestSmall(int index) => Check(fixture.driver.SetRequests(new object(), new[] { smallPages[index].id }),
        "Small group request was rejected");
    RequestSmall(0);
    RequestSmall(1);
    fixture.Drive();
    Check(fixture.state.IoBytesInFlight == 60 * MiB, "Two small file reads must already be in flight");
    Check(fixture.driver.SetRequests(new object(), new[] { large.id }, priority: 1), "Large group request was rejected");
    var largePage = fixture.state.GetState(large.id);

    int largeStarted = 0, largeReady = 0;
    for (int step = 1; step <= 8; step++)
    {
        if (step % 2 == 0) RequestSmall(1 + step / 2);
        var pending = fixture.provider.operations.FirstOrDefault(op => !op.done && op.phase == "asset") ??
            fixture.provider.operations.First(op => !op.done);
        pending.done = true;
        fixture.Drive();
        Check(largePage.Wanted && fixture.driver.BudgetBlockedGroups == 0, "The whole large group must remain admitted");
        long inFlight = definitions.Where(page => fixture.state.GetState(page.id).Status == Stream.Phase.LoadingBundle)
            .Sum(page => page.fileBytes);
        Check(fixture.state.IoBytesInFlight == inFlight && inFlight <= fixture.budget.maxIoBytesInFlight &&
            fixture.state.IoBytesStarted <= fixture.budget.ioBytesPerStep &&
            fixture.state.ActiveOperations == fixture.provider.operations.Count(op => !op.done) &&
            fixture.state.ActiveOperations <= fixture.budget.concurrentOperations &&
            fixture.state.OperationsStarted <= fixture.budget.startsPerStep, "File or operation accounting exceeded the shared limits");
        if (largeStarted == 0 && largePage.Status == Stream.Phase.LoadingBundle) largeStarted = step;
        if (largeReady == 0 && largePage.Status == Stream.Phase.Ready) largeReady = step;
    }

    bool smallReady = fixture.state.GetState(smallPages[2].id).Status == Stream.Phase.Ready;
    Console.WriteLine($"Bundle IO headroom: large start={largeStarted}, ready={largeReady}, first later small ready={smallReady} after 8 steps (0 means no event).");
    return largeStarted > 0 && largeReady > 0 && smallReady;
}

        if (definition.id == FailingBundle) throw new System.IO.IOException("Controlled bundle read failure.");
