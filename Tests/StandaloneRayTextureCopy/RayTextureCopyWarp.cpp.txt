class Reader{

ComPtr<ID3D12Resource> Buffer(ID3D12Device* device,UINT64 bytes,D3D12_HEAP_TYPE type,D3D12_RESOURCE_STATES state,D3D12_RESOURCE_FLAGS flags=D3D12_RESOURCE_FLAG_NONE){

ComPtr<ID3D12Resource> Upload(ID3D12Device* device,const void* source,UINT64 bytes){

    Hr(buffer->Map(0,&empty,&mapped),"Map upload");std::memcpy(mapped,source,static_cast<std::size_t>(bytes));buffer->Unmap(0,nullptr);return buffer;

    barrier.Transition.StateAfter=after;barrier.Transition.Subresource=D3D12_RESOURCE_BARRIER_ALL_SUBRESOURCES;list->ResourceBarrier(1,&barrier);
