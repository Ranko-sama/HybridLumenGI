cmake_minimum_required(VERSION 3.20)
project(HLGIHardwareSurfaceWarp LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This standalone WARP runner requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(HardwareSurfaceWarp HardwareSurfaceWarp.cpp)
target_link_libraries(HardwareSurfaceWarp PRIVATE d3d12 dxgi)
if(MSVC)
    target_compile_options(HardwareSurfaceWarp PRIVATE /W4 /WX /permissive- /utf-8)
endif()
