    public struct Vector3 : IEquatable<Vector3>
    {
        public float x,y,z; public Vector3(float a,float b,float c){x=a;y=b;z=c;}
        public static Vector3 zero => new(0,0,0); public static Vector3 one => new(1,1,1);

        public static Vector3 operator +(Vector3 a,Vector3 b)=>new(a.x+b.x,a.y+b.y,a.z+b.z);

    public struct Vector4 : IEquatable<Vector4>
    {
        public float x,y,z,w; public Vector4(float a,float b,float c,float d){x=a;y=b;z=c;w=d;}
        public static implicit operator Vector4(Vector3 value)=>new(value.x,value.y,value.z,0);
        public bool Equals(Vector4 b)=>x.Equals(b.x)&&y.Equals(b.y)&&z.Equals(b.z)&&w.Equals(b.w);
        public override bool Equals(object b)=>b is Vector4 v&&Equals(v);
        public override int GetHashCode()=>HashCode.Combine(x,y,z,w);
        // Unity's operator is approximate while Equals is component-exact. This
        // distinction matters when proving that local SDF metadata did not change.
        public static bool operator ==(Vector4 a,Vector4 b)
        {float x=a.x-b.x,y=a.y-b.y,z=a.z-b.z,w=a.w-b.w;return x*x+y*y+z*z+w*w<1e-10f;}
        public static bool operator !=(Vector4 a,Vector4 b)=>!(a==b);
    }
    public struct Bounds : IEquatable<Bounds>
    {
        public Vector3 center,size; public Bounds(Vector3 c,Vector3 s){center=c;size=s;}
        public Vector3 min=>center-size*.5f; public Vector3 max=>center+size*.5f;
        public void Encapsulate(Vector3 p)
        {
            Vector3 a=min,b=max;
            var low=new Vector3(MathF.Min(a.x,p.x),MathF.Min(a.y,p.y),MathF.Min(a.z,p.z));
            var high=new Vector3(MathF.Max(b.x,p.x),MathF.Max(b.y,p.y),MathF.Max(b.z,p.z));
            center=(low+high)*.5f;size=high-low;
        }
        public bool Contains(Vector3 p)=>p.x>=min.x&&p.y>=min.y&&p.z>=min.z&&p.x<=max.x&&p.y<=max.y&&p.z<=max.z;
        public bool Equals(Bounds b)=>center.Equals(b.center)&&size.Equals(b.size);
        public override bool Equals(object b)=>b is Bounds v&&Equals(v);
        public override int GetHashCode()=>HashCode.Combine(center,size);
    }
    public struct Matrix4x4 : IEquatable<Matrix4x4>
    {
        public N.Matrix4x4 value;
        public static Matrix4x4 identity=>new(){value=N.Matrix4x4.Identity};
        public static Matrix4x4 Translate(Vector3 p)=>new(){value=N.Matrix4x4.CreateTranslation(p.x,p.y,p.z)};
        public static Matrix4x4 Scale(Vector3 p)=>new(){value=N.Matrix4x4.CreateScale(p.x,p.y,p.z)};
        public static Matrix4x4 operator *(Matrix4x4 a,Matrix4x4 b)=>new(){value=b.value*a.value};
        public float determinant=>value.GetDeterminant();
        public Matrix4x4 inverse {get{N.Matrix4x4.Invert(value,out var m);return new(){value=m};}}
        public float this[int i]=>i switch {0=>value.M11,1=>value.M12,2=>value.M13,3=>value.M14,4=>value.M21,5=>value.M22,6=>value.M23,7=>value.M24,8=>value.M31,9=>value.M32,10=>value.M33,11=>value.M34,12=>value.M41,13=>value.M42,14=>value.M43,_=>value.M44};
        public float this[int r,int c]=>this[c*4+r];
        public Vector3 MultiplyPoint3x4(Vector3 p){var q=N.Vector3.Transform(new N.Vector3(p.x,p.y,p.z),value);return new(q.X,q.Y,q.Z);}
        public bool Equals(Matrix4x4 b)=>value.Equals(b.value);
        public override bool Equals(object b)=>b is Matrix4x4 m&&Equals(m);
        public override int GetHashCode()=>value.GetHashCode();
        public static bool operator ==(Matrix4x4 a,Matrix4x4 b)=>a.Equals(b);
        public static bool operator !=(Matrix4x4 a,Matrix4x4 b)=>!a.Equals(b);
    }

    public class ComputeBuffer : IDisposable
    {
        public static int AttemptedFloatVoxels;
        public static bool FailNextFloatUpload;
        public int count; public string name;
        public ComputeBuffer(int c,int stride,ComputeBufferType t){count=c;}
        public void SetData<T>(T[] data){if(data.Length>count)throw new Exception("oversized buffer upload");}
        public void SetData<T>(T[] data,int start,int destination,int length)
        {
            if(start<0||destination<0||start+length>data.Length||destination+length>count)throw new Exception("invalid buffer range");
            if(typeof(T)!=typeof(float))return;
            AttemptedFloatVoxels+=length;
            if(!FailNextFloatUpload)return;
            FailNextFloatUpload=false;
            throw new InvalidOperationException("Simulated voxel SetData failure");
        }
        public void Release(){}public void Dispose(){}
    }
