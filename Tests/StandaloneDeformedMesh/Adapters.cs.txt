    public class Mesh:Object
    {
        private Vector3[] positions=Array.Empty<Vector3>();
        private readonly List<int[]> indices=new List<int[]>();
        private readonly List<MeshTopology> topology=new List<MeshTopology>();
        public bool isReadable=true;
        public Rendering.IndexFormat indexFormat;
        public Bounds bounds;
        public Vector3[] vertices {get=>(Vector3[])positions.Clone();set=>positions=(Vector3[])value.Clone();}
        public Vector3[] normals=Array.Empty<Vector3>();public Vector4[] tangents=Array.Empty<Vector4>();public Vector2[] uv=Array.Empty<Vector2>();
        public int vertexCount=>positions.Length;public int vertexBufferCount=>1;public int GetVertexBufferStride(int stream)=>48;
        public int subMeshCount
        {
            get=>indices.Count;
            set{while(indices.Count<value){indices.Add(Array.Empty<int>());topology.Add(MeshTopology.Triangles);}while(indices.Count>value){indices.RemoveAt(indices.Count-1);topology.RemoveAt(topology.Count-1);}}
        }
        public uint GetIndexCount(int sub)=>(uint)indices[sub].Length;
        public int[] GetIndices(int sub,bool applyBaseVertex)=>(int[])indices[sub].Clone();
        public MeshTopology GetTopology(int sub)=>topology[sub];
        public void SetIndices(int[] values,MeshTopology kind,int sub,bool recalculateBounds)
        {if(sub>=subMeshCount)subMeshCount=sub+1;indices[sub]=(int[])values.Clone();topology[sub]=kind;}
    }

    public class SkinnedMeshRenderer:Renderer
    {
        public Mesh sharedMesh;public bool FailBake;public int BakeCount;
        public void BakeMesh(Mesh destination,bool useScale)
        {
            BakeCount++;if(FailBake)throw new InvalidOperationException("Controlled capture failure");
            destination.vertices=sharedMesh.vertices;destination.normals=sharedMesh.normals;destination.tangents=sharedMesh.tangents;destination.uv=sharedMesh.uv;
            destination.subMeshCount=sharedMesh.subMeshCount;
            for(int i=0;i<sharedMesh.subMeshCount;i++)destination.SetIndices(sharedMesh.GetIndices(i,true),sharedMesh.GetTopology(i),i,false);
            destination.bounds=sharedMesh.bounds;
        }
    }

    public static class ControlledBake
    {
        public sealed class Gate {public readonly ManualResetEventSlim Started=new ManualResetEventSlim(false),Release=new ManualResetEventSlim(false);}
        private static readonly Dictionary<int,Gate> gates=new Dictionary<int,Gate>();
        public static Gate For(int marker){lock(gates){if(!gates.TryGetValue(marker,out var gate))gates.Add(marker,gate=new Gate());return gate;}}
    }
    public static class HybridLumenGIMeshSdfBake
    {
        public static float[] Bake(Vector3[] vertices,int[] triangles,int size,Func<int,int,bool> progress,out Bounds bounds,out bool closed)
        {
            // Deterministic barriers control real Task.Run work. Returning even after cancellation
            // exercises the component's own generation/cancellation checks, not a fake scheduler.
            var gate=ControlledBake.For((int)vertices[0].x);gate.Started.Set();
            if(!gate.Release.Wait(5000))throw new TimeoutException("Test did not release worker barrier");
            bounds=new Bounds(new Vector3(0,0,0),new Vector3(1,1,1));closed=true;return new float[size*size*size];
        }
    }
