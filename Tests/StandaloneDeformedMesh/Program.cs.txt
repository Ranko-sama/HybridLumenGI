    static Mesh Mesh(int marker)
    {
        var mesh=new Mesh{vertices=new[]{new Vector3(marker,0,0),new Vector3(marker,1,0),new Vector3(marker,0,1)},
            normals=new Vector3[3],tangents=new Vector4[3],uv=new Vector2[3],bounds=new Bounds(new Vector3(marker,0,0),new Vector3(1,1,1)),subMeshCount=1};
        mesh.SetIndices(new[]{0,1,2},MeshTopology.Triangles,0,false);return mesh;
    }
    static Owner Add(int marker,bool skinned=false)
    {

        Mesh mesh=Mesh(marker);

        typeof(HybridLumenGIDeformedMesh).GetMethod("OnEnable",PrivateInstance).Invoke(component,null);
        return new Owner{component=component,proxy=proxy,renderer=renderer,original=mesh};
    }
    static void Disable(Owner owner)
    {

    }

    }

        Mesh pendingA=PendingMesh(owner);Require(!ReferenceEquals(pendingA,owner.original),"Capture aliased producer-owned mesh.");
        Finish(owner,10);HybridLumenGIDeformedMesh.UploadBudgetBytesPerFrame=0;Frame();
        Require(!HybridLumenGIDeformedMesh.HasSnapshot(owner.renderer) && Job(owner)!=null && Texture3D.Created==0,"Completed worker bypassed the publication upload budget.");
        int preparations=0,publications=0;

        };
        HybridLumenGIDeformedMesh.UploadBudgetBytesPerFrame=8L<<20;Frame();
        Require(preparations==2 && publications==1 && owner.component.PublishedFrame>=0,"Snapshot construction/publication observation missing.");
        Require(ReferenceEquals(owner.renderer.GetComponent<MeshFilter>().sharedMesh,owner.original),"GI publication mutated the visible renderer's original mesh.");
        HybridLumenGIMeshSdf.BeforeSet=HybridLumenGISurfaceCharts.BeforeSet=null;owner.proxy.OnDirty=null;
        long oneReservation=HybridLumenGIDeformedMesh.ReservedBytes;
        var sdfA=HybridLumenGIDeformedMesh.GetSdf(owner.renderer,null);var chartsA=HybridLumenGIDeformedMesh.GetCharts(owner.renderer,null);

        owner.component.RequestCapture(Mesh(20));Frame();Started(20);Mesh pendingB=PendingMesh(owner);
        owner.component.RequestCapture(Mesh(30));Finish(owner,20);Frame();Started(30);
        Require(pendingB.Destroyed && ReferenceEquals(HybridLumenGIDeformedMesh.GetSourceMesh(owner.renderer),pendingA),"Cancelled older generation replaced the live snapshot or leaked its private mesh.");
        Mesh pendingC=PendingMesh(owner);Finish(owner,30);Frame();

        Require(!pendingA.Destroyed && !sdfA.Destroyed && !chartsA.Destroyed && HybridLumenGIDeformedMesh.ReservedBytes==2*oneReservation,"Replacement destroyed/released old snapshot before final-context completion.");
        int previousFences=RecordedGpu.Fences.Count;RenderPipelineManager.End();
        Require(RecordedGpu.Fences.Count==previousFences+1 && RecordedGpu.Submits>0 && !pendingA.Destroyed,"Retirement did not submit a final-context fence or assumed it completed immediately.");
        Frame();RenderPipelineManager.End();
        Require(!pendingA.Destroyed && HybridLumenGIDeformedMesh.ReservedBytes==2*oneReservation,"Elapsed context/frame released unfinished GPU work.");
        RecordedGpu.CompleteSubmitted();RenderPipelineManager.End();
        Require(pendingA.Destroyed && sdfA.Destroyed && sdfA.Texture.Destroyed && chartsA.Destroyed && chartsA.CaptureMesh.Destroyed &&
            !pendingC.Destroyed && HybridLumenGIDeformedMesh.ReservedBytes==oneReservation,"Completed old snapshot did not retire all owned resources exactly once.");

        owner.component.RequestCapture(Mesh(40));Frame();Started(40);Mesh rejected=PendingMesh(owner);Finish(owner,40);
        HybridLumenGISurfaceCharts.FailSet=true;Frame();HybridLumenGISurfaceCharts.FailSet=false;
        Require(rejected.Destroyed && ReferenceEquals(HybridLumenGIDeformedMesh.GetSourceMesh(owner.renderer),pendingC) &&
            ReferenceEquals(HybridLumenGIDeformedMesh.GetSdf(owner.renderer,null),sdfC) && ReferenceEquals(HybridLumenGIDeformedMesh.GetCharts(owner.renderer,null),chartsC) &&
            HybridLumenGIDeformedMesh.ReservedBytes==oneReservation,"Failed publication exposed mixed/invalid state or lost the previous snapshot.");
        Disable(owner);Require(!pendingC.Destroyed && ReferenceEquals(HybridLumenGIDeformedMesh.GetSourceMesh(owner.renderer),owner.original),"Disable failed to withdraw atomically or destroyed GPU consumers immediately.");
        RenderPipelineManager.End();Require(!pendingC.Destroyed,"Disable treated fence submission as completion.");
        RecordedGpu.CompleteSubmitted();RenderPipelineManager.End();Require(pendingC.Destroyed && HybridLumenGIDeformedMesh.ReservedBytes==0,"Disabled snapshot did not retire after actual completion.");
    }
    static void FailedCaptureConsumesBudgetAndBacksOff()
    {
        Defaults();HybridLumenGIDeformedMesh.VertexBudgetPerFrame=3;
        var bad=Add(100,true);var skin=(SkinnedMeshRenderer)bad.renderer;skin.FailBake=true;bad.component.updateIntervalFrames=4;

        Frame();Require(skin.BakeCount==1 && Job(good)==null && good.component.Status=="Capture vertex budget pending" &&
            HybridLumenGIDeformedMesh.ReservedBytes==0,"Failed capture bypassed vertex admission or leaked its reservation.");
        RenderPipelineManager.Begin();Require(skin.BakeCount==1,"Repeated camera context in one frame repeated failed capture.");
        Frame();Started(200);Require(skin.BakeCount==1,"Failure retried before its capture interval.");

        Require(Job(good)==null && !HybridLumenGIDeformedMesh.HasSnapshot(good.renderer) && HybridLumenGIDeformedMesh.ReservedBytes==0,"Cancelled disabled worker published or leaked while new workers were paused.");
    }

        Console.WriteLine("Boundary: deterministic SDF/chart result stubs, recorded Unity API calls and controlled real Task.Run barriers. No SDF/chart numeric retest, Unity engine, native fence or physical GPU execution is claimed.");
    }
    static void OpenNewSession()=>typeof(HybridLumenGIResourceRetirement).GetMethod("InitializeRuntime",PrivateStatic).Invoke(null,null);
    static void ShutdownWithdrawalAndNewSession()
    {
        Defaults();var owner=Add(310);Frame();Started(310);Finish(owner,310);Frame();
        Mesh published=HybridLumenGIDeformedMesh.GetSourceMesh(owner.renderer);
        owner.component.RequestCapture(Mesh(320));Frame();Started(320);Mesh pending=PendingMesh(owner);
        int withdrawals=0,parent=0,child=0;bool read=false;
        Action withdraw=()=>{Require(HybridLumenGIResourceRetirement.AdmissionClosed,"Owner withdrew before admission closed");withdrawals++;HybridLumenGIResourceRetirement.Retire(()=>{Require(read,"Old reader lost its parent storage");parent++;HybridLumenGIResourceRetirement.Retire(()=>child++);});Require(parent==0,"Bulk withdrawal reentered drain");};
        HybridLumenGIResourceRetirement.RegisterShutdownOwner(withdraw);
        RenderPipelineManager.Reader(()=>{Require(!published.Destroyed&&parent==0,"Recorded graph lost withdrawn resources");read=true;});
        Application.Quit();Application.Quit();
        Require(withdrawals==1&&!HybridLumenGIDeformedMesh.HasSnapshot(owner.renderer)&&!published.Destroyed&&!pending.Destroyed,"Shutdown was not idempotent atomic withdrawal");
        owner.component.RequestCapture(Mesh(330));var late=Add(340);
        Require(Job(late)==null&&HybridLumenGIDeformedMesh.PendingWorkerCount==1,"Closed session admitted new deformation work");
        // Complete the original context, then parent destruction during a later EndContext
        // enqueues children. Their own context submission remains required.
        RenderPipelineManager.End();RecordedGpu.CompleteSubmitted();RenderPipelineManager.End();
        Require(parent==1&&child==0&&published.Destroyed&&!pending.Destroyed,"Parent/child fencing or worker retention failed");
        RecordedGpu.CompleteSubmitted();HybridLumenGIResourceRetirement.IdleUpdate();Require(child==1,"EndContext nested retirement lost its completion");
        long held=HybridLumenGIDeformedMesh.ReservedBytes;Require(held>0,"Cancelled running worker lost reservation");
        OpenNewSession();Require(!HybridLumenGIResourceRetirement.AdmissionClosed&&HybridLumenGIDeformedMesh.ReservedBytes==held,"New session cleared old ownership");
        Finish(owner,320);HybridLumenGIResourceRetirement.IdleUpdate();
        Require(pending.Destroyed&&HybridLumenGIDeformedMesh.ReservedBytes==0&&!HybridLumenGIDeformedMesh.HasSnapshot(owner.renderer),"Old cancelled result published or survived new-session cleanup");
        Disable(owner);Disable(late);HybridLumenGIResourceRetirement.UnregisterShutdownOwner(withdraw);
        Require(HybridLumenGIResourceRetirement.OwnedCount==0,"Late owner disable enqueued duplicate retirement");
        Console.WriteLine("PASS: close-before-withdraw; no new deformation admission; old recorded reader completion; parent/child EndContext fences; cancelled workers retain storage through new session and cannot publish; late disable idempotence.");
    }
    sealed class Completion:HybridLumenGIAssetPageStreaming.ICompletion

    static void ShutdownCompletionProgress()
    {
        OpenNewSession();HybridLumenGIResourceRetirement.IdleUpdate();
        Require(HybridLumenGIResourceRetirement.OwnedCount==0,"Progress case requires clean ownership");
        int parent=0,child=0,firstFence=RecordedGpu.Fences.Count;
        RecordedGpu.CompleteNewFenceOnQuery=2;
        Action withdraw=()=>HybridLumenGIResourceRetirement.Retire(()=>{parent++;HybridLumenGIResourceRetirement.Retire(()=>child++);});
        HybridLumenGIResourceRetirement.RegisterShutdownOwner(withdraw);Application.Quit();
        HybridLumenGIResourceRetirement.UnregisterShutdownOwner(withdraw);RecordedGpu.CompleteNewFenceOnQuery=0;
        Console.WriteLine("Completion-transition shutdown: parent="+parent+", child="+child+", owned="+HybridLumenGIResourceRetirement.OwnedCount+", pending="+HybridLumenGIResourceRetirement.PendingCount);
        Require(parent==1&&child==1&&HybridLumenGIResourceRetirement.OwnedCount==0&&HybridLumenGIResourceRetirement.PendingCount==0,"A completed fence stranded its parent/child destroy callbacks");
        Require(RecordedGpu.Fences.Count==firstFence+2,"Child did not receive a separate submission");
        for(int i=firstFence;i<RecordedGpu.Fences.Count;i++)Require(RecordedGpu.Fences[i].Submitted&&RecordedGpu.Fences[i].Completed&&RecordedGpu.Fences[i].QueryCount>=2,"Adapter completed before a submitted fence was actually queried");
        foreach(bool throws in new[]{false,true})
        {
            OpenNewSession();HybridLumenGIResourceRetirement.IdleUpdate();RecordedGpu.AutoCompleteImmediate=true;
            int destroyed=0;var prerequisite=new Completion{ThrowQuery=throws};
            HybridLumenGIResourceRetirement.Retire(()=>destroyed++,prerequisite);
            Application.Quit();
            Require(destroyed==0&&HybridLumenGIResourceRetirement.OwnedCount==1&&prerequisite.Queries<=3,"Unready/throwing prerequisite was released or synchronously busy-waited");
            prerequisite.ThrowQuery=false;prerequisite.Ready=true;HybridLumenGIResourceRetirement.IdleUpdate();
            Require(destroyed==1&&HybridLumenGIResourceRetirement.OwnedCount==0,"Retained prerequisite owner could not finish after external completion");
        }
        RecordedGpu.AutoCompleteImmediate=false;OpenNewSession();
        Console.WriteLine("PASS: deterministic submitted fence false-to-true transition completes parent and separately fenced child in one shutdown; unready/throwing prerequisites retain ownership without busy waiting.");
    }
    static void DetachedPassRetirement()
    {
        OpenNewSession();HybridLumenGIResourceRetirement.IdleUpdate();
        int parent=0,child=0,readers=0;long bytes=640;int firstFence=RecordedGpu.Fences.Count;
        RenderPipelineManager.Begin();
        RenderPipelineManager.Reader(()=>{Require(parent==0&&bytes==640,"Detached owner lost storage before last reader");readers++;});
        RenderPipelineManager.SubmitReadersBeforeEnd();RenderPipelineManager.End();
        HybridLumenGIResourceRetirement.RetireDetachedPass(()=>{
            Require(readers==1,"Detached owner destroyed before submitted reader completed");parent++;
            HybridLumenGIResourceRetirement.Retire(()=>{child++;bytes=0;});
        });
        // No idle certificate: shutdown retains ownership without manufacturing a context.
        Application.Quit();Require(parent==0&&bytes==640&&RecordedGpu.Fences.Count==firstFence,"Shutdown bypassed detached-pass idle boundary");
        // Even a genuine context alone cannot submit this dedicated parent ticket.
        RenderPipelineManager.Begin();RenderPipelineManager.End();
        Require(RecordedGpu.Fences.Count==firstFence,"Detached pass entered the generic context batch");
        HybridLumenGIResourceRetirement.IdleUpdate();
        Require(parent==0&&child==0&&bytes==640&&RecordedGpu.Fences.Count==firstFence+1,"Idle was mistaken for parent completion");
        RecordedGpu.CompleteSubmitted();HybridLumenGIResourceRetirement.IdleUpdate();
        Require(parent==1&&child==0&&bytes==640&&RecordedGpu.Fences.Count==firstFence+2,"Child lost its separate fence/accounting");
        RecordedGpu.CompleteSubmitted();HybridLumenGIResourceRetirement.IdleUpdate();HybridLumenGIResourceRetirement.IdleUpdate();
        Require(parent==1&&child==1&&bytes==0&&HybridLumenGIResourceRetirement.OwnedCount==0&&HybridLumenGIResourceRetirement.PendingCount==0,"Detached parent/child not exact once");
        // Already certified idle shutdown must finish without requiring another Update.
        OpenNewSession();HybridLumenGIResourceRetirement.IdleUpdate();RecordedGpu.AutoCompleteImmediate=true;
        Action withdraw=()=>HybridLumenGIResourceRetirement.RetireDetachedPass(()=>parent++);
        HybridLumenGIResourceRetirement.RegisterShutdownOwner(withdraw);Application.Quit();
        HybridLumenGIResourceRetirement.UnregisterShutdownOwner(withdraw);RecordedGpu.AutoCompleteImmediate=false;
        Require(parent==2&&HybridLumenGIResourceRetirement.OwnedCount==0,"Known idle shutdown unnecessarily awaited a future Update");
        OpenNewSession();Console.WriteLine("PASS: detached whole-pass owner waits for real idle and submitted last reader; parent and child complete separately/exactly once with retained bytes; no-idle shutdown retained, known-idle shutdown progresses.");
    }
    static void IdleAndShutdownRetirement()
    {
        RenderPipelineManager.End();HybridLumenGIResourceRetirement.IdleUpdate();
        int destroyed=0;var prerequisite=new Completion();
        HybridLumenGIResourceRetirement.Retire(()=>destroyed++,prerequisite);
        HybridLumenGIResourceRetirement.IdleUpdate();Require(destroyed==0,"Immediate submission was mistaken for completion.");
        var fence=RecordedGpu.Fences[^1];Require(fence.Immediate,"Idle did not submit only its own immediate fence.");
        fence.Completed=true;HybridLumenGIResourceRetirement.IdleUpdate();Require(destroyed==0,"Unfinished prerequisite was discarded.");
        prerequisite.Ready=true;fence.FailQuery=true;HybridLumenGIResourceRetirement.IdleUpdate();Require(destroyed==0,"Failed completion query authorized destruction.");
        fence.FailQuery=false;HybridLumenGIResourceRetirement.IdleUpdate();HybridLumenGIResourceRetirement.IdleUpdate();Require(destroyed==1,"Idle completion failed exact-once retirement without a render context.");

        // Two recorded readers share one context. Retirement occurs only after both are withdrawn.
        RenderPipelineManager.Begin();bool readerA=false,readerB=false;
        RenderPipelineManager.Reader(()=>{Require(destroyed==1,"First camera read destroyed storage.");readerA=true;});
        HybridLumenGIResourceRetirement.Retire(()=>{Require(readerA&&readerB,"Released before both readers completed.");destroyed++;});
        RenderPipelineManager.Reader(()=>{Require(destroyed==1,"Second existing camera reader lost its storage.");readerB=true;});
        RenderPipelineManager.End();HybridLumenGIResourceRetirement.IdleUpdate();
        Require(destroyed==1&&!readerA&&!readerB,"Submission completed a reader or freed its storage.");
        RecordedGpu.CompleteSubmitted();HybridLumenGIResourceRetirement.IdleUpdate();Require(destroyed==2,"Submitted context ticket required another render context to poll.");

        // A later end-event listener is still ambiguous even though our own callback returned.
        RenderPipelineManager.Begin();RenderPipelineManager.End();int before=RecordedGpu.Fences.Count;
        HybridLumenGIResourceRetirement.Retire(()=>destroyed++);HybridLumenGIResourceRetirement.IdleUpdate();
        Require(RecordedGpu.Fences.Count==before&&destroyed==2,"End-event tail was incorrectly authorized for immediate submission.");
        RenderPipelineManager.Begin();RenderPipelineManager.End();RecordedGpu.CompleteSubmitted();HybridLumenGIResourceRetirement.IdleUpdate();Require(destroyed==3,"A genuine subsequent context failed to cover the ambiguous tail.");

        // Shutdown shares one deadline, and late Retire reuses it without re-entering the drain.
        RecordedGpu.AutoCompleteImmediate=true;Application.Quit();
        long deadline=(long)typeof(HybridLumenGIResourceRetirement).GetField("shutdownDeadline",PrivateStatic).GetValue(null);
        HybridLumenGIResourceRetirement.Retire(()=>{destroyed++;HybridLumenGIResourceRetirement.Retire(()=>destroyed++);});
        Application.Quit();Require(destroyed==5&&HybridLumenGIResourceRetirement.OwnedCount==0,"Late shutdown or reentrant retirement was lost/destroyed twice.");
        Require(deadline==(long)typeof(HybridLumenGIResourceRetirement).GetField("shutdownDeadline",PrivateStatic).GetValue(null),"Repeated shutdown restarted its time budget.");
        // Enter shutdown while a context is active, then finish the parent in ordinary idle.
        // Its nested Retire must not recursively destroy the still-listed parent or spin.
        RenderPipelineManager.Begin();int parent=0,child=0;
        HybridLumenGIResourceRetirement.Retire(()=>{parent++;HybridLumenGIResourceRetirement.Retire(()=>child++);});
        RenderPipelineManager.End();RecordedGpu.CompleteSubmitted();
        var elapsed=System.Diagnostics.Stopwatch.StartNew();HybridLumenGIResourceRetirement.IdleUpdate();
        Require(parent==1&&child==1&&elapsed.ElapsedMilliseconds<1000,"Ordinary PollOwned reentered destruction or spun inside shutdown drain.");
        int attempts=0;HybridLumenGIResourceRetirement.Retire(()=>{attempts++;throw new Exception("Controlled destruction failure");});
        HybridLumenGIResourceRetirement.IdleUpdate();Application.Quit();Require(attempts==1&&HybridLumenGIResourceRetirement.FailedCount==1&&Debug.Exceptions.Count==1,"Failed native destruction was retried or hidden.");
        RecordedGpu.AutoCompleteImmediate=false;
    }

    static void UnknownSubmissionIsNotRepaired()

        Console.WriteLine("PASS: actual retirement source: idle completion, prerequisite/query failure, two-reader fence, ambiguous event tail, late/reentrant shutdown, shared deadline, failed destroy, and unknown context submission retained.");
    }
