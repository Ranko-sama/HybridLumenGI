    static Mesh Mesh(int marker)
    {
        var mesh=new Mesh{vertices=new[]{new Vector3(marker,0,0),new Vector3(marker,1,0),new Vector3(marker,0,1)},
            normals=new Vector3[3],tangents=new Vector4[3],uv=new Vector2[3],bounds=new Bounds(new Vector3(marker,0,0),new Vector3(1,1,1)),subMeshCount=1};
        mesh.SetIndices(new[]{0,1,2},MeshTopology.Triangles,0,false);return mesh;
    }
    static Owner Add(int marker,bool skinned=false)
    {
        var go=new GameObject();var proxy=go.Add<HybridLumenGIProxy>();Renderer renderer;
        Mesh mesh=Mesh(marker);
        if(skinned){var skin=go.Add<SkinnedMeshRenderer>();skin.sharedMesh=mesh;renderer=skin;}
        else{renderer=go.Add<MeshRenderer>();go.Add<MeshFilter>().sharedMesh=mesh;}
        proxy.SourceRenderer=renderer;var component=go.Add<HybridLumenGIDeformedMesh>();component.sdfResolution=16;component.updateIntervalFrames=1;
        typeof(HybridLumenGIDeformedMesh).GetMethod("OnEnable",PrivateInstance).Invoke(component,null);
        return new Owner{component=component,proxy=proxy,renderer=renderer,original=mesh};
    }
    static void Disable(Owner owner)
    {
        owner.proxy.OnDirty=null;owner.component.enabled=false;
        typeof(HybridLumenGIDeformedMesh).GetMethod("OnDisable",PrivateInstance).Invoke(owner.component,null);
    }
    static void Frame(){Time.frameCount++;RenderPipelineManager.Begin();}
    static object Job(Owner owner)
    {

        foreach(var job in jobs)if(ReferenceEquals(job.GetType().GetField("owner",PrivateInstance).GetValue(job),owner.component))return job;
        return null;
    }
    static Mesh PendingMesh(Owner owner)
    {
        var job=Job(owner);var snapshot=job.GetType().GetField("snapshot",PrivateInstance).GetValue(job);
        return (Mesh)snapshot.GetType().GetField("mesh",PrivateInstance).GetValue(snapshot);
    }

    }

    static void LimitCaptureTime()
    {

    static void SpendCaptureTime()
    {
        // A lower-bound delay inside the existing capture hook, not a performance assertion
        // or an injected product clock. One whole attempt exceeds the 0.1 ms allowance.
        var elapsed=System.Diagnostics.Stopwatch.StartNew();
        while(elapsed.Elapsed.TotalMilliseconds<1)System.Threading.Thread.SpinWait(32);
    }
    static void CaptureTimeBudgetDefersAndPublishes()
    {
        Defaults();LimitCaptureTime();HybridLumenGIDeformedMesh.VertexBudgetPerFrame=6;
        var first=Add(1100,true);var second=Add(1101,true);
        var firstSkin=(SkinnedMeshRenderer)first.renderer;var secondSkin=(SkinnedMeshRenderer)second.renderer;
        var requested=typeof(HybridLumenGIDeformedMesh).GetField("requested",PrivateInstance);

        string oldStatus=second.component.Status;long firstReservation=0;int publications=0;Mesh firstMesh=null;
        firstSkin.Captured=_=>{firstReservation=HybridLumenGIDeformedMesh.ReservedBytes;SpendCaptureTime();};
        secondSkin.Captured=_=>
        {
            Require(publications==1&&Job(first)==null&&ReferenceEquals(HybridLumenGIDeformedMesh.GetSourceMesh(first.renderer),firstMesh),
                "New capture ran before the previously completed job was published.");
            SpendCaptureTime();
        };
        typeof(HybridLumenGIDeformedMesh).GetField("cursor",PrivateStatic).SetValue(null,0);
        Frame();Started(1100);firstMesh=PendingMesh(first);
        Require(firstSkin.BakeCount==1&&first.component.PendingFrame==Time.frameCount&&Job(second)==null&&secondSkin.BakeCount==0&&
            second.component.PendingFrame==-1&&second.component.Status==oldStatus&&
            object.Equals(requested.GetValue(second.component),oldRequest)&&object.Equals(generation.GetValue(second.component),oldGeneration)&&
            object.Equals(lastCapture.GetValue(second.component),oldCapture)&&
            (int)typeof(HybridLumenGIDeformedMesh).GetField("cursor",PrivateStatic).GetValue(null)==1&&
            firstReservation>0&&HybridLumenGIDeformedMesh.ReservedBytes==firstReservation&&
            HybridLumenGIDeformedMesh.VertexBudgetPerFrame==6&&HybridLumenGIDeformedMesh.MaximumWorkers==2,
            "Exceeded allowance consumed the deferred owner's turn, request, interval or reservation.");
        RenderPipelineManager.Begin();
        Require(firstSkin.BakeCount==1&&secondSkin.BakeCount==0,"Another camera context reopened the exhausted engine-frame allowance.");
        Finish(first,1100);
        // Publication itself exceeds the capture allowance, but is outside that allowance.
        first.proxy.OnDirty=()=>{publications++;SpendCaptureTime();};
        Frame();Started(1101);
        Require(publications==1&&firstSkin.BakeCount==1&&secondSkin.BakeCount==1&&second.component.PendingFrame==Time.frameCount&&
            (int)typeof(HybridLumenGIDeformedMesh).GetField("cursor",PrivateStatic).GetValue(null)==2,
            "Next frame failed to publish completed work and resume the deferred owner's turn.");
        PublishCompleted(second,1101);

            "Enabled time allowance or paused new captures blocked completed-job publication.");
        Disable(first);Disable(second);DrainCompletedRetirements();
        Require(HybridLumenGIDeformedMesh.ReservedBytes==0,"Time-limited successful captures failed normal retirement.");Defaults();
    }
