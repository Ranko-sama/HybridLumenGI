    static Mesh Mesh(int marker)
    {
        var mesh=new Mesh{vertices=new[]{new Vector3(marker,0,0),new Vector3(marker,1,0),new Vector3(marker,0,1)},
            normals=new Vector3[3],tangents=new Vector4[3],uv=new Vector2[3],bounds=new Bounds(new Vector3(marker,0,0),new Vector3(1,1,1)),subMeshCount=1};
        mesh.SetIndices(new[]{0,1,2},MeshTopology.Triangles,0,false);return mesh;
    }
    static Owner Add(int marker,bool skinned=false)
    {
        var go=new GameObject();var proxy=go.Add<HybridLumenGIProxy>();Renderer renderer;
        Mesh mesh=Mesh(marker);
        if(skinned){var skin=go.Add<SkinnedMeshRenderer>();skin.sharedMesh=mesh;renderer=skin;}
        else{renderer=go.Add<MeshRenderer>();go.Add<MeshFilter>().sharedMesh=mesh;}
        proxy.SourceRenderer=renderer;var component=go.Add<HybridLumenGIDeformedMesh>();component.sdfResolution=16;component.updateIntervalFrames=1;
        typeof(HybridLumenGIDeformedMesh).GetMethod("OnEnable",PrivateInstance).Invoke(component,null);
        return new Owner{component=component,proxy=proxy,renderer=renderer,original=mesh};
    }
    static void Disable(Owner owner)
    {
        owner.proxy.OnDirty=null;owner.component.enabled=false;
        typeof(HybridLumenGIDeformedMesh).GetMethod("OnDisable",PrivateInstance).Invoke(owner.component,null);
    }
    static void Frame(){Time.frameCount++;RenderPipelineManager.Begin();}
    static object Job(Owner owner)
    {
        var jobs=(IList)typeof(HybridLumenGIDeformedMesh).GetField("Jobs",PrivateStatic).GetValue(null);
        foreach(var job in jobs)if(ReferenceEquals(job.GetType().GetField("owner",PrivateInstance).GetValue(job),owner.component))return job;
        return null;
    }
    static Mesh PendingMesh(Owner owner)
    {
        var job=Job(owner);var snapshot=job.GetType().GetField("snapshot",PrivateInstance).GetValue(job);
        return (Mesh)snapshot.GetType().GetField("mesh",PrivateInstance).GetValue(snapshot);
    }
    static void Started(int marker)=>Require(ControlledBake.For(marker).Started.Wait(5000),"Worker did not reach controlled bake: "+marker);

    {

        var work=(Task)job.GetType().GetField("work",PrivateInstance).GetValue(job);
        ControlledBake.For(marker).Release.Set();Require(work.Wait(5000),"Worker did not finish after its barrier.");
    }

    static void AtomicGenerationAndRetirement()
    {
        Defaults();var owner=Add(10);HybridLumenGIDeformedMesh.MaximumWorkers=0;Frame();
        Require(Job(owner)==null && owner.component.PendingFrame==-1,"MaximumWorkers=0 still admitted a capture.");
        HybridLumenGIDeformedMesh.MaximumWorkers=1;Frame();Started(10);
        Mesh pendingA=PendingMesh(owner);Require(!ReferenceEquals(pendingA,owner.original),"Capture aliased producer-owned mesh.");
        Finish(owner,10);HybridLumenGIDeformedMesh.UploadBudgetBytesPerFrame=0;Frame();
        Require(!HybridLumenGIDeformedMesh.HasSnapshot(owner.renderer) && Job(owner)!=null && Texture3D.Created==0,"Completed worker bypassed the publication upload budget.");
        int preparations=0,publications=0;
        Action beforeSet=()=>{preparations++;Require(ReferenceEquals(HybridLumenGIDeformedMesh.GetSourceMesh(owner.renderer),owner.original),"Partially built snapshot became visible before atomic publication.");};
        HybridLumenGIMeshSdf.BeforeSet=beforeSet;HybridLumenGISurfaceCharts.BeforeSet=beforeSet;
        owner.proxy.OnDirty=()=>
        {
            publications++;Mesh mesh=HybridLumenGIDeformedMesh.GetSourceMesh(owner.renderer);
            var sdf=HybridLumenGIDeformedMesh.GetSdf(owner.renderer,null);var charts=HybridLumenGIDeformedMesh.GetCharts(owner.renderer,null);
            Require(ReferenceEquals(mesh,pendingA) && ReferenceEquals(sdf.SourceMesh,mesh) && ReferenceEquals(charts.SourceMesh,mesh) && charts.CaptureMesh!=null,"Mesh/SDF/charts were not published as one snapshot.");
        };
        HybridLumenGIDeformedMesh.UploadBudgetBytesPerFrame=8L<<20;Frame();
        Require(preparations==2 && publications==1 && owner.component.PublishedFrame>=0,"Snapshot construction/publication observation missing.");
        Require(ReferenceEquals(owner.renderer.GetComponent<MeshFilter>().sharedMesh,owner.original),"GI publication mutated the visible renderer's original mesh.");
        HybridLumenGIMeshSdf.BeforeSet=HybridLumenGISurfaceCharts.BeforeSet=null;owner.proxy.OnDirty=null;
        long oneReservation=HybridLumenGIDeformedMesh.ReservedBytes;
        var sdfA=HybridLumenGIDeformedMesh.GetSdf(owner.renderer,null);var chartsA=HybridLumenGIDeformedMesh.GetCharts(owner.renderer,null);

        owner.component.RequestCapture(Mesh(20));Frame();Started(20);Mesh pendingB=PendingMesh(owner);
        owner.component.RequestCapture(Mesh(30));Finish(owner,20);Frame();Started(30);
        Require(pendingB.Destroyed && ReferenceEquals(HybridLumenGIDeformedMesh.GetSourceMesh(owner.renderer),pendingA),"Cancelled older generation replaced the live snapshot or leaked its private mesh.");
        Mesh pendingC=PendingMesh(owner);Finish(owner,30);Frame();
        var sdfC=HybridLumenGIDeformedMesh.GetSdf(owner.renderer,null);var chartsC=HybridLumenGIDeformedMesh.GetCharts(owner.renderer,null);
        Require(ReferenceEquals(HybridLumenGIDeformedMesh.GetSourceMesh(owner.renderer),pendingC) &&
            ReferenceEquals(sdfC.SourceMesh,pendingC) && ReferenceEquals(chartsC.SourceMesh,pendingC),"Replacement snapshot published mixed identities.");
        Require(!pendingA.Destroyed && !sdfA.Destroyed && !chartsA.Destroyed && HybridLumenGIDeformedMesh.ReservedBytes==2*oneReservation,"Replacement destroyed/released old snapshot before final-context completion.");
        int previousFences=RecordedGpu.Fences.Count;RenderPipelineManager.End();
        Require(RecordedGpu.Fences.Count==previousFences+1 && RecordedGpu.Submits>0 && !pendingA.Destroyed,"Retirement did not submit a final-context fence or assumed it completed immediately.");
        Frame();RenderPipelineManager.End();
        Require(!pendingA.Destroyed && HybridLumenGIDeformedMesh.ReservedBytes==2*oneReservation,"Elapsed context/frame released unfinished GPU work.");
        RecordedGpu.CompleteSubmitted();RenderPipelineManager.End();
        Require(pendingA.Destroyed && sdfA.Destroyed && sdfA.Texture.Destroyed && chartsA.Destroyed && chartsA.CaptureMesh.Destroyed &&
            !pendingC.Destroyed && HybridLumenGIDeformedMesh.ReservedBytes==oneReservation,"Completed old snapshot did not retire all owned resources exactly once.");

        owner.component.RequestCapture(Mesh(40));Frame();Started(40);Mesh rejected=PendingMesh(owner);Finish(owner,40);
        HybridLumenGISurfaceCharts.FailSet=true;Frame();HybridLumenGISurfaceCharts.FailSet=false;
        Require(rejected.Destroyed && ReferenceEquals(HybridLumenGIDeformedMesh.GetSourceMesh(owner.renderer),pendingC) &&
            ReferenceEquals(HybridLumenGIDeformedMesh.GetSdf(owner.renderer,null),sdfC) && ReferenceEquals(HybridLumenGIDeformedMesh.GetCharts(owner.renderer,null),chartsC) &&
            HybridLumenGIDeformedMesh.ReservedBytes==oneReservation,"Failed publication exposed mixed/invalid state or lost the previous snapshot.");
        Disable(owner);Require(!pendingC.Destroyed && ReferenceEquals(HybridLumenGIDeformedMesh.GetSourceMesh(owner.renderer),owner.original),"Disable failed to withdraw atomically or destroyed GPU consumers immediately.");

    }

    static void LimitCaptureTime()
    {
        Require(CaptureTimeBudget!=null,"Current source lacks the capture-time budget API.");
        CaptureTimeBudget.SetValue(null,.1d);
    }
    static void SpendCaptureTime()
    {
        // A lower-bound delay inside the existing capture hook, not a performance assertion
        // or an injected product clock. One whole attempt exceeds the 0.1 ms allowance.
        var elapsed=System.Diagnostics.Stopwatch.StartNew();
        while(elapsed.Elapsed.TotalMilliseconds<1)System.Threading.Thread.SpinWait(32);
    }
    static void CaptureTimeBudgetDefersAndPublishes()
    {
        Defaults();LimitCaptureTime();HybridLumenGIDeformedMesh.VertexBudgetPerFrame=6;
        var first=Add(1100,true);var second=Add(1101,true);
        var firstSkin=(SkinnedMeshRenderer)first.renderer;var secondSkin=(SkinnedMeshRenderer)second.renderer;
        var requested=typeof(HybridLumenGIDeformedMesh).GetField("requested",PrivateInstance);
        var generation=typeof(HybridLumenGIDeformedMesh).GetField("generation",PrivateInstance);
        var lastCapture=typeof(HybridLumenGIDeformedMesh).GetField("lastCapture",PrivateInstance);
        object oldRequest=requested.GetValue(second.component),oldGeneration=generation.GetValue(second.component),oldCapture=lastCapture.GetValue(second.component);
        string oldStatus=second.component.Status;long firstReservation=0;int publications=0;Mesh firstMesh=null;
        firstSkin.Captured=_=>{firstReservation=HybridLumenGIDeformedMesh.ReservedBytes;SpendCaptureTime();};
        secondSkin.Captured=_=>
        {
            Require(publications==1&&Job(first)==null&&ReferenceEquals(HybridLumenGIDeformedMesh.GetSourceMesh(first.renderer),firstMesh),
                "New capture ran before the previously completed job was published.");
            SpendCaptureTime();
        };
        typeof(HybridLumenGIDeformedMesh).GetField("cursor",PrivateStatic).SetValue(null,0);
        Frame();Started(1100);firstMesh=PendingMesh(first);
        Require(firstSkin.BakeCount==1&&first.component.PendingFrame==Time.frameCount&&Job(second)==null&&secondSkin.BakeCount==0&&
            second.component.PendingFrame==-1&&second.component.Status==oldStatus&&
            object.Equals(requested.GetValue(second.component),oldRequest)&&object.Equals(generation.GetValue(second.component),oldGeneration)&&
            object.Equals(lastCapture.GetValue(second.component),oldCapture)&&
            (int)typeof(HybridLumenGIDeformedMesh).GetField("cursor",PrivateStatic).GetValue(null)==1&&
            firstReservation>0&&HybridLumenGIDeformedMesh.ReservedBytes==firstReservation&&
            HybridLumenGIDeformedMesh.VertexBudgetPerFrame==6&&HybridLumenGIDeformedMesh.MaximumWorkers==2,
            "Exceeded allowance consumed the deferred owner's turn, request, interval or reservation.");
        RenderPipelineManager.Begin();
        Require(firstSkin.BakeCount==1&&secondSkin.BakeCount==0,"Another camera context reopened the exhausted engine-frame allowance.");
        Finish(first,1100);
        // Publication itself exceeds the capture allowance, but is outside that allowance.
        first.proxy.OnDirty=()=>{publications++;SpendCaptureTime();};
        Frame();Started(1101);
        Require(publications==1&&firstSkin.BakeCount==1&&secondSkin.BakeCount==1&&second.component.PendingFrame==Time.frameCount&&
            (int)typeof(HybridLumenGIDeformedMesh).GetField("cursor",PrivateStatic).GetValue(null)==2,
            "Next frame failed to publish completed work and resume the deferred owner's turn.");
        PublishCompleted(second,1101);
        Require(Job(second)==null&&HybridLumenGIDeformedMesh.HasSnapshot(second.renderer),
            "Enabled time allowance or paused new captures blocked completed-job publication.");
        Disable(first);Disable(second);DrainCompletedRetirements();
        Require(HybridLumenGIDeformedMesh.ReservedBytes==0,"Time-limited successful captures failed normal retirement.");Defaults();
    }

    static void VertexBudgetKeepsDeferredCaptureTurn()
    {
        Defaults();
        const int unchangedMarker=1620, oldPose=1621, newPose=1622, fittingMarker=1623;
        var unchanged=Add(unchangedMarker,true);
        var moving=Add(oldPose,true);
        var vertices=moving.original.vertices;
        Array.Resize(ref vertices,4);
        vertices[3]=new Vector3(oldPose,1,1);
        moving.original.vertices=vertices;
        moving.original.normals=new Vector3[4];
        moving.original.tangents=new Vector4[4];
        moving.original.uv=new Vector2[4];
        moving.original.SetIndices(new[]{0,1,2,1,3,2},MeshTopology.Triangles,0,false);

        Frame();Started(unchangedMarker);Started(oldPose);
        PublishCompleted(unchanged,unchangedMarker);PublishCompleted(moving,oldPose);
        Mesh oldMesh=HybridLumenGIDeformedMesh.GetSourceMesh(moving.renderer);
        var unchangedSkin=(SkinnedMeshRenderer)unchanged.renderer;
        var movingSkin=(SkinnedMeshRenderer)moving.renderer;
        movingSkin.Captured=mesh=>
        {
            Vector3[] baked=mesh.vertices;
            for(int i=0;i<baked.Length;i++)baked[i].x+=1;
            mesh.vertices=baked;
        };
        var fitting=Add(fittingMarker);
        HybridLumenGIDeformedMesh.VertexBudgetPerFrame=6;
        typeof(HybridLumenGIDeformedMesh).GetField("cursor",PrivateStatic).SetValue(null,0);
        Frame();
        Require(unchangedSkin.BakeCount==2 && Job(unchanged)==null && movingSkin.BakeCount==1 && Job(moving)==null &&
            moving.component.Status=="Capture vertex budget pending" && Job(fitting)!=null,
            "The unchanged capture must charge its vertices while leaving the remaining budget available to a fitting capture.");
        PublishCompleted(fitting,fittingMarker);

        Frame();
        Require(movingSkin.BakeCount==2 && Job(moving)!=null && unchangedSkin.BakeCount==2 &&
            ReferenceEquals(HybridLumenGIDeformedMesh.GetSourceMesh(moving.renderer),oldMesh),
            "The owner deferred by the vertex budget did not get its capture turn in the next eligible frame.");
        Mesh newMesh=PendingMesh(moving);
        PublishCompleted(moving,newPose);
        PublishedTogether(moving,newMesh,newPose);
        Frame();
        Require(unchangedSkin.BakeCount==3 && movingSkin.BakeCount==2,
            "The original owner must resume without exceeding the frame vertex budget.");
        Disable(unchanged);Disable(moving);Disable(fitting);DrainCompletedRetirements();
        Require(HybridLumenGIDeformedMesh.ReservedBytes==0,"Capture turn regression left a reservation.");
        Defaults();
        Console.WriteLine("PASS: a vertex-budget waiter captures and publishes its changed pose while fitting work uses the remaining vertices.");
    }
    static void CompletedPublicationIsFair()
    {
        Defaults();HybridLumenGIDeformedMesh.MaximumWorkers=1;
        long Upload(Owner owner)
        {
            var job=Job(owner);var snapshot=job.GetType().GetField("snapshot",PrivateInstance).GetValue(job);
            return (long)snapshot.GetType().GetField("uploadBytes",PrivateInstance).GetValue(snapshot);
        }
        // Queue an unfinished older job, then a younger job which completes first.
        var older=Add(1400);Frame();Started(1400);Mesh olderMesh=PendingMesh(older);
        var refresh=Add(1401);HybridLumenGIDeformedMesh.MaximumWorkers=2;Frame();Started(1401);
        Mesh firstRefresh=PendingMesh(refresh);long oneUpload=Upload(older);
        Require(oneUpload>0&&Upload(refresh)==oneUpload,"Fairness inputs must have equal actual upload reservations.");
        HybridLumenGIDeformedMesh.UploadBudgetBytesPerFrame=oneUpload;
        Finish(refresh,1401);HybridLumenGIDeformedMesh.MaximumWorkers=0;Frame();
        Require(Job(older)!=null&&!HybridLumenGIDeformedMesh.HasSnapshot(older.renderer)&&Job(refresh)==null&&
            ReferenceEquals(HybridLumenGIDeformedMesh.GetSourceMesh(refresh.renderer),firstRefresh),
            "An unfinished older worker blocked a later ready snapshot.");

        // The old worker remains queued while a third owner and a newer demand from
        // the already-published owner both capture. Completion order is deliberately reversed.
        var later=Add(1402);HybridLumenGIDeformedMesh.MaximumWorkers=2;Frame();Started(1402);
        Mesh laterMesh=PendingMesh(later);
        refresh.component.RequestCapture(Mesh(1403));HybridLumenGIDeformedMesh.MaximumWorkers=3;Frame();Started(1403);
        Mesh refreshedMesh=PendingMesh(refresh);
        Require(Upload(later)==oneUpload&&Upload(refresh)==oneUpload,"Newer captures changed the one-snapshot upload premise.");
        Finish(refresh,1403);Finish(later,1402);Finish(older,1400);
        HybridLumenGIDeformedMesh.MaximumWorkers=0;Frame();
        bool olderPublished=ReferenceEquals(HybridLumenGIDeformedMesh.GetSourceMesh(older.renderer),olderMesh);
        bool laterPublished=HybridLumenGIDeformedMesh.HasSnapshot(later.renderer);
        bool refreshPublished=ReferenceEquals(HybridLumenGIDeformedMesh.GetSourceMesh(refresh.renderer),refreshedMesh);
        Require(olderPublished&&!laterPublished&&!refreshPublished&&Job(older)==null&&Job(later)!=null&&Job(refresh)!=null&&
            ReferenceEquals(HybridLumenGIDeformedMesh.GetSourceMesh(refresh.renderer),firstRefresh),
            "One-upload ready queue violated capture order: older="+olderPublished+", later="+laterPublished+", refresh="+refreshPublished);

        // A completed but withdrawn entry precedes the remaining ready refresh.
        // Its private cleanup must neither consume the upload allowance nor block publication.
        Disable(later);Frame();
        Require(laterMesh.Destroyed&&Job(later)==null&&Job(refresh)==null&&
            ReferenceEquals(HybridLumenGIDeformedMesh.GetSourceMesh(refresh.renderer),refreshedMesh)&&
            ReferenceEquals(HybridLumenGIDeformedMesh.GetSourceMesh(older.renderer),olderMesh),
            "Invalid completed work blocked or consumed upload allowance before a valid ready snapshot.");
        Disable(older);Disable(refresh);DrainCompletedRetirements();
        Require(HybridLumenGIDeformedMesh.ReservedBytes==0,"Fair publication/withdrawal left snapshot reservations.");Defaults();
        Console.WriteLine("PASS: equal actual upload costs: unfinished head does not block ready work; oldest ready capture wins over later owner/new demand; invalid completed entry cleans without spending publication quota.");
    }
    static void OversizedCompletionReleasesCaptureSlot()
    {
        Defaults();
        HybridLumenGIDeformedMesh.MaximumWorkers=1;
        var large=Add(1500);
        large.component.sdfResolution=32;
        Frame();Started(1500);
        var largeJob=Job(large);
        var largeMesh=PendingMesh(large);
        long largeReservation=HybridLumenGIDeformedMesh.ReservedBytes;
        var small=Add(1501);
        HybridLumenGIDeformedMesh.UploadBudgetBytesPerFrame=32L*1024;
        Frame();
        Require(Job(small)==null,"An unfinished capture must still occupy the only worker slot.");

        Finish(large,1500);
        Frame();Started(1501);
        var smallMesh=PendingMesh(small);
        Require(ReferenceEquals(Job(large),largeJob) && !largeMesh.Destroyed &&
            !HybridLumenGIDeformedMesh.HasSnapshot(large.renderer) && HybridLumenGIDeformedMesh.ReservedBytes>largeReservation,
            "The oversized completion must retain its original pending snapshot while the fitting capture starts.");
        Finish(small,1501);Frame();
        PublishedTogether(small,smallMesh,1501);
        Require(ReferenceEquals(Job(large),largeJob) && !HybridLumenGIDeformedMesh.HasSnapshot(large.renderer),
            "The old result must remain pending under the smaller upload budget.");

        HybridLumenGIDeformedMesh.UploadBudgetBytesPerFrame=8L<<20;
        Frame();
        PublishedTogether(large,largeMesh,1500);
        Require(Job(large)==null && Job(small)==null,"Both snapshots should finish without another capture.");
        Disable(large);Disable(small);DrainCompletedRetirements();
        Require(HybridLumenGIDeformedMesh.ReservedBytes==0,"Completed snapshots left a reservation after retirement.");
        Defaults();
        Console.WriteLine("PASS: an oversized completed snapshot yields the capture slot, a fitting snapshot publishes, and the retained completion publishes after the budget recovers.");
    }
