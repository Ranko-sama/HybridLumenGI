    static void Finish(Batch batch,CommandBuffer cmd)
    {
        if(batch.State==Batch.BatchState.Prepared){batch.Dispose();}
        else if(batch.State!=Batch.BatchState.Released)
        {
            cmd.ThrowAt=cmd.ThrowBeforeAt=-1;cmd.OnCall=null;cmd.NextFence=new FenceSignal();
            if(batch.State!=Batch.BatchState.SealedAwaitingFence)
                Check(batch.SealAfterConsumers(cmd,out var reason),"final same-command retirement fence "+reason);
            cmd.NextFence.Complete=true;Batch.PollRetired();
        }
        Check(batch.State==Batch.BatchState.Released,"retired");batch.Dispose();
    }

    static void RandomContractSequences()
    {
        scenarios++;var rng=new Random(0x5517);var source=new Fixture().Freeze();
        // Independent observable contract oracle: once recorded never record again; only an
        // uncanceled recording can consume; a final fence closes use; completion alone retires.
        for(int trial=0;trial<100;trial++)
        {
            var b=Prepare(source);var cmd=new CommandBuffer();bool everRecorded=false,canceled=false,sealedUse=false,released=false;
            for(int step=0;step<20;step++)
            {
                switch(rng.Next(6))
                {
                    case 0: {bool expect=!everRecorded&&!canceled;bool actual=b.Record(cmd,out _);Check(actual==expect,"oracle record");if(actual)everRecorded=true;break;}
                    case 1:Check(b.Bind(cmd,new(),0,out _)==(everRecorded&&!canceled&&!sealedUse&&!released),"oracle consume");break;
                    case 2: {bool expect=everRecorded&&!sealedUse&&!released;bool actual=b.SealAfterConsumers(cmd,out _);Check(actual==expect,"oracle seal");if(actual)sealedUse=true;break;}
                    case 3:b.Dispose();canceled=true;if(!everRecorded)released=true;break;
                    case 4:Batch.PollRetired();Check((b.State==Batch.BatchState.Released)==released,"no implicit age retirement");break;
                    case 5:if(sealedUse){cmd.NextFence.Complete=true;Batch.PollRetired();released=true;}break;
                }
                Check(Batch.TotalAllocatedBytes==(released?0:352),"oracle exact live byte accounting");
            }
            if(!released){if(!everRecorded)b.Dispose();else{if(!sealedUse)Check(b.SealAfterConsumers(cmd,out _),"final oracle fence");cmd.NextFence.Complete=true;Batch.PollRetired();}}
            Clean();
        }
    }

        Check(AsyncGPUReadback.Pending.Count == 1, "original upload needs its actual readback callback");
        AsyncGPUReadback.CompleteNext(readback);
        Batch.PollRetired();
        long originalArenaBytes = Pool.TotalAllocatedBytes;

        var expanded = Prepare(shared.Freeze(), large.Freeze(19));
        Check(expanded.TextureUploadCount == 2 && expanded.TextureGeneration != originalHandle.y,
            "larger batch must create a distinct matching page because it cannot fit in the original arena");
        var expandedCommands = new CommandBuffer();
        RecordBatch(expanded, expandedCommands);
        Finish(expanded, expandedCommands);
        Check(AsyncGPUReadback.Pending.Count == 1, "expanded arena readback deliberately remains in flight");
        long heldArenas = Pool.TotalAllocatedBytes;

        var repeat = Prepare(shared.Freeze());
        var repeatCommands = new CommandBuffer();
        RecordBatch(repeat, repeatCommands);
        var usedHandle = Table(repeatCommands)[0].textures0;
        Console.WriteLine($"Matching duplicate pending, original generation={originalHandle.y}, repeated Batch generation={usedHandle.y}.");
        Check(repeat.TextureUploadCount == 0 && usedHandle.x == originalHandle.x && usedHandle.y == originalHandle.y &&
            repeat.GetResources(out _, out _, out var actualMetadata, out var actualTexels) &&
            ReferenceEquals(actualMetadata, originalMetadata) && ReferenceEquals(actualTexels, originalTexels) &&
            Pool.TotalAllocatedBytes == heldArenas && AsyncGPUReadback.Pending.Count == 1,
            "a pending duplicate replaced the current ready map or forced another allocation");
        Finish(repeat, repeatCommands);

        long reducedBudget = heldArenas - originalArenaBytes + 272 + 80;
        Check(Batch.TryCreate(new[]{shared.Freeze()}, reducedBudget, out var deferred, out _),
            "an arena that no longer fits prevented the original pending-page fallback");
        var deferredCommands = new CommandBuffer();
        RecordBatch(deferred, deferredCommands);
        Check(Table(deferredCommands)[0].textures0.y == 0 && deferred.TextureUploadCount == 0 &&
            Batch.TotalAllocatedBytes == reducedBudget && AsyncGPUReadback.Pending.Count == 1,
            "an unfittable ready arena bypassed the budget or duplicated the pending page");
        Finish(deferred, deferredCommands);
        AsyncGPUReadback.CompleteNext(null, true);
        Pool.ClearReusable();
        SystemInfo.supportsAsyncGPUReadback = false;
        Clean();
    }

    static void DynamicTextureCannotStarveWaitingMap()
    {
        scenarios++;
        SystemInfo.supportsAsyncGPUReadback = true;
        const int quota = 1 << 20;
        // One current RT copy, including its metadata, uses the entire allowance.
        var dynamicTexture = new RenderTexture { width = 516, height = 127, mipmapCount = 1, graphicsFormat = 42 };
        var dynamicMaterial = new Fixture(dynamicTexture);
        var stableMaterial = new Fixture(Texture(2, 2, 1));
        HybridLumenGIHitMaterialUInt4 stableHandle = default;
        for (int step = 0; step < 3; step++)
        {
            Pool.BeginFrame(2002 + step, quota);
            dynamicTexture.updateCount++;
            var batch = Prepare(dynamicMaterial.Freeze(), stableMaterial.Freeze(19));
            var command = new CommandBuffer();
            RecordBatch(batch, command);
            var table = Table(command);
            Check((table[0].textures0.y != 0) == (step != 1) && (table[1].textures0.y != 0) == (step != 0),
                $"frame {step}: waiting texture did not take its turn or a deferred texture was exposed");
            Check(Pool.UploadBytesUsed == (step == 1 ? 128 : quota), $"frame {step}: unexpected admitted copy bytes");

            HybridLumenGIRayTextureMetadata[] ready = null;
            if (step == 1)
            {
                stableHandle = table[1].textures0;
                Check(batch.GetResources(out _, out _, out var metadata, out _), "stable page metadata resource");
                ready = new HybridLumenGIRayTextureMetadata[metadata.Count];
                ready[stableHandle.x] = command.Copies.OfType<HybridLumenGIRayTextureMetadata[]>().Single()[0];
                ready[stableHandle.x].flags = 1;
            }
            if (step == 2)
                Check(batch.TextureUploadCount == 1 && table[1].textures0.x == stableHandle.x &&
                    table[1].textures0.y == stableHandle.y, "the RT resumes copying while the completed stable page stays ready");
            Finish(batch, command);
            if (ready != null) AsyncGPUReadback.CompleteNext(ready);
        }
        Pool.ClearReusable();
        SystemInfo.supportsAsyncGPUReadback = false;
        Clean();
    }
