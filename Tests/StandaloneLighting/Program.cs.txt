    // Independent double point-to-box oracle: clamp the closest point explicitly.
    static double DistanceToBox(L.Data light,R.Domain box)
    {
        double sum=0;
        for(int axis=0;axis<3;axis++)
        {
            double p=axis==0?light.positionType.x:axis==1?light.positionType.y:light.positionType.z;
            double q=p;if(q<box.Minimum[axis])q=box.Minimum[axis];else if(q>box.Maximum[axis])q=box.Maximum[axis];
            double delta=p-q;sum+=delta*delta;
        }
        return Math.Sqrt(sum);
    }
    static void RelevanceNumerics()
    {
        var domain=Box(-2,2);
        foreach(float x in new[]{0f,2f,11f,12f,12.00001f,12.1f,100f,-12f,-100f})
        {
            var light=Pack(Point(x));bool outside=R.DefinitelyOutside(light,domain);double distance=DistanceToBox(light,domain);
            if(distance<=10) Check(!outside,"independent distance oracle: intersecting sphere excluded");
            if(distance>10.01) Check(outside,"independent distance oracle: well separated sphere retained");
            if(outside)
                foreach(float y in new[]{-2f,0f,2f})foreach(float z in new[]{-2f,0f,2f})foreach(float px in new[]{-2f,0f,2f})
                {float dx=x-px,d2=MathF.Max(dx*dx+y*y+z*z,.0001f),factor=d2/100f,smooth=Math.Clamp(1-factor*factor,0,1);Check(smooth==0,"excluded light retained nonzero float attenuation at domain point");}
        }
        Check(!R.DefinitelyOutside(Pack(Point(12.00001f)),domain),"numerical boundary band must remain");
        var spot=Point(12);spot.type=LightType.Spot;Check(!R.DefinitelyOutside(Pack(spot),domain),"spot boundary sphere excluded");
        var overflow=Pack(Point(100));overflow.directionRange.w=1e30f;Check(!R.DefinitelyOutside(overflow,domain),"range-square overflow excluded");
        overflow=Pack(Point(1e30f));Check(!R.DefinitelyOutside(overflow,domain),"shader offset/dot overflow excluded");
        overflow.positionType.x=float.NaN;Check(!R.DefinitelyOutside(overflow,domain),"nonfinite source excluded");
        Check(!R.Domain.TryCreate(new(float.NaN,0,0),new(1,1,1),out _)&&!R.Domain.TryCreate(new(2,0,0),new(1,1,1),out _),"invalid domain admitted");
        Check(!domain.TryInclude(new(0,0,0),new(float.PositiveInfinity,1,1),out _),"unknown world bounds silently omitted");

        // Fixed inverse-projection fixtures, independent direct float homogeneous evaluation.
        var perspective=new Matrix4x4{m00=1,m11=1,m23=-1,m32=9.999f,m33=.001f};
        var translated=perspective;translated.m02=49.995f;translated.m03=.005f;translated.m13=.03f;
        foreach(var matrix in new[]{Matrix4x4.identity,perspective,translated})
        {
            Check(R.TryScreenDomain(matrix,out var screen),"finite screen interval rejected");
            foreach(float x in new[]{-1f,0f,1f})foreach(float y in new[]{-1f,0f,1f})foreach(float z in new[]{0f,.5f,1f})
            {
                float w=matrix[3,0]*x+matrix[3,1]*y+matrix[3,2]*z+matrix[3,3];
                for(int row=0;row<3;row++)
                {float value=(matrix[row,0]*x+matrix[row,1]*y+matrix[row,2]*z+matrix[row,3])/w;Check(value>=screen.Minimum[row]&&value<=screen.Maximum[row],"screen domain missed float reconstructed point");}
            }
        }
        var bad=perspective;bad.m33=0;Check(!R.TryScreenDomain(bad,out _),"unbounded w accepted");
        bad=perspective;bad.m00=float.NaN;Check(!R.TryScreenDomain(bad,out _),"nonfinite inverse accepted");
        Check(R.StandardMatrixMatches(perspective,perspective),"equivalent standard matrix rejected");
        bad=perspective;bad.m01=.25f;Check(!R.StandardMatrixMatches(bad,perspective),"custom projection accepted as equivalent");
        Console.WriteLine("PASS independent double distance cases, conservative float attenuation boundaries/overflow, finite screen interval and unknown-domain rejection. CPU adapter evidence only.");
    }
